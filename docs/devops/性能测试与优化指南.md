```python
from locust import HttpUser, task, between, constant_pacing

class HomeCareUser(HttpUser):
    wait_time = between(1, 3)

    def on_start(self):
        # 登录获取 Token
        response = self.client.post("/api/v1/auth/login", json={
            "username": "testuser",
            "password": "testpass",
        })
        self.token = response.json()["access_token"]
        self.headers = {"Authorization": f"Bearer {self.token}"}

    @task(3)
    def get_medicines(self):
        self.client.get("/api/v1/medicines", headers=self.headers)

    @task(2)
    def create_medicine(self):
        self.client.post(
            "/api/v1/medicines",
            json={
                "name": "测试药品",
                "dosage": "1片",
                "frequency": "每日三次",
            },
            headers=self.headers,
        )

    @task(1)
    def get_risks(self):
        self.client.get("/api/v1/risks", headers=self.headers)

    @task(1)
    def get_health_records(self):
        self.client.get("/api/v1/health-records", headers=self.headers)
```

## 3. 后端优化

### 3.1 数据库优化

```sql
-- 索引优化
CREATE INDEX idx_medicines_member_id ON medicines(member_id);
CREATE INDEX idx_medicines_name ON medicines(name);
CREATE INDEX idx_health_records_member_type_time
  ON health_records(member_id, type, measured_at DESC);

-- 查询优化
-- 避免 SELECT *
SELECT id, name, dosage FROM medicines WHERE member_id = ?;

-- 使用连接代替子查询
SELECT m.name, COUNT(r.id) as record_count
FROM medicines m
LEFT JOIN medication_records r ON r.medicine_id = m.id
WHERE m.member_id = ?
GROUP BY m.id;

-- 分页优化
SELECT * FROM medicines
WHERE member_id = ? AND id < ?
ORDER BY id DESC
LIMIT 20;
```

### 3.2 缓存优化

```python
# 热点数据缓存
@cache(ttl=300)
async def get_medicine_detail(medicine_id):
    return await db.get(Medicine, medicine_id)

# 查询结果缓存
@cache(ttl=60)
async def get_medicine_list(member_id, page):
    return await db.query(Medicine).filter(
        Medicine.member_id == member_id
    ).offset(page * 20).limit(20).all()
```

### 3.3 异步优化

```python
# 异步数据库查询
async def get_medicine_list(member_id):
    result = await db.execute(
        select(Medicine).where(Medicine.member_id == member_id)
    )
    return result.scalars().all()

# 并发请求
async def get_dashboard_data(member_id):
    medicines, records, risks = await asyncio.gather(
        get_medicines(member_id),
        get_health_records(member_id),
        get_risks(member_id),
    )
    return {
        "medicines": medicines,
        "records": records,
        "risks": risks,
    }
```
