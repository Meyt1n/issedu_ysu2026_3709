## 2. Elasticsearch 集成

### 2.1 客户端配置

```python
from elasticsearch import AsyncElasticsearch

class ElasticsearchClient:
    def __init__(self, hosts: list, api_key: str = None):
        self.client = AsyncElasticsearch(
            hosts=hosts,
            api_key=api_key,
            timeout=30,
            max_retries=3,
            retry_on_timeout=True,
            sniff_on_start=True,
            sniff_on_connection_fail=True,
        )

    async def close(self):
        await self.client.close()

    async def index_document(self, index: str, doc_id: str, document: dict):
        await self.client.index(index=index, id=doc_id, document=document)

    async def search(self, index: str, query: dict) -> dict:
        return await self.client.search(index=index, body=query)
```

### 2.2 索引管理

```python
class IndexManager:
    def __init__(self, es: AsyncElasticsearch):
        self.es = es

    async def create_index(self, index_name: str, mapping: dict, settings: dict = None):
        await self.es.indices.create(
            index=index_name,
            mappings=mapping,
            settings=settings or {},
        )

    async def delete_index(self, index_name: str):
        await self.es.indices.delete(index=index_name)

    async def reindex(self, source: str, dest: str):
        await self.es.reindex(
            body={"source": {"index": source}, "dest": {"index": dest}},
            wait_for_completion=True,
        )

    async def refresh(self, index_name: str):
        await self.es.indices.refresh(index=index_name)
```

## 4. 查询设计

### 4.1 全文搜索

```python
class SearchService:
    def __init__(self, es: AsyncElasticsearch):
        self.es = es

    async def search_medicine(self, query: str, page: int = 1, size: int = 20) -> dict:
        body = {
            "query": {
                "bool": {
                    "should": [
                        {"match": {"name": {"query": query, "boost": 3.0}}},
                        {"match": {"generic_name": {"query": query, "boost": 2.0}}},
                        {"match": {"ingredients": query}},
                        {"match": {"indications": query}},
                        {"match": {"name.pinyin": query}},
                    ],
                    "minimum_should_match": 1,
                }
            },
            "highlight": {
                "fields": {
                    "name": {},
                    "generic_name": {},
                    "indications": {},
                }
            },
            "from": (page - 1) * size,
            "size": size,
        }

        result = await self.es.search(index="medicines", body=body)
        return self._parse_result(result)
```

### 4.2 模糊搜索

```python
    async def fuzzy_search(self, query: str, index: str) -> dict:

                        "prefix_length": 1,

                }
            }
        }
        return await self.es.search(index=index, body=body)
```

### 4.3 自动补全

```python
    async def autocomplete(self, prefix: str, index: str = "medicines") -> list[str]:
        body = {
            "suggest": {
                "medicine-suggest": {
                    "prefix": prefix,
                    "completion": {
                        "field": "name.suggest",
                        "size": 10,
                        "skip_duplicates": True,
                    },
                }
            }
        }
        result = await self.es.search(index=index, body=body)
        suggestions = result.suggest["medicine-suggest"][0].options
        return [s.text for s in suggestions]
```

```python
    def personalize_query(self, query: dict, user_preferences: dict) -> dict:
        functions = []

        if "favorite_categories" in user_preferences:
            functions.append({
                "filter": {"terms": {"category": user_preferences["favorite_categories"]}},
                "weight": 3,
            })

                "query": query,
                "functions": functions,
                "boost_mode": "sum",
            }
        }
```

```python
class QueryOptimizer:
    @staticmethod
    def use_filter_instead_of_query():
        # 使用 filter 代替 query（不计算评分）
        return {
            "query": {
                "bool": {
                    "filter": [
                        {"term": {"category": "感冒药"}},
                        {"range": {"price": {"lte": 50}}},
                    ],
                    "must": [
                        {"match": {"name": "感冒"}},
                    ],
                }
            }
        }

    @staticmethod
    def avoid_deep_pagination():
        # 使用 search_after 代替 from/size
        return {
            "query": {"match_all": {}},
            "size": 20,
            "search_after": [1620000000000, "last_id"],
            "sort": [{"created_at": "desc"}, {"_id": "asc"}],
        }
```
