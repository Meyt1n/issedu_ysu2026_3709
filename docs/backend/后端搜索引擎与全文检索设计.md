# 后端搜索引擎与全文检索设计

## 1. 概述

### 1.1 设计目标

1. 搜索响应 < 500ms
2. 搜索准确率 > 90%
3. 支持中文分词

5. 支持高亮显示

## 2. Elasticsearch 集成

### 2.1 连接配置

```python
from elasticsearch import AsyncElasticsearch

class ElasticsearchClient:
    def __init__(self, hosts: list[str], api_key: str = None):
        self.client = AsyncElasticsearch(
            hosts=hosts,
            api_key=api_key,
            timeout=30,
            max_retries=3,
            retry_on_timeout=True,
        )

    async def health_check(self) -> dict:
        return await self.client.cluster.health()

    async def close(self):
        await self.client.close()
```

### 2.2 索引管理

class IndexManager:
    def __init__(self, es_client: ElasticsearchClient):

## 4. 文档操作

### 4.1 索引文档

```python
class DocumentIndexer:
    def __init__(self, es_client: ElasticsearchClient):
        self.es = es_client.client

    async def index_medicine(self, medicine: dict):
        await self.es.index(
            index='medicines',
            id=str(medicine['id']),
            document=medicine,
        )

    async def bulk_index(self, index_name: str, documents: list[dict]):
        operations = []
        for doc in documents:
            operations.append({'index': {'_index': index_name, '_id': str(doc['id'])}})
            operations.append(doc)

        await self.es.bulk(operations=operations)

    async def update_document(self, index_name: str, doc_id: str, update: dict):
        await self.es.update(
            index=index_name,
            id=doc_id,
            doc=update,
        )

    async def delete_document(self, index_name: str, doc_id: str):
        await self.es.delete(index=index_name, id=doc_id)
```

### 4.2 数据同步

```python
class DataSyncService:
    def __init__(self, db, es_indexer: DocumentIndexer):
        self.db = db
        self.indexer = es_indexer

    async def sync_medicine(self, medicine_id: str):
        medicine = await self.db.get_medicine(medicine_id)
        if medicine:
            await self.indexer.index_medicine(medicine.to_search_doc())
        else:
            await self.indexer.delete_document('medicines', medicine_id)

    async def full_sync(self, index_name: str, batch_size: int = 1000):
        offset = 0
        while True:
            batch = await self.db.get_batch(index_name, offset, batch_size)
            if not batch:
                break
            await self.indexer.bulk_index(index_name, batch)
            offset += batch_size
```

## 5. 搜索查询

### 5.1 基础搜索

```python
class SearchService:
    def __init__(self, es_client: ElasticsearchClient):
        self.es = es_client.client

    async def search_medicines(
        self,
        keyword: str,
        category: str = None,
        page: int = 1,
        size: int = 20,
        sort_by: str = 'relevance',
    ) -> dict:
        query = {
            'bool': {
                'must': [
                    {
                        'multi_match': {
                            'query': keyword,
                            'fields': ['name^3', 'generic_name^2', 'indications'],
                            'type': 'best_fields',
                            'fuzziness': 'AUTO',
                        }
                    }
                ],
                'filter': [],
            }
        }

        if category:
            query['bool']['filter'].append({'term': {'category': category}})

        sort = self._build_sort(sort_by)

        result = await self.es.search(
            index='medicines',
            query=query,
            from_=(page - 1) * size,
            size=size,
            sort=sort,
            highlight={
                'fields': {
                    'name': {},
                    'indications': {},
                }
            },
        )

        return {
            'total': result['hits']['total']['value'],
            'items': [
                {
                    **hit['_source'],
                    'highlight': hit.get('highlight', {}),
                    'score': hit['_score'],
                }
                for hit in result['hits']['hits']
            ],
            'page': page,
            'size': size,
        }

    def _build_sort(self, sort_by: str) -> list:
        if sort_by == 'price_asc':
            return [{'price': 'asc'}]
        elif sort_by == 'price_desc':
            return [{'price': 'desc'}]
        elif sort_by == 'sales':
            return [{'sales': 'desc'}]
        elif sort_by == 'rating':
            return [{'rating': 'desc'}]
        return []
```

    async def autocomplete(self, prefix: str, size: int = 10) -> list[str]:

    async def search_with_aggregations(self, keyword: str) -> dict:
        result = await self.es.search(
            index='medicines',

                    'query': keyword,

## 7. 搜索优化

### 7.1 性能优化

```python
class SearchOptimizer:
    @staticmethod
    def optimize_query(query: dict) -> dict:
        # 使用 filter 代替 must 进行精确匹配
        # 减少返回字段
        query['_source'] = ['id', 'name', 'price', 'category']
        return query

    @staticmethod
    def add_caching(query: dict) -> dict:
        query['request_cache'] = True
        return query
```
