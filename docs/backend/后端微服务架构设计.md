# 后端微服务架构设计

> 本文档是家健镜系统微服务架构的完整设计说明，覆盖服务拆分、服务通信、服务发现、网关、配置中心。

## 1. 微服务概述

### 1.1 设计目标

1. 独立部署：每个服务可独立部署和扩展
2. 技术异构：不同服务可使用不同技术栈
3. 故障隔离：单个服务故障不影响全局
4. 团队自治：每个团队负责一个服务
5. 弹性扩展：根据负载独立扩展

### 1.2 服务拆分

| 服务 | 职责 | 数据存储 |
| --- | --- | --- |
| 用户服务 | 用户认证、权限、家庭管理 | PostgreSQL |
| 药品服务 | 药品管理、用药计划 | PostgreSQL |
| 健康服务 | 体征记录、健康档案 | PostgreSQL |
| 风险服务 | 风险识别、风险评估 | PostgreSQL + Redis |
| 视觉服务 | 图片识别、OCR | GPU + Redis |
| 聊天服务 | 健康助手对话 | PostgreSQL + Vector DB |
| 通知服务 | 推送、短信、邮件 | Redis |
| 同步服务 | 多端数据同步 | PostgreSQL |
| 文件服务 | 文件存储、图片处理 | MinIO |
| 网关服务 | 路由、鉴权、限流 | - |

### 2.2 异步通信

```python
# 事件驱动通信
class EventBus:
    def __init__(self, redis):
        self.redis = redis

    async def publish(self, event):
        await self.redis.xadd(
            "events",
            {"data": json.dumps(event, default=str)},
        )

    async def subscribe(self, handler):
        while True:
            messages = await self.redis.xread(
                {"events": ">"}, count=10, block=5000
            )
            for _, message_list in messages:
                for _, data in message_list:
                    event = json.loads(data[b"data"])
                    await handler(event)
```

### 2.3 服务间调用

```python
class ServiceClient:
    def __init__(self, service_name, discovery):
        self.service_name = service_name
        self.discovery = discovery

    async def call(self, method, path, data=None):
        # 1. 服务发现
        instances = await self.discovery.get_instances(self.service_name)
        instance = self._load_balance(instances)

        # 2. 发起请求
        url = f"http://{instance.host}:{instance.port}{path}"
        async with aiohttp.ClientSession() as session:
            async with session.request(method, url, json=data) as resp:
                return await resp.json()

    def _load_balance(self, instances):
        # 轮询负载均衡
        return instances[self._counter % len(instances)]
```

## 3. 服务发现

### 3.1 服务注册

```python
class ServiceRegistry:
    def __init__(self, redis):
        self.redis = redis

    async def register(self, service_name, host, port):
        instance_id = f"{service_name}:{host}:{port}"
        await self.redis.hset(
            f"services:{service_name}",
            instance_id,
            json.dumps({"host": host, "port": port, "registered_at": time.time()}),
        )
        # 心跳
        await self.redis.expire(f"services:{service_name}", 30)

    async def deregister(self, service_name, host, port):
        instance_id = f"{service_name}:{host}:{port}"
        await self.redis.hdel(f"services:{service_name}", instance_id)

    async def get_instances(self, service_name):
        instances = await self.redis.hgetall(f"services:{service_name}")
        return [json.loads(v) for v in instances.values()]
```

### 3.2 健康检查

```python
class HealthChecker:
    def __init__(self, registry):
        self.registry = registry

    async def check(self):
        services = await self.registry.get_all_services()
        for service_name, instances in services.items():
            for instance in instances:
                try:
                    async with aiohttp.ClientSession() as session:
                        async with session.get(
                            f"http://{instance['host']}:{instance['port']}/health",
                            timeout=5,
                        ) as resp:
                            if resp.status != 200:
                                await self.registry.deregister(
                                    service_name, instance["host"], instance["port"]
                                )
                except Exception:
                    await self.registry.deregister(
                        service_name, instance["host"], instance["port"]
                    )
```

### 4.2 网关实现

```python
class APIGateway:
    def __init__(self, routes, auth, limiter, circuit_breaker):
        self.routes = routes
        self.auth = auth
        self.limiter = limiter
        self.circuit_breaker = circuit_breaker

    async def handle(self, request):
        # 1. 路由匹配
        route = self._match_route(request.path)
        if not route:
            return Response(status_code=404)

        # 2. 认证
        if route.requires_auth:
            user = await self.auth.authenticate(request)
            if not user:
                return Response(status_code=401)

        # 3. 限流
        if not await self.limiter.allow(request):
            return Response(status_code=429)

        # 4. 熔断
        if self.circuit_breaker.is_open(route.service):
            return Response(status_code=503)

        # 5. 转发请求
        try:
            response = await self._forward(request, route)
            self.circuit_breaker.record_success(route.service)
            return response
        except Exception as e:
            self.circuit_breaker.record_failure(route.service)
            raise
```

## 5. 配置中心

### 5.1 配置管理

```python
class ConfigCenter:
    def __init__(self, redis):
        self.redis = redis
        self.listeners = {}

    async def get(self, service_name, key):
        value = await self.redis.get(f"config:{service_name}:{key}")
        return json.loads(value) if value else None

    async def set(self, service_name, key, value):
        await self.redis.set(f"config:{service_name}:{key}", json.dumps(value))
        # 通知变更
        await self.redis.publish(
            f"config_changes:{service_name}",
            json.dumps({"key": key, "value": value}),
        )

    async def watch(self, service_name, callback):
        pubsub = self.redis.pubsub()
        await pubsub.subscribe(f"config_changes:{service_name}")
        async for message in pubsub.listen():
            if message["type"] == "message":
                data = json.loads(message["data"])
                await callback(data["key"], data["value"])
```
