# 后端认证与授权设计

> 本文档是家健镜系统后端认证与授权的完整设计说明，覆盖 JWT 认证、OAuth2.0、RBAC 权限、API 密钥、多因素认证。

## 1. 概述

### 1.1 设计目标

1. 安全：防止未授权访问
2. 灵活：支持多种认证方式
3. 可扩展：支持细粒度权限
4. 高性能：认证开销小
5. 可审计：权限变更可追溯

### 1.2 认证方式

| 方式 | 说明 | 适用场景 |
| --- | --- | --- |
| JWT | 无状态令牌认证 | APP、Web |
| OAuth2.0 | 第三方授权登录 | 微信、支付宝 |
| API Key | 服务间认证 | 内部服务 |
| 短信验证码 | 手机号登录 | 移动端 |
| 多因素认证 | 双重验证 | 高安全操作 |

## 2. JWT 认证

### 2.1 Token 结构

```python
# JWT 由三部分组成：header.payload.signature
# header: 算法和类型
# payload: 用户信息和声明
# signature: 签名

{
  "alg": "HS256",
  "typ": "JWT"
}
{
  "sub": "user_id_123",
  "name": "张三",
  "role": "user",
  "iat": 1693000000,
  "exp": 1693003600
}
```

### 2.2 Token 生成

```python
import jwt
from datetime import datetime, timedelta

class JWTManager:
    def __init__(self, secret_key: str, algorithm: str = "HS256"):
        self.secret_key = secret_key
        self.algorithm = algorithm
        self.access_token_expire = timedelta(hours=2)
        self.refresh_token_expire = timedelta(days=7)

    def create_access_token(self, user_id: str, role: str) -> str:
        payload = {
            "sub": user_id,
            "role": role,
            "type": "access",
            "iat": datetime.utcnow(),
            "exp": datetime.utcnow() + self.access_token_expire,
        }
        return jwt.encode(payload, self.secret_key, algorithm=self.algorithm)

    def create_refresh_token(self, user_id: str) -> str:
        payload = {
            "sub": user_id,
            "type": "refresh",
            "iat": datetime.utcnow(),
            "exp": datetime.utcnow() + self.refresh_token_expire,
        }
        return jwt.encode(payload, self.secret_key, algorithm=self.algorithm)

    def verify_token(self, token: str) -> dict:
        try:
            payload = jwt.decode(
                token,
                self.secret_key,
                algorithms=[self.algorithm],
            )
            return payload
        except jwt.ExpiredSignatureError:
            raise AuthError("Token 已过期")
        except jwt.InvalidTokenError:
            raise AuthError("无效的 Token")
```

### 2.3 Token 刷新

```python
@router.post("/auth/refresh")
async def refresh_token(refresh_token: str = Body(...)):
    try:
        payload = jwt_manager.verify_token(refresh_token)
        if payload.get("type") != "refresh":
            raise AuthError("无效的刷新令牌")

        user_id = payload["sub"]
        user = await user_service.get_user(user_id)

        # 生成新的 access token
        new_access_token = jwt_manager.create_access_token(
            user_id=user.id,
            role=user.role,
        )

        return {
            "access_token": new_access_token,
            "token_type": "bearer",
        }
    except AuthError as e:
        raise HTTPException(status_code=401, detail=str(e))
```

## 3. OAuth2.0 集成

### 3.1 授权码流程

```python
# 1. 重定向到授权服务器
@router.get("/auth/oauth/{provider}")
async def oauth_login(provider: str):
    provider_config = get_provider_config(provider)
    auth_url = f"{provider_config.auth_endpoint}?response_type=code&client_id={provider_config.client_id}&redirect_uri={provider_config.redirect_uri}&scope={provider_config.scope}"
    return RedirectResponse(auth_url)

# 2. 处理回调
@router.get("/auth/oauth/{provider}/callback")
async def oauth_callback(provider: str, code: str):
    provider_config = get_provider_config(provider)

    # 用 code 交换 token
    token_response = requests.post(
        provider_config.token_endpoint,
        data={
            "grant_type": "authorization_code",
            "code": code,
            "client_id": provider_config.client_id,
            "client_secret": provider_config.client_secret,
            "redirect_uri": provider_config.redirect_uri,
        },
    )
    tokens = token_response.json()

    # 获取用户信息
    user_info = requests.get(
        provider_config.userinfo_endpoint,
        headers={"Authorization": f"Bearer {tokens['access_token']}"},
    ).json()

    # 查找或创建用户
    user = await user_service.get_or_create_oauth_user(
        provider=provider,
        provider_user_id=user_info["sub"],
        email=user_info.get("email"),
        name=user_info.get("name"),
    )

    # 生成系统 JWT
    access_token = jwt_manager.create_access_token(user.id, user.role)
    return {"access_token": access_token, "token_type": "bearer"}
```

## 4. RBAC 权限模型

### 4.1 角色定义

```python
from enum import Enum

class Role(str, Enum):
    ADMIN = "admin"
    DOCTOR = "doctor"
    CARETAKER = "caretaker"
    USER = "user"

# 权限定义
class Permission(str, Enum):
    # 用户管理
    USER_READ = "user:read"
    USER_WRITE = "user:write"
    USER_DELETE = "user:delete"
    # 药品管理
    MEDICINE_READ = "medicine:read"
    MEDICINE_WRITE = "medicine:write"
    MEDICINE_DELETE = "medicine:delete"
    # 健康数据
    HEALTH_READ = "health:read"
    HEALTH_WRITE = "health:write"
    # 系统管理
    SYSTEM_CONFIG = "system:config"
    SYSTEM_AUDIT = "system:audit"

# 角色权限映射
ROLE_PERMISSIONS = {
    Role.ADMIN: [p for p in Permission],
    Role.DOCTOR: [
        Permission.USER_READ,
        Permission.MEDICINE_READ,
        Permission.MEDICINE_WRITE,
        Permission.HEALTH_READ,
        Permission.HEALTH_WRITE,
    ],
    Role.CARETAKER: [
        Permission.MEDICINE_READ,
        Permission.MEDICINE_WRITE,
        Permission.HEALTH_READ,
        Permission.HEALTH_WRITE,
    ],
    Role.USER: [
        Permission.MEDICINE_READ,
        Permission.MEDICINE_WRITE,
        Permission.HEALTH_READ,
        Permission.HEALTH_WRITE,
    ],
}
```

### 4.2 权限校验装饰器

```python
from functools import wraps
from fastapi import Depends, HTTPException

def require_permission(permission: Permission):
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, current_user: User = Depends(get_current_user), **kwargs):
            if permission not in ROLE_PERMISSIONS.get(current_user.role, []):
                raise HTTPException(
                    status_code=403,
                    detail="权限不足",
                )
            return await func(*args, current_user=current_user, **kwargs)
        return wrapper
    return decorator

# 使用
@router.delete("/medicines/{medicine_id}")
@require_permission(Permission.MEDICINE_DELETE)
async def delete_medicine(medicine_id: str, current_user: User):
    await medicine_service.delete(medicine_id)
    return {"message": "删除成功"}
```

### 4.3 数据权限

```python
class DataPermissionService:
    async def check_data_access(
        self,
        user: User,
        resource_type: str,
        resource_id: str,
    ) -> bool:
        # 管理员可以访问所有数据
        if user.role == Role.ADMIN:
            return True

        # 医生可以访问患者数据
        if user.role == Role.DOCTOR:
            return await self._check_doctor_access(user.id, resource_type, resource_id)

        # 看护人可以访问被看护人数据
        if user.role == Role.CARETAKER:
            return await self._check_caretaker_access(user.id, resource_type, resource_id)

        # 普通用户只能访问自己的数据
        resource = await self._get_resource(resource_type, resource_id)
        return resource.owner_id == user.id
```

## 5. API 密钥认证

### 5.1 API Key 管理

```python
import secrets
from datetime import datetime

class APIKeyService:
    async def create_api_key(
        self,
        user_id: str,
        name: str,
        scopes: list[str],
    ) -> APIKey:
        key = secrets.token_urlsafe(32)
        api_key = APIKey(
            id=secrets.token_hex(8),
            user_id=user_id,
            name=name,
            key_hash=hashlib.sha256(key.encode()).hexdigest(),
            scopes=scopes,
            created_at=datetime.utcnow(),
            last_used_at=None,
            expires_at=None,
        )
        await self._save(api_key)
        # 只返回一次明文 key
        api_key.plain_key = key
        return api_key

    async def verify_api_key(self, key: str) -> APIKey | None:
        key_hash = hashlib.sha256(key.encode()).hexdigest()
        api_key = await self._get_by_hash(key_hash)
        if api_key and api_key.is_valid():
            api_key.last_used_at = datetime.utcnow()
            await self._update(api_key)
            return api_key
        return None
```

### 5.2 API Key 中间件

```python
from fastapi import Security, HTTPException
from fastapi.security import APIKeyHeader

api_key_header = APIKeyHeader(name="X-API-Key", auto_error=False)

async def get_api_key_user(
    api_key: str | None = Security(api_key_header),
) -> User:
    if not api_key:
        raise HTTPException(status_code=401, detail="缺少 API Key")
```

## 6. 多因素认证

### 6.1 TOTP 实现

```python
import pyotp

class TOTPService:
    def generate_secret(self) -> str:
        return pyotp.random_base32()

    def get_provisioning_uri(self, secret: str, email: str) -> str:
        return pyotp.totp.TOTP(secret).provisioning_uri(
            name=email,
            issuer_name="家健镜",
        )

    def verify_code(self, secret: str, code: str) -> bool:
        totp = pyotp.TOTP(secret)
        return totp.verify(code, valid_window=1)

# 登录流程
@router.post("/auth/login/mfa")
async def mfa_login(user_id: str, code: str):
    user = await user_service.get_user(user_id)
    if not user.mfa_enabled:
```

## 7. 安全最佳实践

### 7.1 密码存储

```python
import bcrypt

class PasswordService:
    def hash_password(self, password: str) -> str:
        salt = bcrypt.gensalt(rounds=12)
        return bcrypt.hashpw(password.encode(), salt).decode()

    def verify_password(self, password: str, hashed: str) -> bool:
        return bcrypt.checkpw(password.encode(), hashed.encode())
```
