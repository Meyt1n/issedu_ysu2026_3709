# 后端国际化与多语言设计

> 本文档是家健镜系统后端国际化（i18n）与多语言支持的完整设计说明，覆盖语言检测、消息翻译、日期时间格式化、数据库多语言、API 多语言。面向后端开发者，作为国际化实现的权威依据。

## 1. 国际化概述

### 1.1 设计目标

1. **多语言支持**：支持中文、英文等多种语言
2. **自动检测**：根据请求头或用户偏好自动选择语言
3. **易于扩展**：新增语言只需添加翻译文件
4. **统一管理**：所有用户可见文本集中管理
5. **回退机制**：翻译缺失时回退到默认语言

### 1.2 支持语言

| 语言代码 | 语言名称 | 地区 | 默认 |
| --- | --- | --- | --- |
| zh-CN | 简体中文 | 中国大陆 | 是 |
| zh-TW | 繁体中文 | 中国台湾 | - |
| en-US | 英语 | 美国 | - |
| ja-JP | 日语 | 日本 | - |

## 2. 语言检测

### 2.1 检测优先级

1. URL 参数 `?lang=zh-CN`
2. 用户偏好设置（数据库）
3. 请求头 `Accept-Language`
4. 默认语言 `zh-CN`

### 2.2 语言中间件

```python
from fastapi import Request
from starlette.middleware.base import BaseHTTPMiddleware

SUPPORTED_LANGUAGES = {"zh-CN", "zh-TW", "en-US", "ja-JP"}
DEFAULT_LANGUAGE = "zh-CN"

class I18nMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        language = self._detect_language(request)
        request.state.language = language

        # 设置到上下文变量
        set_current_language(language)

        response = await call_next(request)
        response.headers["Content-Language"] = language
        return response

    def _detect_language(self, request: Request) -> str:
        # 1. URL 参数
        lang_param = request.query_params.get("lang")
        if lang_param and lang_param in SUPPORTED_LANGUAGES:
            return lang_param

        # 2. 用户偏好（从 JWT 或数据库）
        user_lang = self._get_user_language(request)
        if user_lang and user_lang in SUPPORTED_LANGUAGES:
            return user_lang

        # 3. Accept-Language 头
        accept_language = request.headers.get("Accept-Language", "")
        for lang in self._parse_accept_language(accept_language):
            if lang in SUPPORTED_LANGUAGES:
                return lang

        # 4. 默认
        return DEFAULT_LANGUAGE

    def _parse_accept_language(self, header: str) -> list[str]:
        # 解析 "zh-CN,zh;q=0.9,en;q=0.8"
        languages = []
        for part in header.split(","):
            part = part.strip()
            if ";q=" in part:
                lang, q = part.split(";q=")
                languages.append((lang.strip(), float(q)))
            else:
                languages.append((part, 1.0))
        languages.sort(key=lambda x: x[1], reverse=True)
        return [lang for lang, _ in languages]
```

### 3.4 翻译服务

```python
import json
import os
from typing import Any

class TranslationService:
    def __init__(self, i18n_dir: str):
        self.translations: dict[str, dict] = {}
        self._load_translations(i18n_dir)

    def _load_translations(self, i18n_dir: str):
        for filename in os.listdir(i18n_dir):
            if filename.endswith(".json"):
                lang = filename[:-5]
                with open(os.path.join(i18n_dir, filename), "r", encoding="utf-8") as f:
                    self.translations[lang] = json.load(f)

    def translate(
        self,
        key: str,
        language: str | None = None,
        **kwargs: Any,
    ) -> str:
        if language is None:
            language = get_current_language()

        # 查找翻译
        translation = self._lookup(key, language)

        # 回退到默认语言
        if translation is None:
            translation = self._lookup(key, DEFAULT_LANGUAGE)

        # 回退到 key 本身
        if translation is None:
            return key

        # 替换占位符
        if kwargs:
            try:
                translation = translation.format(**kwargs)
            except KeyError:
                pass

        return translation

    def _lookup(self, key: str, language: str) -> str | None:
        if language not in self.translations:
            return None

        # 支持点分隔的 key，如 "medicine.name"
        keys = key.split(".")
        value = self.translations[language]
        for k in keys:
            if isinstance(value, dict) and k in value:
                value = value[k]
            else:
                return None
        return value if isinstance(value, str) else None


# 全局实例
translation_service = TranslationService(
    os.path.join(os.path.dirname(__file__), "i18n")
)

def t(key: str, **kwargs: Any) -> str:
    '''便捷翻译函数'''
    return translation_service.translate(key, **kwargs)
```

## 4. 错误消息国际化

### 4.1 异常国际化

```python
class I18nBusinessError(Exception):
    def __init__(self, code: str, message_key: str, status_code: int = 400, **kwargs):
        self.code = code
        self.message_key = message_key
        self.status_code = status_code
        self.kwargs = kwargs
        super().__init__(message_key)

@app.exception_handler(I18nBusinessError)
async def i18n_business_error_handler(request: Request, exc: I18nBusinessError):
    language = getattr(request.state, "language", DEFAULT_LANGUAGE)
    message = translation_service.translate(
        exc.message_key, language=language, **exc.kwargs
    )
    return JSONResponse(
        status_code=exc.status_code,
        content={"error": {"code": exc.code, "message": message}},
    )
```

### 5.1 日期格式化

```python
from datetime import datetime
import pytz

DATE_FORMATS = {
    "zh-CN": "%Y年%m月%d日",
    "zh-TW": "%Y年%m月%d日",
    "en-US": "%m/%d/%Y",
    "ja-JP": "%Y年%m月%d日",

TIME_FORMATS = {
    "zh-CN": "%H:%M",
    "zh-TW": "%H:%M",
    "en-US": "%I:%M %p",
    "ja-JP": "%H:%M",
}

DATETIME_FORMATS = {
    "zh-CN": "%Y年%m月%d日 %H:%M",
    "zh-TW": "%Y年%m月%d日 %H:%M",
    "en-US": "%m/%d/%Y %I:%M %p",
    "ja-JP": "%Y年%m月%d日 %H:%M",
}

def format_date(dt: datetime, language: str | None = None) -> str:
    if language is None:
        language = get_current_language()
    fmt = DATE_FORMATS.get(language, DATE_FORMATS[DEFAULT_LANGUAGE])
    return dt.strftime(fmt)

def format_datetime(dt: datetime, language: str | None = None) -> str:
    if language is None:
        language = get_current_language()
    fmt = DATETIME_FORMATS.get(language, DATETIME_FORMATS[DEFAULT_LANGUAGE])
    return dt.strftime(fmt)
```

### 5.2 相对时间

```python
def format_relative_time(dt: datetime, language: str | None = None) -> str:
    if language is None:
        language = get_current_language()

    now = datetime.now()
    delta = now - dt
    seconds = delta.total_seconds()

    templates = {
        "zh-CN": {
            "just_now": "刚刚",
            "minutes_ago": "{n}分钟前",
            "hours_ago": "{n}小时前",
            "days_ago": "{n}天前",
        },
        "en-US": {
            "just_now": "just now",
            "minutes_ago": "{n}m ago",
            "hours_ago": "{n}h ago",
            "days_ago": "{n}d ago",
        },
    }

    tpl = templates.get(language, templates[DEFAULT_LANGUAGE])

    if seconds < 60:
        return tpl["just_now"]
    elif seconds < 3600:
        return tpl["minutes_ago"].format(n=int(seconds / 60))
    elif seconds < 86400:
        return tpl["hours_ago"].format(n=int(seconds / 3600))
    else:
        return tpl["days_ago"].format(n=int(seconds / 86400))
```

```python
class MedicineTemplate(Base):
    __tablename__ = "medicine_templates"

    template_id = Column(UUID, primary_key=True)
```

## 7. API 多语言响应

### 7.1 响应序列化

```python
class I18nResponseMixin:
    def to_dict(self, language: str | None = None) -> dict:
        if language is None:
            language = get_current_language()

        data = super().to_dict()

        # 处理多语言字段
        for field in self.i18n_fields:
            value = getattr(self, field)
            if isinstance(value, dict):
                data[field] = value.get(language, value.get(DEFAULT_LANGUAGE, ""))

        # 处理日期时间
        for field in self.datetime_fields:
            value = getattr(self, field)
            if value:
                data[field] = format_datetime(value, language)

        return data
```

## 9. 翻译管理

### 9.1 翻译完整性检查

```python
def check_translation_completeness() -> dict:
    '''检查各语言翻译完整性'''
    base_lang = DEFAULT_LANGUAGE
    base_keys = _extract_keys(translation_service.translations[base_lang])

    report = {}
    for lang, translations in translation_service.translations.items():
        if lang == base_lang:
            continue
        lang_keys = _extract_keys(translations)
        missing = base_keys - lang_keys
        extra = lang_keys - base_keys
        report[lang] = {
            "total": len(base_keys),
            "translated": len(base_keys - missing),
            "missing": list(missing),
            "extra": list(extra),
            "completeness": round((len(base_keys - missing) / len(base_keys)) * 100, 2),
        }
    return report

def _extract_keys(d: dict, prefix: str = "") -> set[str]:
    keys = set()
    for k, v in d.items():
        full_key = f"{prefix}.{k}" if prefix else k
        if isinstance(v, dict):
            keys.update(_extract_keys(v, full_key))
        else:
            keys.add(full_key)
    return keys
```
