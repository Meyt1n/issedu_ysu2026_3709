## 2. 服务注册

```python
class ServiceRegistry:
    def __init__(self, etcd_client):
        self.etcd = etcd_client

    async def register(self, service_name: str, address: str, port: int, ttl: int = 30):
        key = f"/services/{service_name}/{address}:{port}"
        value = json.dumps({
            "address": address,
            "port": port,
            "registered_at": datetime.utcnow().isoformat(),
        })

        # 创建租约
        lease = await self.etcd.grant_lease(ttl)
        await self.etcd.put(key, value, lease=lease.id)

        # 定期续约
        async def keep_alive():
            while True:
                await asyncio.sleep(ttl // 2)
                await self.etcd.keep_alive(lease.id)

        asyncio.create_task(keep_alive())
```

## 3. 服务发现

```python
class ServiceDiscovery:
    def __init__(self, etcd_client):
        self.etcd = etcd_client
        self.cache = {}

    async def discover(self, service_name: str) -> list[ServiceInstance]:
        # 查缓存
        if service_name in self.cache:
            return self.cache[service_name]

        # 查 etcd
        result = await self.etcd.get_prefix(f"/services/{service_name}/")
        instances = []
        for kv in result.kvs:
            data = json.loads(kv.value.decode())
            instances.append(ServiceInstance(**data))

        self.cache[service_name] = instances
        return instances
```

## 4. 负载均衡

```python
class LoadBalancer:
    def __init__(self, strategy: str = "round_robin"):
        self.strategy = strategy
        self.counter = 0

    def select(self, instances: list[ServiceInstance]) -> ServiceInstance:
        if not instances:
            raise NoAvailableInstanceError()

        if self.strategy == "round_robin":
            instance = instances[self.counter % len(instances)]
            self.counter += 1
            return instance
        elif self.strategy == "random":
            return random.choice(instances)
        elif self.strategy == "weighted":
            return self._weighted_select(instances)
```
