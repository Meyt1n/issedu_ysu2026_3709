# 后端数据库迁移管理设计

> 本文档是家健镜系统后端数据库迁移管理的完整设计说明，覆盖迁移工具、迁移脚本、版本管理、回滚策略、数据迁移。

## 1. 概述

### 1.1 设计目标

1. 迁移可追溯
2. 支持回滚
3. 零停机迁移
4. 多环境一致
5. 自动化执行

### 1.2 迁移工具

| 工具 | 语言 | 特点 |
| --- | --- | --- |
| Alembic | Python | SQLAlchemy 官方，自动生成 |
| Flyway | Java/SQL | 简单，基于 SQL 脚本 |
| Liquibase | XML/YAML/SQL | 功能强大，多数据库 |

## 2. Alembic 配置

### 2.1 初始化

```bash
# 初始化 Alembic
alembic init alembic

# 生成迁移脚本
alembic revision --autogenerate -m "create medicines table"

# 执行迁移
alembic upgrade head

# 回滚
alembic downgrade -1
```

### 2.3 环境配置

```python
# alembic/env.py
from logging.config import fileConfig
from sqlalchemy import engine_from_config, pool
from alembic import context
from app.models import Base
from app.config import settings

config = context.config
config.set_main_option("sqlalchemy.url", settings.DATABASE_URL)

if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = Base.metadata

def run_migrations_offline():
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )
    with context.begin_transaction():
        context.run_migrations()

def run_migrations_online():
    connectable = engine_from_config(
        config.get_section(config.config_ini_section),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )
    with connectable.connect() as connection:
        context.configure(
            connection=connection,
            target_metadata=target_metadata,
        )
        with context.begin_transaction():
            context.run_migrations()

if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
```

## 3. 迁移脚本

### 3.1 创建表

```python
# alembic/versions/001_create_medicines.py
from alembic import op
import sqlalchemy as sa

revision = '001_create_medicines'
down_revision = None
branch_labels = None
depends_on = None

def upgrade():
    op.create_table(
        'medicines',
        sa.Column('id', sa.UUID(), primary_key=True),
        sa.Column('name', sa.String(255), nullable=False),
        sa.Column('dosage', sa.String(100)),
        sa.Column('frequency', sa.String(50)),
        sa.Column('member_id', sa.UUID(), sa.ForeignKey('members.id')),
        sa.Column('created_at', sa.DateTime(), server_default=sa.func.now()),
        sa.Column('updated_at', sa.DateTime(), onupdate=sa.func.now()),
    )
    op.create_index('idx_medicines_member_id', 'medicines', ['member_id'])

def downgrade():
    op.drop_index('idx_medicines_member_id', table_name='medicines')
    op.drop_table('medicines')
```

### 3.2 添加列

```python
# alembic/versions/002_add_medicine_category.py
revision = '002_add_medicine_category'
down_revision = '001_create_medicines'

def upgrade():
    op.add_column('medicines', sa.Column('category', sa.String(50)))
    op.add_column('medicines', sa.Column('description', sa.Text()))

def downgrade():
    op.drop_column('medicines', 'description')
    op.drop_column('medicines', 'category')
```

### 3.3 数据迁移

```python
# alembic/versions/003_migrate_medicine_data.py
revision = '003_migrate_medicine_data'
down_revision = '002_add_medicine_category'

def upgrade():
    # 数据迁移
    conn = op.get_bind()

    # 1. 添加新列
    op.add_column('medicines', sa.Column('type', sa.String(20)))

    # 2. 数据转换
    conn.execute(sa.text('''
        UPDATE medicines
        SET type = CASE
            WHEN dosage LIKE '%片%' THEN 'tablet'
            WHEN dosage LIKE '%胶囊%' THEN 'capsule'
            ELSE 'other'
        END
    '''))

    # 3. 设置非空约束
    op.alter_column('medicines', 'type', nullable=False)

def downgrade():
    op.drop_column('medicines', 'type')
```

### 3.4 索引迁移

```python
# alembic/versions/004_add_indexes.py
revision = '004_add_indexes'
down_revision = '003_migrate_medicine_data'

def upgrade():
    # 创建复合索引
    op.create_index(
        'idx_medicines_member_created',
        'medicines',
        ['member_id', 'created_at'],
    )

    # 创建唯一索引
    op.create_unique_constraint(
        'uq_medicines_name_member',
        'medicines',
        ['name', 'member_id'],
    )

def downgrade():
    op.drop_constraint('uq_medicines_name_member', 'medicines', type_='unique')
    op.drop_index('idx_medicines_member_created', table_name='medicines')
```

## 4. 零停机迁移

### 4.1 扩展-收缩模式

```python
# 阶段1：扩展（添加新列，双写）
# alembic/versions/005_expand_schema.py
def upgrade():
    op.add_column('medicines', sa.Column('new_dosage', sa.String(100)))

# 应用代码同时写新旧两列
# INSERT INTO medicines (dosage, new_dosage) VALUES (?, ?)

# 阶段2：数据迁移
# alembic/versions/006_migrate_data.py
def upgrade():
    conn = op.get_bind()
    conn.execute(sa.text("UPDATE medicines SET new_dosage = dosage WHERE new_dosage IS NULL"))

# 阶段3：收缩（删除旧列）
# alembic/versions/007_shrink_schema.py
def upgrade():
    op.drop_column('medicines', 'dosage')
    op.alter_column('medicines', 'new_dosage', new_column_name='dosage')
```

### 4.2 大表添加索引

```sql
-- PostgreSQL：并发创建索引，不锁表
CREATE INDEX CONCURRENTLY idx_medicines_name
ON medicines(name);

-- 注意：CONCURRENTLY 不能在事务中执行
-- Alembic 中需要特殊处理
```

```python
# Alembic 中使用并发索引
def upgrade():
    op.execute('COMMIT')  # 结束当前事务
    op.execute('CREATE INDEX CONCURRENTLY idx_name ON medicines(name)')
    op.execute('BEGIN')  # 重新开始事务
```

## 6. CI/CD 集成

### 6.1 自动迁移

```yaml
# .github/workflows/migrate.yml
name: Database Migration

on:
  push:
    branches: [master]

jobs:
  migrate:
    runs-on: ubuntu-latest
    environment: production
    steps:
      - uses: actions/checkout@v3

      - name: Setup Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.11'

      - name: Install dependencies
        run: pip install -r requirements.txt

      - name: Run migrations
        env:
          DATABASE_URL: ${{ secrets.DATABASE_URL }}
        run: alembic upgrade head

      - name: Verify migration
        run: alembic current
```

### 6.2 迁移检查

```yaml
      - name: Check for ungenerated migrations
        run: |
          alembic revision --autogenerate -m "check"
          if [ -n "$(git status --porcelain alembic/versions/)" ]; then
            echo "::error::有未生成的迁移脚本"
            exit 1
          fi
```

## 7. 迁移最佳实践

### 7.1 迁移原则

1. 每个迁移只做一件事
2. 必须实现 downgrade
3. 大表操作要考虑锁表
4. 数据迁移要分批处理
5. 迁移前备份数据
