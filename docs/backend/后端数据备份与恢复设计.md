# 后端数据备份与恢复设计

> 本文档是家健镜系统后端数据备份与恢复的完整设计说明，覆盖备份策略、备份方式、恢复流程、备份验证、灾难恢复。

## 1. 概述

### 1.1 设计目标

1. RPO < 1 小时：最多丢失 1 小时数据
2. RTO < 4 小时：4 小时内恢复服务
3. 备份完整：所有关键数据都有备份
4. 恢复可靠：备份可验证、可恢复
5. 安全合规：备份加密、访问控制

### 1.2 备份类型

| 类型 | 说明 | 频率 | 保留时间 |
| --- | --- | --- | --- |
| 全量备份 | 完整数据备份 | 每天 | 30 天 |
| 增量备份 | 上次备份后的变更 | 每小时 | 7 天 |
| 快照 | 存储快照 | 每 6 小时 | 3 天 |
| 归档 | 长期归档 | 每月 | 1 年 |

## 2. 数据库备份

### 2.1 PostgreSQL 备份

```bash
#!/bin/bash
# 全量备份脚本

BACKUP_DIR="/backup/postgresql"
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="$BACKUP_DIR/homecare_$DATE.dump"

# 创建备份
pg_dump -h localhost -U admin -F c -b -v -f "$BACKUP_FILE" homecare

# 压缩
gzip "$BACKUP_FILE"

# 加密
gpg --symmetric --cipher-algo AES256 "$BACKUP_FILE.gz"

# 上传到 S3
aws s3 cp "$BACKUP_FILE.gz.gpg" s3://homecare-backups/postgresql/

# 清理本地旧备份
find "$BACKUP_DIR" -name "*.dump.gz.gpg" -mtime +7 -delete
```

### 2.3 自动备份脚本

```python
import subprocess
import boto3
from datetime import datetime, timedelta

class PostgreSQLBackup:
    def __init__(self, config):
        self.config = config
        self.s3 = boto3.client('s3')

    def full_backup(self):
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        filename = f"homecare_full_{timestamp}.dump"
        filepath = f"/backup/{filename}"

        # 执行 pg_dump
        subprocess.run([
            'pg_dump',
            '-h', self.config['host'],
            '-U', self.config['user'],
            '-F', 'c',
            '-f', filepath,
            self.config['database'],
        ], check=True)

        # 上传 S3
        self.s3.upload_file(
            filepath,
            self.config['bucket'],
            f"postgresql/full/{filename}",
        )

        return filepath

    def cleanup_old_backups(self, days=30):
        cutoff = datetime.now() - timedelta(days=days)
        # 列出并删除旧备份
        response = self.s3.list_objects_v2(
            Bucket=self.config['bucket'],
            Prefix='postgresql/full/',
        )
        for obj in response.get('Contents', []):
            if obj['LastModified'].replace(tzinfo=None) < cutoff:
                self.s3.delete_object(
                    Bucket=self.config['bucket'],
                    Key=obj['Key'],
                )
```

## 3. 文件备份

### 3.1 用户上传文件备份

```python
class FileBackupService:
    def __init__(self, source_dir, backup_bucket):
        self.source_dir = source_dir
        self.backup_bucket = backup_bucket
        self.s3 = boto3.client('s3')

    def backup_files(self):
        for root, dirs, files in os.walk(self.source_dir):
            for filename in files:
                filepath = os.path.join(root, filename)
                relative_path = os.path.relpath(filepath, self.source_dir)

                # 检查是否需要备份（基于修改时间）
                if self._needs_backup(relative_path, filepath):
                    self.s3.upload_file(
                        filepath,
                        self.backup_bucket,
                        f"files/{relative_path}",
                    )

    def _needs_backup(self, key, filepath):
        try:
            response = self.s3.head_object(
                Bucket=self.backup_bucket,
                Key=f"files/{key}",
            )
            remote_mtime = response['LastModified'].replace(tzinfo=None)
            local_mtime = datetime.fromtimestamp(os.path.getmtime(filepath))
            return local_mtime > remote_mtime
        except self.s3.exceptions.ClientError:
            return True
```

## 4. 恢复流程

### 4.1 数据库恢复

```bash
#!/bin/bash
# 恢复脚本

BACKUP_FILE=$1

# 停止应用服务
systemctl stop homecare-backend

# 断开所有连接
psql -U admin -c "SELECT pg_terminate_backend(pid) FROM pg_stat_activity WHERE datname='homecare';"

# 删除现有数据库
psql -U admin -c "DROP DATABASE IF EXISTS homecare;"
psql -U admin -c "CREATE DATABASE homecare;"

# 恢复备份
pg_restore -h localhost -U admin -d homecare -v "$BACKUP_FILE"

# 启动应用服务
systemctl start homecare-backend
```

### 4.2 时间点恢复（PITR）

```bash
#!/bin/bash
# 基于 WAL 的时间点恢复

TARGET_TIME=$1

# 1. 恢复最近的全量备份
pg_restore -U admin -d homecare /backup/latest_full.dump

# 2. 配置 recovery.conf
cat > /var/lib/postgresql/data/recovery.conf <<EOF
restore_command = 'cp /backup/wal/%f %p'
recovery_target_time = '$TARGET_TIME'
recovery_target_action = 'promote'
EOF

# 3. 重启数据库
systemctl restart postgresql

# 4. 等待恢复完成
tail -f /var/log/postgresql/postgresql.log
```

### 4.3 恢复验证

```python
class RestoreVerifier:
    def verify(self, db_connection):
        checks = []

        # 1. 检查记录数
        result = db_connection.execute("SELECT COUNT(*) FROM users")
        user_count = result.scalar()
        checks.append(('user_count', user_count > 0))

        # 2. 检查最新数据时间
        result = db_connection.execute("SELECT MAX(created_at) FROM medicines")
        latest_date = result.scalar()
        checks.append(('data_freshness', latest_date > datetime.now() - timedelta(days=1)))

        # 3. 检查数据完整性
        result = db_connection.execute('''
            SELECT COUNT(*) FROM medicines m
            LEFT JOIN members mem ON m.member_id = mem.id
            WHERE mem.id IS NULL
''')
        orphan_count = result.scalar()
        checks.append(('referential_integrity', orphan_count == 0))

        return all(passed for _, passed in checks)
```

## 6. 备份监控

### 6.1 备份状态监控

```python
class BackupMonitor:
    def check_backup_status(self):
        status = {
            'last_full_backup': self._get_last_backup_time('full'),
            'last_incremental_backup': self._get_last_backup_time('incremental'),
            'backup_size': self._get_total_backup_size(),
            'backup_count': self._get_backup_count(),
        }

        # 检查备份是否过期
        if status['last_full_backup'] < datetime.now() - timedelta(hours=25):
            self._alert('全量备份超过 25 小时未执行')

        return status
```

class CrossRegionReplication:

        s3 = boto3.client('s3')

        s3.put_bucket_replication(
            Bucket=source_bucket,
            ReplicationConfiguration={

                    'Status': 'Enabled',
                    'Destination': {
                        'Bucket': f'arn:aws:s3:::{dest_bucket}',
                        'StorageClass': 'STANDARD_IA',

                }],
            },
        )
