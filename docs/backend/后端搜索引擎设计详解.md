# 后端搜索引擎设计详解

> 本文档是家健镜系统后端搜索引擎的完整设计说明，覆盖 Elasticsearch 集成、索引设计、查询优化、同义词、搜索排序。

## 1. 概述

### 1.1 设计目标

1. 查询延迟 < 200ms
2. 支持全文检索
3. 支持模糊匹配
4. 支持中文分词
5. 支持智能排序

### 1.2 搜索场景

| 场景 | 数据 | 特点 |
| --- | --- | --- |
| 药品搜索 | 药品库 | 名称、成分、适应症 |
| 健康知识 | 知识库 | 文章、问答 |
| 医生搜索 | 医生信息 | 姓名、科室、医院 |
| 历史记录 | 用户数据 | 用药、体征记录 |

### 2.2 索引管理

```python
class IndexManager:
    def __init__(self, es: AsyncElasticsearch):
        self.es = es

    async def create_index(self, index_name: str, mapping: dict, settings: dict = None):
        await self.es.indices.create(
            index=index_name,
            mappings=mapping,
            settings=settings or {},
        )

    async def delete_index(self, index_name: str):
        await self.es.indices.delete(index=index_name)

    async def reindex(self, source: str, dest: str):
        await self.es.reindex(
            body={"source": {"index": source}, "dest": {"index": dest}},
            wait_for_completion=True,
        )
```

## 4. 文档操作

### 4.1 文档写入

```python
class DocumentService:
    def __init__(self, es: AsyncElasticsearch):
        self.es = es

    async def index_document(self, index: str, doc_id: str, document: dict):
        await self.es.index(
            index=index,
            id=doc_id,
            document=document,
        )

    async def bulk_index(self, index: str, documents: list[dict]):
        operations = []
        for doc in documents:
            operations.append({"index": {"_index": index, "_id": doc["id"]}})
            operations.append(doc)

        await self.es.bulk(operations=operations)

    async def update_document(self, index: str, doc_id: str, update: dict):
        await self.es.update(
            index=index,
            id=doc_id,
            doc=update,
        )

    async def delete_document(self, index: str, doc_id: str):
        await self.es.delete(index=index, id=doc_id)
```

## 5. 查询设计

### 5.1 全文搜索

```python
class SearchService:
    def __init__(self, es: AsyncElasticsearch):
        self.es = es

    async def search_medicine(self, query: str, page: int = 1, size: int = 20) -> dict:
        body = {
            "query": {
                "bool": {
                    "should": [
                        {"match": {"name": {"query": query, "boost": 3.0}}},
                        {"match": {"generic_name": {"query": query, "boost": 2.0}}},
                        {"match": {"ingredients": query}},
                        {"match": {"indications": query}},
                        {"match": {"name.pinyin": query}},
                    ],
                    "minimum_should_match": 1,
                }
            },
            "highlight": {
                "fields": {
                    "name": {},
                    "generic_name": {},
                    "indications": {},
                }
            },
            "from": (page - 1) * size,
            "size": size,
        }

        result = await self.es.search(index="medicines", body=body)
        return self._parse_result(result)
```

    async def fuzzy_search(self, query: str, index: str) -> dict:

### 5.3 自动补全

```python
    async def autocomplete(self, prefix: str, index: str = "medicines") -> list[str]:
        body = {
            "suggest": {
                "medicine-suggest": {
                    "prefix": prefix,
                    "completion": {
                        "field": "name_suggest",
                        "size": 10,
                        "skip_duplicates": True,
                    },
                }
            }
        }
        result = await self.es.search(index=index, body=body)
        suggestions = result.suggest["medicine-suggest"][0].options
        return [s.text for s in suggestions]
```

class SynonymManager:
    def __init__(self, es: AsyncElasticsearch):
        self.es = es

    async def update_synonyms(self, synonyms: list[str]):
        # 更新同义词文件
        # 或使用 synonyms_set API
        await self.es.cluster.put_settings(
            body={
                "persistent": {
                    "indices.analysis.synonym": {
                        "synonyms": synonyms,
                    }
                }
            }
        )

        # 刷新索引
        await self.es.indices.refresh(index="medicines")
```

    def personalize_query(self, query: dict, user_preferences: dict) -> dict:
        # 根据用户偏好调整排序
        functions = []

        if "favorite_categories" in user_preferences:
            functions.append({
                "filter": {"terms": {"category": user_preferences["favorite_categories"]}},
                "weight": 3,
            })

        if "recent_views" in user_preferences:
            functions.append({
                "filter": {"ids": {"values": user_preferences["recent_views"]}},
                "weight": 2,
            })

        return {
            "function_score": {
                "query": query,
                "functions": functions,
                "boost_mode": "sum",
            }
        }
```
