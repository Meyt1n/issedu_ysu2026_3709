## 2. 消息队列选型

### 2.1 RabbitMQ

```python
import pika

class RabbitMQClient:
    def __init__(self, host: str, port: int, username: str, password: str):
        self.credentials = pika.PlainCredentials(username, password)
        self.parameters = pika.ConnectionParameters(
            host=host,
            port=port,
            credentials=self.credentials,
            heartbeat=60,
        )
        self.connection = None
        self.channel = None

    def connect(self):
        self.connection = pika.BlockingConnection(self.parameters)
        self.channel = self.connection.channel()

    def declare_queue(self, queue_name: str, durable: bool = True):
        self.channel.queue_declare(queue=queue_name, durable=durable)

    def publish(self, exchange: str, routing_key: str, body: str):
        self.channel.basic_publish(
            exchange=exchange,
            routing_key=routing_key,
            body=body,
            properties=pika.BasicProperties(
                delivery_mode=2,  # 持久化
                content_type='application/json',
            ),
        )

    def consume(self, queue_name: str, callback):
        self.channel.basic_qos(prefetch_count=1)
        self.channel.basic_consume(
            queue=queue_name,
            on_message_callback=callback,
        )
        self.channel.start_consuming()

    def close(self):
        if self.connection:
            self.connection.close()
```

### 2.2 Kafka

```python
from kafka import KafkaProducer, KafkaConsumer
import json

class KafkaClient:
    def __init__(self, bootstrap_servers: list):
        self.bootstrap_servers = bootstrap_servers
        self.producer = KafkaProducer(
            bootstrap_servers=bootstrap_servers,
            value_serializer=lambda v: json.dumps(v).encode('utf-8'),
            acks='all',
            retries=3,
        )

    def send(self, topic: str, message: dict, key: str = None):
        future = self.producer.send(
            topic,
            value=message,
            key=key.encode('utf-8') if key else None,
        )
        return future.get()

    def consume(self, topic: str, group_id: str):
        consumer = KafkaConsumer(
            topic,
            bootstrap_servers=self.bootstrap_servers,
            group_id=group_id,
            auto_offset_reset='earliest',
            enable_auto_commit=False,
            value_deserializer=lambda m: json.loads(m.decode('utf-8')),
        )
        return consumer
```

### 2.3 Redis Stream

```python
import redis

class RedisStreamClient:
    def __init__(self, redis_url: str):
        self.redis = redis.from_url(redis_url)

    def publish(self, stream: str, message: dict):
        return self.redis.xadd(stream, message)

    def consume(self, stream: str, group: str, consumer: str, count: int = 10):
        return self.redis.xreadgroup(
            groupname=group,
            consumername=consumer,
            streams={stream: '>'},
            count=count,
            block=5000,
        )

    def ack(self, stream: str, group: str, message_id: str):
        self.redis.xack(stream, group, message_id)
```

### 3.2 消费者

```python
class MessageConsumer:
    def __init__(self, mq_client):
        self.mq = mq_client
        self.handlers = {}

    def register_handler(self, message_type: str, handler):
        self.handlers[message_type] = handler

    async def start_consuming(self, queue_name: str):
        def callback(ch, method, properties, body):
            try:
                message = json.loads(body)
                handler = self.handlers.get(message['type'])
                if handler:
                    handler(message)
                ch.basic_ack(delivery_tag=method.delivery_tag)
            except Exception as e:
                # 处理失败，拒绝并重新入队或进入死信队列
                ch.basic_nack(delivery_tag=method.delivery_tag, requeue=False)

        self.mq.consume(queue_name, callback)
```

## 4. 消息可靠性

### 4.1 消息确认

```python
class ReliableConsumer:
    def __init__(self, mq_client):
        self.mq = mq_client

    async def consume_with_retry(self, queue_name: str, handler, max_retries: int = 3):
        def callback(ch, method, properties, body):
            message = json.loads(body)
            retry_count = message.get('retry_count', 0)

            try:
                handler(message)
                ch.basic_ack(delivery_tag=method.delivery_tag)
            except Exception as e:
                if retry_count < max_retries:
                    # 重新入队，增加重试次数
                    message['retry_count'] = retry_count + 1
                    self.mq.publish(queue_name, json.dumps(message))
                else:
                    # 超过重试次数，进入死信队列
                    self.mq.publish('dead_letter', json.dumps(message))
                ch.basic_ack(delivery_tag=method.delivery_tag)

        self.mq.consume(queue_name, callback)
```

### 4.2 幂等消费

```python
class IdempotentConsumer:
    def __init__(self, redis_client):
        self.redis = redis_client

    async def consume(self, message: dict, handler):
        message_id = message.get('message_id')
        if not message_id:
            return handler(message)

        # 检查是否已处理
        processed = await self.redis.get(f"processed:{message_id}")
        if processed:
            return  # 已处理，跳过

        # 处理消息
        result = handler(message)

        # 标记已处理
        await self.redis.setex(f"processed:{message_id}", 86400, "1")

        return result
```

### 4.3 事务消息

```python
class TransactionalMessage:
    def __init__(self, db, mq_client):
        self.db = db
        self.mq = mq_client

    async def send_with_db_operation(self, db_operation, message: dict, queue: str):
        async with self.db.transaction():
            # 1. 执行数据库操作
            await db_operation()

            # 2. 保存消息到本地消息表
            message_id = str(uuid.uuid4())
            await self.db.execute(
                '''INSERT INTO outbox_messages (id, queue, message, status)
                   VALUES ($1, $2, $3, 'pending')''',
                message_id, queue, json.dumps(message),
            )

        # 3. 异步发送消息（由后台任务轮询 outbox 表）
        return message_id
```

## 5. 死信队列

### 5.1 死信配置

```python
class DeadLetterQueue:
    def __init__(self, mq_client):
        self.mq = mq_client

    def setup(self, main_queue: str, dlq_queue: str):
        # 声明死信交换机
        self.mq.channel.exchange_declare(exchange='dlx', exchange_type='direct')

        # 声明死信队列
        self.mq.channel.queue_declare(queue=dlq_queue, durable=True)
        self.mq.channel.queue_bind(queue=dlq_queue, exchange='dlx', routing_key=dlq_queue)

        # 主队列绑定死信交换机
        self.mq.channel.queue_declare(
            queue=main_queue,
            durable=True,
            arguments={
                'x-dead-letter-exchange': 'dlx',
                'x-dead-letter-routing-key': dlq_queue,
            },
        )

    async def process_dead_letters(self, dlq_queue: str):
        def callback(ch, method, properties, body):
            message = json.loads(body)
            # 记录死信
            await self._log_dead_letter(message)
            # 告警通知
            await self._alert(message)
            ch.basic_ack(delivery_tag=method.delivery_tag)

        self.mq.consume(dlq_queue, callback)
```

## 6. 延迟队列

### 6.1 RabbitMQ 延迟队列

```python
class DelayedQueue:
    def __init__(self, mq_client):
        self.mq = mq_client

    def setup(self, queue_name: str, delay_ms: int):
        # 声明延迟队列
        self.mq.channel.queue_declare(
            queue=f"{queue_name}_delay",
            durable=True,
            arguments={
                'x-message-ttl': delay_ms,
                'x-dead-letter-exchange': '',
                'x-dead-letter-routing-key': queue_name,
            },
        )

        # 声明目标队列
        self.mq.channel.queue_declare(queue=queue_name, durable=True)

    def publish_delayed(self, queue_name: str, message: dict, delay_ms: int):
        self.mq.publish(
            exchange='',
            routing_key=f"{queue_name}_delay",
            body=json.dumps(message),
        )
```

class MedicationReminderScheduler:
    def __init__(self, delayed_queue: DelayedQueue):
        self.delayed_queue = delayed_queue

    async def schedule_reminder(self, user_id: str, medicine_id: str, remind_time: datetime):
        delay_ms = int((remind_time - datetime.utcnow()).total_seconds() * 1000)
        if delay_ms <= 0:
            return

        message = {
            'user_id': user_id,
            'medicine_id': medicine_id,
            'type': 'medication_reminder',
        }
        self.delayed_queue.publish_delayed('reminders', message, delay_ms)

## 7. 监控与运维

### 7.1 队列监控

```python
class QueueMonitor:
    def __init__(self, mq_client):
        self.mq = mq_client

    def get_queue_stats(self, queue_name: str) -> dict:
        info = self.mq.channel.queue_declare(queue=queue_name, passive=True)
        return {
            'queue': queue_name,
            'message_count': info.method.message_count,
            'consumer_count': info.method.consumer_count,
        }

    async def alert_if_queue_backlog(self, queue_name: str, threshold: int = 1000):
        stats = self.get_queue_stats(queue_name)
        if stats['message_count'] > threshold:
            await self._send_alert(f"队列 {queue_name} 积压 {stats['message_count']} 条消息")
```
