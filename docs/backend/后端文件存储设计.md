# 后端文件存储设计

> 本文档是家健镜系统文件存储的完整设计说明，覆盖存储方案、文件管理、图片处理、权限控制、备份策略。

## 1. 文件存储概述

### 1.1 设计目标

1. 安全存储：文件加密存储，防止未授权访问
2. 快速访问：CDN 加速，图片按需加载
3. 空间可控：自动清理，限制存储用量
4. 可扩展：支持本地和对象存储切换
5. 可追溯：文件操作有审计日志

### 1.2 文件类型

| 类型 | 格式 | 大小限制 | 存储方式 |
| --- | --- | --- | --- |
| 图片 | JPG/PNG/WebP | 10MB | 对象存储 + 缩略图 |
| 视频 | MP4/MOV | 100MB | 对象存储 + 抽帧 |
| 文档 | PDF/Markdown | 20MB | 对象存储 |
| 头像 | JPG/PNG | 2MB | 对象存储 + 裁剪 |
| 备份 | ZIP/TAR | 1GB | 对象存储 |

### 1.3 存储方案

| 方案 | 适用场景 | 优点 | 缺点 |
| --- | --- | --- | --- |
| 本地文件系统 | 开发/小规模 | 简单、无依赖 | 不可扩展、无高可用 |
| MinIO | 自托管 | S3 兼容、私有化 | 需要维护 |
| S3/OSS | 云部署 | 高可用、弹性 | 依赖云服务商 |

决策：开发环境用本地文件系统，生产环境用 MinIO 或 S3。

## 2. 存储抽象层

### 2.1 存储接口

```python
from abc import ABC, abstractmethod

class StorageBackend(ABC):
    @abstractmethod
    async def upload(self, file_path, data, content_type):
        pass

    @abstractmethod
    async def download(self, file_path):
        pass

    @abstractmethod
    async def delete(self, file_path):
        pass

    @abstractmethod
    async def exists(self, file_path):
        pass

    @abstractmethod
    async def get_url(self, file_path, expires=3600):
        pass
```

### 2.2 本地存储实现

```python
class LocalStorageBackend(StorageBackend):
    def __init__(self, base_path, base_url):
        self.base_path = Path(base_path)
        self.base_url = base_url
        self.base_path.mkdir(parents=True, exist_ok=True)

    async def upload(self, file_path, data, content_type):
        full_path = self.base_path / file_path
        full_path.parent.mkdir(parents=True, exist_ok=True)
        with open(full_path, "wb") as f:
            f.write(data)
        return f"{self.base_url}/{file_path}"

    async def download(self, file_path):
        full_path = self.base_path / file_path
        with open(full_path, "rb") as f:
            return f.read()

    async def delete(self, file_path):
        full_path = self.base_path / file_path
        if full_path.exists():
            full_path.unlink()

    async def exists(self, file_path):
        return (self.base_path / file_path).exists()

    async def get_url(self, file_path, expires=3600):
        return f"{self.base_url}/{file_path}"
```

### 2.3 S3/MinIO 实现

```python
from minio import Minio
from minio.error import S3Error

class S3StorageBackend(StorageBackend):
    def __init__(self, endpoint, access_key, secret_key, bucket, secure=True):
        self.client = Minio(
            endpoint, access_key=access_key,
            secret_key=secret_key, secure=secure,
        )
        self.bucket = bucket
        self._ensure_bucket()

    def _ensure_bucket(self):
        if not self.client.bucket_exists(self.bucket):
            self.client.make_bucket(self.bucket)

    async def upload(self, file_path, data, content_type):
        from io import BytesIO
        self.client.put_object(
            self.bucket, file_path,
            BytesIO(data), length=len(data),
            content_type=content_type,
        )
        return file_path

    async def download(self, file_path):
        response = self.client.get_object(self.bucket, file_path)
        return response.read()

    async def delete(self, file_path):
        self.client.remove_object(self.bucket, file_path)

    async def exists(self, file_path):
        try:
            self.client.stat_object(self.bucket, file_path)
            return True
        except S3Error:
            return False

    async def get_url(self, file_path, expires=3600):
        return self.client.presigned_get_object(
            self.bucket, file_path, expires=timedelta(seconds=expires)
        )
```

## 3. 文件管理服务

### 3.1 文件上传

```python
class FileService:
    def __init__(self, storage, db, image_processor):
        self.storage = storage
        self.db = db
        self.image_processor = image_processor

    async def upload_file(self, file, user_id, household_id, file_type):
        self._validate_file(file)
        data = await file.read()

        file_id = str(uuid.uuid4())
        ext = Path(file.filename).suffix.lower()
        file_path = f"{household_id}/{file_type}/{file_id}{ext}"

        if file_type == "image":
            data = await self.image_processor.optimize(data)
            thumbnail = await self.image_processor.make_thumbnail(data)
            await self.storage.upload(
                f"{file_path}.thumb.jpg", thumbnail, "image/jpeg"
            )

        await self.storage.upload(file_path, data, file.content_type)

        file_ref = FileReference(
            file_id=file_id,
            household_id=household_id,
            uploaded_by=user_id,
            file_name=file.filename,
            file_path=file_path,
            file_type=file_type,
            content_type=file.content_type,
            file_size=len(data),
        )
        self.db.add(file_ref)
        await self.db.commit()
        return file_ref

    def _validate_file(self, file):
        if file.size > MAX_FILE_SIZE:
            raise BusinessError("FILE_TOO_LARGE", "文件过大")
        if file.content_type not in ALLOWED_CONTENT_TYPES:
            raise BusinessError("INVALID_FILE_TYPE", "不支持的文件类型")
```

## 4. 图片处理

### 4.1 图片优化

```python
from PIL import Image
import io

class ImageProcessor:
    def __init__(self, max_width=1920, quality=85):
        self.max_width = max_width
        self.quality = quality

    async def optimize(self, data):
        img = Image.open(io.BytesIO(data))
        img = self._auto_rotate(img)

        if img.width > self.max_width:
            ratio = self.max_width / img.width
            new_height = int(img.height * ratio)
            img = img.resize((self.max_width, new_height), Image.LANCZOS)

        output = io.BytesIO()
        if img.mode in ("RGBA", "P"):
            img = img.convert("RGB")
        img.save(output, format="JPEG", quality=self.quality, optimize=True)
        return output.getvalue()

    async def make_thumbnail(self, data, size=(300, 300)):
        img = Image.open(io.BytesIO(data))
        img.thumbnail(size, Image.LANCZOS)

        output = io.BytesIO()
        if img.mode in ("RGBA", "P"):
            img = img.convert("RGB")
        img.save(output, format="JPEG", quality=80)
        return output.getvalue()

    def _auto_rotate(self, img):
        try:
            from PIL import ImageOps
            return ImageOps.exif_transpose(img)
        except Exception:
            return img
```

## 5. 存储配额

### 5.1 配额管理

```python
class StorageQuota:
    def __init__(self, db):
        self.db = db

    async def get_usage(self, household_id):
        result = await self.db.execute(
            select(
                func.sum(FileReference.file_size).label("total_size"),
                func.count(FileReference.file_id).label("file_count"),
            ).where(FileReference.household_id == household_id)
        )
        row = result.one()
        quota = 10 * 1024 * 1024 * 1024
        return {
            "total_size": row.total_size or 0,
            "file_count": row.file_count,
            "quota": quota,
            "usage_percent": round((row.total_size or 0) / quota * 100, 2),
        }

    async def check_quota(self, household_id, file_size):
        usage = await self.get_usage(household_id)
        if usage["total_size"] + file_size > usage["quota"]:
            raise BusinessError("QUOTA_EXCEEDED", "存储空间不足")
```

## 6. 备份策略

### 6.1 备份方案

| 数据 | 备份频率 | 保留时间 | 存储位置 |
| --- | --- | --- | --- |
| 数据库 | 每日全量 + WAL | 30 天 | 异地对象存储 |
| 用户文件 | 每日增量 | 90 天 | 异地对象存储 |
| 配置文件 | 每次变更 | 永久 | Git 仓库 |

### 6.2 备份脚本

```bash
#!/bin/bash
BACKUP_DIR="/backup/files"
DATE=$(date +%Y%m%d)

rsync -av --delete /data/files/ "$BACKUP_DIR/$DATE/"
aws s3 sync "$BACKUP_DIR/$DATE/" "s3://homecare-backups/files/$DATE/"
find $BACKUP_DIR -maxdepth 1 -type d -mtime +90 -exec rm -rf {} \;
```
