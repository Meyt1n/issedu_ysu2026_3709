# 后端数据同步与冲突解决

> 本文档是家健镜系统数据同步与冲突解决的完整设计说明，覆盖同步策略、增量同步、冲突检测、冲突解决、离线支持、数据一致性。面向后端开发者，作为数据同步实现的权威依据。

## 1. 同步概述

### 1.1 设计目标

1. **多设备同步**：手机、平板、家庭服务器间数据一致
2. **离线支持**：离线时可操作，联网后自动同步
3. **冲突解决**：多端同时修改时智能合并
4. **增量同步**：只同步变更部分，减少流量
5. **最终一致**：保证最终数据一致，允许短暂不一致

### 1.2 同步架构

```
移动端 A ←→ 同步服务 ←→ 移动端 B
              ↕
         家庭服务器
              ↕
           数据库
```

### 1.3 同步策略

| 策略 | 说明 | 适用场景 |
| --- | --- | --- |
| 事件驱动 | 操作产生事件，通过事件同步 | 健康事件、药品变更 |
| 增量拉取 | 客户端拉取上次同步后的变更 | 启动时、定时同步 |
| 推送通知 | 服务端推送变更通知 | 实时同步 |
| 全量同步 | 首次同步或数据修复 | 新设备、数据恢复 |

### 2.3 增量拉取

```python
class SyncService:
    async def pull_changes(
        self,
        device_id: str,
        user_id: str,
        household_id: str,
        last_sequence: int,
        batch_size: int = 100,
    ) -> SyncBatch:
        # 1. 获取游标
        cursor = await self._get_cursor(device_id, user_id, household_id)

        # 2. 拉取变更
        changes = []

        # 健康事件
        events = await self.db.execute(
            select(HealthEvent)
            .where(
                HealthEvent.household_id == household_id,
                HealthEvent.sync_sequence > last_sequence,
            )
            .order_by(HealthEvent.sync_sequence)
            .limit(batch_size)
        )
        for event in events.scalars():
            changes.append(SyncChange(
                type="event",
                operation="create",  # 事件只有 create
                data=event.to_dict(),
                sequence=event.sync_sequence,
            ))

        # 药品变更
        meds = await self.db.execute(
            select(MedicationChangeLog)
            .where(
                MedicationChangeLog.household_id == household_id,
                MedicationChangeLog.sync_sequence > last_sequence,
            )
            .order_by(MedicationChangeLog.sync_sequence)
            .limit(batch_size)
        )
        for change in meds.scalars():
            changes.append(SyncChange(
                type="medication",
                operation=change.operation,
                data=change.data,
                sequence=change.sync_sequence,
            ))

        # 3. 更新游标
        if changes:
            max_sequence = max(c.sequence for c in changes)
            cursor.last_sequence = max_sequence
            cursor.last_sync_at = datetime.now()
            await self.db.commit()

        return SyncBatch(
            changes=changes,
            has_more=len(changes) == batch_size,
            next_sequence=cursor.last_sequence,
        )
```

```python
class PushThrottler:
    '''合并短时间内的多次变更通知'''
    def __init__(self, delay_seconds: int = 5):
        self.delay = delay_seconds
        self.pending: dict[str, asyncio.TimerHandle] = {}

    async def schedule_notify(self, household_id: str, change_type: str):
        if household_id in self.pending:
            self.pending[household_id].cancel()

        self.pending[household_id] = asyncio.get_event_loop().call_later(
            self.delay,
            lambda: asyncio.create_task(self._do_notify(household_id, change_type)),
        )

    async def _do_notify(self, household_id: str, change_type: str):
        self.pending.pop(household_id, None)
        await notify_change(household_id, change_type)
```

## 4. 冲突检测

### 4.1 冲突场景

| 场景 | 说明 |
| --- | --- |
| 同时修改 | 两端同时修改同一记录 |
| 离线修改 | 一端离线修改，另一端在线修改 |
| 删除冲突 | 一端删除，另一端修改 |
| 并发创建 | 两端创建相同业务标识的记录 |

### 4.2 版本检测

```python
class ConflictDetector:
    def detect(
        self,
        local_version: int,
        server_version: int,
        local_data: dict,
        server_data: dict,
    ) -> Conflict | None:
        # 版本相同，无冲突
        if local_version == server_version:
            return None

        # 本地基于旧版本修改，可能冲突
        if local_version < server_version:
            # 检查是否修改了相同字段
            local_changes = self._get_changed_fields(local_data, server_data)
            if local_changes:
                return Conflict(
                    type="field_conflict",
                    fields=local_changes,
                    local_data=local_data,
                    server_data=server_data,
                )

        return None

    def _get_changed_fields(self, local: dict, server: dict) -> list[str]:
        changed = []
        for key in local:
            if key in server and local[key] != server[key]:
                changed.append(key)
        return changed
```

## 5. 冲突解决

### 5.1 解决策略

| 策略 | 说明 | 适用场景 |
| --- | --- | --- |
| 服务端优先 | 以服务端数据为准 | 简单场景 |
| 客户端优先 | 以客户端数据为准 | 用户主动选择 |
| 字段合并 | 不同字段自动合并 | 大部分场景 |
| 人工选择 | 让用户选择 | 重要数据 |
| 时间戳优先 | 以最新修改为准 | 默认策略 |

### 5.2 自动合并

```python
class ConflictResolver:
    def resolve(
        self,
        conflict: Conflict,
        strategy: str = "field_merge",
    ) -> ResolveResult:
        if strategy == "server_wins":
            return ResolveResult(data=conflict.server_data, resolved=True)

        if strategy == "client_wins":
            return ResolveResult(data=conflict.local_data, resolved=True)

        if strategy == "field_merge":
            return self._field_merge(conflict)

        if strategy == "timestamp":
            return self._timestamp_resolve(conflict)

        # 需要人工解决
        return ResolveResult(resolved=False, requires_user=True)

    def _field_merge(self, conflict: Conflict) -> ResolveResult:
        merged = dict(conflict.server_data)

        for field in conflict.fields:
            # 对于非关键字段，以本地为准
            if field not in ("name", "dosage", "frequency"):
                merged[field] = conflict.local_data[field]
            # 关键字段标记为需要人工解决
            else:
                return ResolveResult(
                    resolved=False,
                    requires_user=True,
                    conflict_fields=[field],
                )

        return ResolveResult(data=merged, resolved=True)
```

### 5.3 人工解决

```python
async def resolve_conflict_manually(
    conflict_id: str,
    choice: str,  # "local" / "server" / "merge"
    merged_data: dict | None = None,
    user_id: str,
    db: AsyncSession,
):
    conflict = await db.get(SyncConflict, conflict_id)

    if choice == "local":
        final_data = conflict.local_data
    elif choice == "server":
        final_data = conflict.server_data
    elif choice == "merge" and merged_data:
        final_data = merged_data
    else:
        raise BusinessError("INVALID_CHOICE", "无效的冲突解决选择")

    # 应用解决结果
    await apply_data(final_data, db)

    # 记录解决
    conflict.resolved = True
    conflict.resolved_by = user_id
    conflict.resolved_at = datetime.now()
    conflict.resolution = choice
    await db.commit()
```

## 6. 离线支持

### 6.1 离线队列

```python
class OfflineQueue:
    '''客户端离线操作队列（本地存储）'''

    def enqueue(self, operation: str, data: dict):
        queue = self._load_queue()
        queue.append({
            "operation_id": str(uuid.uuid4()),
            "operation": operation,
            "data": data,
            "timestamp": datetime.now().isoformat(),
            "status": "pending",
        })
        self._save_queue(queue)

    async def sync(self):
        queue = self._load_queue()
        for item in queue:
            if item["status"] != "pending":
                continue

            try:
                # 发送到服务端
                await self._send_operation(item)
                item["status"] = "synced"
            except Exception as e:
                item["status"] = "failed"
                item["error"] = str(e)

        self._save_queue(queue)
```

```python
async def push_offline_changes(device_id: str, changes: list[dict], db):
    results = []

    for change in changes:
        try:
            # 检查冲突
            conflict = await conflict_detector.check(change, db)
            if conflict:
                results.append({
                    "operation_id": change["operation_id"],
                    "status": "conflict",
                    "conflict": conflict,
                })
                continue

            # 应用变更
            await apply_operation(change, db)
            results.append({
                "operation_id": change["operation_id"],
                "status": "success",
            })
        except Exception as e:
            results.append({
                "operation_id": change["operation_id"],
                "status": "error",
                "error": str(e),
            })

    return results
```
