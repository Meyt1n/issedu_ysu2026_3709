# 后端数据库分库分表设计

> 本文档是家健镜系统数据库分库分表的完整设计说明，覆盖分片策略、路由规则、分布式事务、扩容方案。

## 1. 分库分表概述

### 1.1 设计目标

1. 水平扩展：突破单库性能瓶颈
2. 高可用：单库故障不影响全局
3. 负载均衡：数据均匀分布
4. 透明访问：应用层无感知
5. 平滑扩容：支持在线扩容

### 1.2 适用场景

| 表 | 数据量预估 | 分片策略 |
| --- | --- | --- |
| health_events | 亿级 | 按 household_id 哈希 |
| medication_records | 千万级 | 按 member_id 哈希 |
| vital_records | 千万级 | 按 member_id 哈希 |
| chat_messages | 亿级 | 按 conversation_id 哈希 |
| operation_logs | 亿级 | 按时间范围 |

## 2. 分片策略

### 2.1 哈希分片

```python
class HashShardingStrategy:
    def __init__(self, shard_count=16):
        self.shard_count = shard_count

    def get_shard(self, shard_key):
        hash_value = hashlib.md5(str(shard_key).encode()).hexdigest()
        shard_index = int(hash_value, 16) % self.shard_count
        return f"shard_{shard_index:02d}"

    def get_table(self, shard_key, table_name):
        shard = self.get_shard(shard_key)
        return f"{table_name}_{shard}"
```

### 2.2 范围分片

```python
class RangeShardingStrategy:
    def __init__(self, ranges):
        self.ranges = ranges  # [(start, end, shard), ...]

    def get_shard(self, value):
        for start, end, shard in self.ranges:
            if start <= value < end:
                return shard
        return self.ranges[-1][2]
```

### 2.3 查表法分片

```python
class LookupShardingStrategy:
    def __init__(self, db):
        self.db = db
        self.cache = {}

    async def get_shard(self, household_id):
        if household_id in self.cache:
            return self.cache[household_id]

        mapping = await self.db.execute(
            select(ShardMapping).where(
                ShardMapping.household_id == household_id
            )
        )
        shard = mapping.scalar_one().shard
        self.cache[household_id] = shard
        return shard
```

## 3. 路由层

### 3.1 分片路由

```python
class ShardingRouter:
    def __init__(self, strategy, databases):
        self.strategy = strategy
        self.databases = databases

    async def execute(self, operation, shard_key, *args, **kwargs):
        shard = self.strategy.get_shard(shard_key)
        db = self.databases[shard]

        async with db.session() as session:
            return await operation(session, *args, **kwargs)

    async def execute_all(self, operation, *args, **kwargs):
        results = []
        for shard, db in self.databases.items():
            async with db.session() as session:
                result = await operation(session, *args, **kwargs)
                results.extend(result)
        return results
```

### 3.2 跨分片查询

```python
class CrossShardQuery:
    def __init__(self, router):
        self.router = router

    async def query_all(self, query_func, merge_func=None):
        results = await self.router.execute_all(query_func)

        if merge_func:
            return merge_func(results)
        return results

    async def paginated_query(self, query_func, page, page_size):
        # 跨分片分页：每个分片取足够数据，在内存中排序
        all_results = await self.router.execute_all(
            lambda session: query_func(session, page * page_size, page_size)
        )

        # 排序和分页
        all_results.sort(key=lambda x: x.created_at, reverse=True)
        start = (page - 1) * page_size
        return all_results[start:start + page_size]
```

## 4. 分布式事务

### 4.1 Saga 模式

```python
class SagaOrchestrator:
    def __init__(self):
        self.steps = []

    def add_step(self, action, compensation):
        self.steps.append({"action": action, "compensation": compensation})

    async def execute(self):
        executed = []
        try:
            for step in self.steps:
                await step["action"]()
                executed.append(step)
        except Exception as e:
            # 补偿已执行的步骤
            for step in reversed(executed):
                try:
                    await step["compensation"]()
                except Exception:
                    logger.error("补偿失败")
            raise e
```

### 4.2 最终一致性

```python
class EventualConsistency:
    def __init__(self, event_bus):
        self.event_bus = event_bus

    async def write_with_event(self, operation, event):
        # 1. 执行本地事务
        result = await operation()

        # 2. 发布事件（Outbox 模式）
        await self.event_bus.publish(event)

        return result
```

## 5. 扩容方案

### 5.1 一致性哈希扩容

```python
class ConsistentHashSharding:
    def __init__(self, nodes, virtual_nodes=150):
        self.ring = {}
        self.nodes = nodes
        for node in nodes:
            for i in range(virtual_nodes):
                key = hashlib.md5(f"{node}:{i}".encode()).hexdigest()
                self.ring[int(key, 16)] = node
        self.sorted_keys = sorted(self.ring.keys())

    def get_node(self, key):
        hash_value = int(hashlib.md5(str(key).encode()).hexdigest(), 16)
        for k in self.sorted_keys:
            if k >= hash_value:
                return self.ring[k]
        return self.ring[self.sorted_keys[0]]

    def add_node(self, new_node):
        # 新增节点只影响相邻节点的数据
        pass
```

### 5.2 数据迁移

```python
class DataMigrator:
    def __init__(self, source_db, target_db, batch_size=1000):
        self.source_db = source_db
        self.target_db = target_db
        self.batch_size = batch_size

    async def migrate(self, table, shard_key, old_strategy, new_strategy):
        offset = 0
        while True:
            batch = await self.source_db.execute(
                select(table).offset(offset).limit(self.batch_size)
            )
            records = batch.scalars().all()
            if not records:
                break

            for record in records:
                old_shard = old_strategy.get_shard(getattr(record, shard_key))
                new_shard = new_strategy.get_shard(getattr(record, shard_key))

                if old_shard != new_shard:
                    await self.target_db[new_shard].merge(record)
                    await self.source_db[old_shard].delete(record)

            offset += self.batch_size
```
