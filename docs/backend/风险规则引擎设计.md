# 风险规则引擎设计

> 本文档是家健镜系统风险规则引擎的完整设计说明，覆盖规则模型、规则类型、评估流程、预算控制、去重合并、确认机制。面向后端开发者，作为风险引擎实现的权威依据。

## 1. 风险引擎概述

### 1.1 设计目标

1. **及时发现**：健康数据变化时自动评估风险
2. **准确分级**：按严重程度分级（SEVERE/HIGH/INFO/LOW）
3. **可解释**：每个风险有明确原因和证据
4. **可操作**：提供非医疗处置建议
5. **不打扰**：预算控制，避免信息过载
6. **可追溯**：风险触发和确认都有记录

### 1.2 风险等级

| 等级 | 颜色 | 含义 | 预算控制 | 示例 |
| --- | --- | --- | --- | --- |
| SEVERE | 红色 | 可能危及健康，需立即处理 | 不受预算限制 | 过敏冲突、药品过期 |
| HIGH | 橙色 | 需要关注，建议尽快处理 | 不受预算限制 | 重复成分、库存不足 |
| INFO | 蓝色 | 信息提示，了解即可 | 每日 3 条 | 天气变化建议 |
| LOW | 灰色 | 日常提醒 | 每日 5 条 | 用药知识科普 |

### 1.3 规则类型

| 规则类型 | 触发时机 | 说明 |
| --- | --- | --- |
| medication_expired | 药品变更/每日检查 | 药品已过期 |
| medication_expiring | 每日检查 | 药品 7 天内过期 |
| low_stock | 药品变更/每日检查 | 库存不足 |
| duplicate_ingredient | 药品变更 | 多种药品含相同成分 |
| allergy_conflict | 药品/过敏变更 | 药品成分与过敏冲突 |
| drug_interaction | 药品变更 | 药物相互作用 |
| missed_dose | 任务超时 | 漏服药物 |
| environment_action | 天气变化 | 环境相关健康建议 |
| abnormal_vital | 体征记录 | 生命体征异常 |
| custom | 自定义 | 用户自定义规则 |

**事件触发：**
- `medication_added` → 评估过敏冲突、重复成分、相互作用

**定时触发：**
- 每日 00:05 → 评估药品过期、临期、库存
- 每日 08:00 → 评估环境行动卡

## 3. 评估流程

### 3.1 事件触发评估

```python
class RiskEngine:
    def __init__(self, rules: list[RiskRule]):
        self.rules = rules

    async def evaluate_event(self, event: HealthEvent, db: AsyncSession):
        # 1. 找到匹配的规则
        matching_rules = [
            rule for rule in self.rules
            if rule.enabled and event.event_type in rule.trigger.event_types
        ]

        # 2. 加载成员状态
        member_state = await load_member_state(event.member_id, db)

        # 3. 逐条评估
        for rule in matching_rules:
            try:
                result = await rule.evaluate(event, member_state, db)
                if result:
                    await self._create_risk(result, event, db)
            except Exception as e:
                logger.error("规则评估失败", rule_id=rule.rule_id, error=str(e))

    async def _create_risk(self, result: RiskResult, event: HealthEvent, db: AsyncSession):
        # 1. 计算去重键
        dedup_key = self._calculate_dedup_key(result)

        # 2. 检查是否已有相同风险
        existing = await get_risk_by_dedup_key(
            event.household_id, dedup_key, db
        )

        if existing and not existing.acknowledged:
            # 3. 合并到已有风险
            existing.merged_count += 1
            existing.updated_at = datetime.now()
            await db.commit()
            return existing

        # 4. 创建新风险
        risk = RiskEvent(
            household_id=event.household_id,
            member_id=event.member_id,
            risk_level=result.risk_level,
            rule_id=result.rule_id,
            rule_version=result.rule_version,
            title=result.title,
            description=result.description,
            evidence=result.evidence,
            suggestion=result.suggestion,
            deduplication_key=dedup_key,
            budget_status=self._check_budget(result.risk_level, event.household_id),
            risk_fingerprint=self._calculate_fingerprint(result),
            valid_until=result.valid_until,
        )
        db.add(risk)
        await db.commit()

        # 5. 写入事件
        await create_risk_triggered_event(risk, event.actor_user_id, db)

        return risk
```

class ScheduledRiskEvaluator:
    async def run_daily_check(self, db: AsyncSession):
        # 每日检查所有家庭
        households = await get_all_households(db)
        for household in households:
            members = await get_household_members(household.household_id, db)

                    risk_level="SEVERE",

                        title=med.name,
                        detail=f"过期日期：{med.expiry_date}",

## 4. 规则实现

### 4.1 过敏冲突规则

```python
class AllergyConflictRule(RiskRule):
    rule_id = "allergy_conflict"
    rule_version = "1.2.0"
    risk_level = RiskLevel.SEVERE
    budget_limited = False

    async def evaluate(self, event: HealthEvent, state: MemberState, db):
        if event.event_type not in ("medication_added", "allergy_added"):
            return None

        conflicts = []
        for allergy in state.allergies:
            for med in state.medications.values():
                for ingredient in med.ingredients:
                    if self._is_match(allergy.substance, ingredient.name):
                        conflicts.append((allergy, med, ingredient))

        if not conflicts:
            return None

        allergy, med, ingredient = conflicts[0]
        return RiskResult(
            rule_id=self.rule_id,
            rule_version=self.rule_version,
            risk_level=self.risk_level,
            title=f"过敏冲突：{med.name} 含 {allergy.substance}",
            description=(
                f"{med.name} 含有成分 {ingredient.name}，"
                f"与成员的 {allergy.substance} 过敏（{allergy.severity}）冲突。"
                f"服用可能导致严重过敏反应。"
            ),
            evidence=[
                RiskEvidence(type="allergy", title=f"{allergy.substance}过敏",
                            detail=f"严重程度：{allergy.severity}"),
                RiskEvidence(type="medication", title=med.name,
                            detail=f"成分：{ingredient.name} {ingredient.amount}"),
            ],
            suggestion="建议立即停止服用该药品，并咨询医生或药师更换替代药物。",
            valid_until=None,
        )

    def _is_match(self, allergy_substance: str, ingredient: str) -> bool:
        # 简单匹配，实际可使用药品成分数据库
        return allergy_substance.lower() in ingredient.lower()
```

### 4.2 重复成分规则

```python
class DuplicateIngredientRule(RiskRule):
    rule_id = "duplicate_ingredient"
    rule_version = "1.1.0"
    risk_level = RiskLevel.HIGH
    budget_limited = False

    async def evaluate(self, event, state, db):
        if event.event_type != "medication_added":
            return None

        # 统计成分出现次数
        ingredient_count: dict[str, list[Medication]] = {}
        for med in state.medications.values():
            for ingredient in med.ingredients:
                ingredient_count.setdefault(ingredient.name, []).append(med)

        # 找重复成分
        for ingredient, meds in ingredient_count.items():
            if len(meds) > 1:
                return RiskResult(
                    rule_id=self.rule_id,
                    rule_version=self.rule_version,
                    risk_level=self.risk_level,
                    title=f"重复成分：{ingredient}",
                    description=f"以下药品都含有 {ingredient}：{', '.join(m.name for m in meds)}。同时服用可能导致过量。",
                    evidence=[
                        RiskEvidence(type="medication", title=m.name,
                                    detail=f"含 {ingredient}")
                        for m in meds
                    ],
                    suggestion="建议咨询医生或药师，确认是否需要调整用药方案。",
                )
        return None
```

## 5. 预算控制

### 5.1 为什么需要预算

- 避免信息过载（用户每天收到太多提醒会忽略）
- 严重风险始终可见，一般风险限量
- 提高用户对风险提醒的重视程度

### 5.2 预算规则

| 风险等级 | 每日预算 | 超出处理 |
| --- | --- | --- |
| SEVERE | 无限制 | 始终可见 |
| HIGH | 无限制 | 始终可见 |
| INFO | 3 条/天 | 压制，次日显示 |
| LOW | 5 条/天 | 压制，次日显示 |

### 5.3 预算实现

```python
class BudgetController:
    def __init__(self, db: AsyncSession):
        self.db = db

    async def check_budget(self, household_id: str, risk_level: RiskLevel) -> BudgetStatus:
        if risk_level in (RiskLevel.SEVERE, RiskLevel.HIGH):
            return BudgetStatus.VISIBLE

        # 统计今日已显示的同等级风险
        today_start = datetime.combine(date.today(), time.min)
        result = await self.db.execute(
            select(func.count(RiskEvent.risk_id))
            .where(
                RiskEvent.household_id == household_id,
                RiskEvent.risk_level == risk_level,
                RiskEvent.budget_status == BudgetStatus.VISIBLE,
                RiskEvent.created_at >= today_start,
            )
        )
        count = result.scalar()

        limit = 3 if risk_level == RiskLevel.INFO else 5
        if count >= limit:
            return BudgetStatus.SUPPRESSED

        return BudgetStatus.VISIBLE

    async def get_suppressed_risks(self, household_id: str) -> list[RiskEvent]:
        # 获取被压制的风险，次日自动显示
        result = await self.db.execute(
            select(RiskEvent)
            .where(
                RiskEvent.household_id == household_id,
                RiskEvent.budget_status == BudgetStatus.SUPPRESSED,
            )
            .order_by(RiskEvent.created_at)
        )
        return result.scalars().all()
```

## 6. 去重与合并

### 6.1 去重键

```python
def calculate_dedup_key(result: RiskResult, member_id: str) -> str:
    # 规则 ID + 成员 ID + 关键参数
    key_parts = [result.rule_id, member_id]

    if result.rule_id == "allergy_conflict":
        key_parts.append(result.evidence[0].title)  # 过敏原
    elif result.rule_id == "medication_expired":
        key_parts.append(result.evidence[0].title)  # 药品名
    elif result.rule_id == "duplicate_ingredient":
        key_parts.append(result.title)  # 成分名

    return hashlib.sha256("|".join(key_parts).encode()).hexdigest()
```

### 6.2 合并逻辑

- 相同去重键的风险不重复创建
- 新触发时增加 merged_count
- 更新 updated_at
- 已确认的风险不重新触发（除非规则版本变化）

## 7. 确认机制

### 7.1 确认流程

```python
async def acknowledge_risk(
    risk_id: str,
    user_id: str,
    rule_version: str,
    risk_fingerprint: str,
    note: str | None,
    db: AsyncSession,
) -> RiskEvent:
    risk = await db.get(RiskEvent, risk_id)
    if not risk:
        raise NotFoundError("RISK_NOT_FOUND", "风险不存在")

    if risk.acknowledged:
        raise BusinessError("ALREADY_ACKNOWLEDGED", "该风险已确认")

    # 验证指纹（防止确认过时风险）
    if risk.risk_fingerprint != risk_fingerprint:
        raise VersionConflictError(
            "FINGERPRINT_MISMATCH",
            "风险已变化，请刷新后重新确认",
        )

    risk.acknowledged = True
    risk.acknowledged_at = datetime.now()
    risk.acknowledged_by = user_id
    risk.updated_at = datetime.now()

    await db.commit()

    # 写入确认事件
    await create_event(
        event_type="risk_acknowledged",
        member_id=risk.member_id,
        payload={"risk_id": risk_id, "note": note},
        actor_user_id=user_id,
        db=db,
    )

    return risk
```

### 7.2 确认后行为

- 风险从活跃列表移除
- 相同去重键的风险不再触发（除非规则版本变化）
- 确认记录写入审计日志

## 8. 风险查询

### 8.1 活跃风险列表

```python
async def get_active_risks(
    household_id: str,
    member_id: str | None = None,
    risk_level: RiskLevel | None = None,
    db: AsyncSession,
) -> list[RiskEvent]:
    query = select(RiskEvent).where(
        RiskEvent.household_id == household_id,
        RiskEvent.acknowledged == False,
        RiskEvent.budget_status == BudgetStatus.VISIBLE,
    )

    if member_id:
        query = query.where(RiskEvent.member_id == member_id)
    if risk_level:
        query = query.where(RiskEvent.risk_level == risk_level)

    query = query.order_by(
        case(
            (RiskEvent.risk_level == "SEVERE", 0),
            (RiskEvent.risk_level == "HIGH", 1),
            (RiskEvent.risk_level == "INFO", 2),
            else_=3,
        ),
        RiskEvent.created_at.desc(),
    )

    result = await db.execute(query)
    return result.scalars().all()
```

### 8.2 风险历史

```python
async def get_risk_history(
    household_id: str,
    member_id: str | None = None,
    start_date: date | None = None,
    end_date: date | None = None,
    db: AsyncSession,
) -> list[RiskEvent]:
    query = select(RiskEvent).where(RiskEvent.household_id == household_id)
    if member_id:
        query = query.where(RiskEvent.member_id == member_id)
    if start_date:
        query = query.where(RiskEvent.created_at >= start_date)
    if end_date:
        query = query.where(RiskEvent.created_at <= end_date)
    query = query.order_by(RiskEvent.created_at.desc()).limit(200)

    result = await db.execute(query)
    return result.scalars().all()
```
