# 后端分布式事务设计

> 本文档是家健镜系统后端分布式事务的完整设计说明，覆盖 2PC、TCC、Saga、本地消息表、事务消息。

## 1. 概述

### 1.1 设计目标

1. 数据一致性：跨服务数据最终一致
2. 高可用：事务协调器单点故障可恢复
3. 高性能：减少分布式锁和阻塞
4. 可扩展：支持服务动态增减
5. 可观测：事务状态可追踪

### 1.2 事务方案对比

| 方案 | 一致性 | 性能 | 复杂度 | 适用场景 |
| --- | --- | --- | --- | --- |
| 2PC | 强一致 | 低 | 中 | 短事务、强一致 |
| TCC | 最终一致 | 中 | 高 | 资金类业务 |
| Saga | 最终一致 | 高 | 中 | 长事务、补偿 |
| 本地消息表 | 最终一致 | 高 | 低 | 异步场景 |
| 事务消息 | 最终一致 | 高 | 中 | 消息驱动 |

## 2. Saga 模式

### 2.1 Saga 编排

```python
class SagaOrchestrator:
    def __init__(self):
        self.steps = []
        self.compensations = []

    def add_step(self, action, compensation):
        self.steps.append(action)
        self.compensations.append(compensation)

    async def execute(self, context):
        executed = []
        try:
            for i, step in enumerate(self.steps):
                result = await step(context)
                executed.append(i)
                context[f"step_{i}_result"] = result
            return context
        except Exception as e:
            # 反向执行补偿
            for i in reversed(executed):
                try:
                    await self.compensations[i](context)
                except Exception as comp_error:
                    # 补偿失败需要人工介入
                    await self._handle_compensation_failure(i, comp_error)
            raise
```

### 2.2 用药创建 Saga

```python
class CreateMedicineSaga:
    def __init__(self, medicine_service, inventory_service, notification_service):
        self.medicine_service = medicine_service
        self.inventory_service = inventory_service
        self.notification_service = notification_service

    async def execute(self, medicine_data):
        saga = SagaOrchestrator()

        # 步骤1：创建药品
        saga.add_step(
            action=lambda ctx: self.medicine_service.create(ctx["medicine"]),
            compensation=lambda ctx: self.medicine_service.delete(ctx["medicine_id"]),
        )

        # 步骤2：扣减库存
        saga.add_step(
            action=lambda ctx: self.inventory_service.deduct(ctx["medicine_id"], ctx["quantity"]),
            compensation=lambda ctx: self.inventory_service.restore(ctx["medicine_id"], ctx["quantity"]),
        )

        # 步骤3：发送通知
        saga.add_step(
            action=lambda ctx: self.notification_service.send_medicine_created(ctx["medicine"]),
            compensation=lambda ctx: self.notification_service.cancel(ctx["notification_id"]),
        )

        context = {"medicine": medicine_data, "quantity": 1}
        return await saga.execute(context)
```

## 3. TCC 模式

### 3.1 TCC 接口

```python
from abc import ABC, abstractmethod

class TCCService(ABC):
    @abstractmethod
    async def try_(self, transaction_id: str, data: dict):
        '''预留资源'''
        pass

    @abstractmethod
    async def confirm(self, transaction_id: str):
        '''确认提交'''
        pass

    @abstractmethod
    async def cancel(self, transaction_id: str):
        '''取消回滚'''
        pass

class InventoryTCCService(TCCService):
    async def try_(self, transaction_id: str, data: dict):
        # 冻结库存
        await self.inventory_repo.freeze(
            medicine_id=data["medicine_id"],
            quantity=data["quantity"],
            transaction_id=transaction_id,
        )

    async def confirm(self, transaction_id: str):
        # 扣减冻结库存
        await self.inventory_repo.deduct_frozen(transaction_id)

    async def cancel(self, transaction_id: str):
        # 解冻库存
        await self.inventory_repo.unfreeze(transaction_id)
```

### 3.2 TCC 协调器

```python
class TCCCoordinator:
    def __init__(self, transaction_log_repo):
        self.transaction_log_repo = transaction_log_repo
        self.participants = {}

    def register_participant(self, name: str, service: TCCService):
        self.participants[name] = service

    async def execute(self, transaction_id: str, actions: dict[str, dict]):
        # 记录事务
        await self.transaction_log_repo.create(
            transaction_id=transaction_id,
            status="TRYING",
            participants=list(actions.keys()),
        )

        # Try 阶段
        try:
            for name, data in actions.items():
                await self.participants[name].try_(transaction_id, data)
        except Exception:
            # Cancel 阶段
            for name in actions:
                await self.participants[name].cancel(transaction_id)
            await self.transaction_log_repo.update_status(transaction_id, "CANCELLED")
            raise

        # Confirm 阶段
        for name in actions:
            await self.participants[name].confirm(transaction_id)

        await self.transaction_log_repo.update_status(transaction_id, "CONFIRMED")
```

## 4. 本地消息表

### 4.1 消息表设计

```sql
CREATE TABLE local_message (
    id UUID PRIMARY KEY,
    message_type VARCHAR(100) NOT NULL,
    payload JSONB NOT NULL,
    status VARCHAR(20) DEFAULT 'PENDING',
    retry_count INT DEFAULT 0,
    max_retry INT DEFAULT 5,
    next_retry_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_local_message_status ON local_message(status, next_retry_at);
```

### 4.2 事务写入

```python
class MedicineService:
    def __init__(self, db, message_repo, message_publisher):
        self.db = db
        self.message_repo = message_repo
        self.message_publisher = message_publisher

    async def create_medicine(self, medicine_data):
        async with self.db.transaction():
            # 1. 业务操作
            medicine = await self.medicine_repo.create(medicine_data)

            # 2. 写入消息表（同一个事务）
            message = LocalMessage(
                id=uuid4(),
                message_type="medicine_created",
                payload={"medicine_id": medicine.id, "name": medicine.name},
                status="PENDING",
            )
            await self.message_repo.create(message)

        return medicine
```

### 4.3 消息投递

```python
class MessagePublisher:
    def __init__(self, message_repo, mq_producer):
        self.message_repo = message_repo
        self.mq_producer = mq_producer

    async def publish_pending(self):
        messages = await self.message_repo.get_pending(limit=100)

        for message in messages:
            try:
                # 发送到 MQ
                await self.mq_producer.send(
                    topic=message.message_type,
                    payload=message.payload,
                )
                # 更新状态
                await self.message_repo.update_status(message.id, "SENT")
            except Exception as e:
                # 重试
                message.retry_count += 1
                if message.retry_count >= message.max_retry:
                    await self.message_repo.update_status(message.id, "FAILED")
                else:
                    message.next_retry_at = datetime.utcnow() + timedelta(minutes=message.retry_count)
                    await self.message_repo.update(message)
```

class TransactionMessageProducer:
    def __init__(self, producer, local_transaction_checker):
        self.producer = producer
        self.local_transaction_checker = local_transaction_checker

    async def send_in_transaction(self, topic: str, payload: dict, arg: dict):

## 6. 幂等性设计

### 6.1 幂等键

```python
class IdempotencyService:
    def __init__(self, redis_client):
        self.redis = redis_client

    async def check_and_set(self, key: str, ttl: int = 86400) -> bool:
        # 返回 True 表示首次请求，False 表示重复请求
        result = await self.redis.set(key, "1", ex=ttl, nx=True)
        return result is not None

    async def get_result(self, key: str):
        return await self.redis.get(f"result:{key}")

    async def set_result(self, key: str, result, ttl: int = 86400):
        await self.redis.set(f"result:{key}", json.dumps(result), ex=ttl)

# 使用
@router.post("/medicines")
async def create_medicine(
    request: MedicineCreateRequest,
    idempotency_key: str = Header(..., alias="Idempotency-Key"),
):
    if not await idempotency_service.check_and_set(idempotency_key):
        # 重复请求，返回之前的结果
        cached = await idempotency_service.get_result(idempotency_key)
        if cached:
            return json.loads(cached)

    result = await medicine_service.create(request)
    await idempotency_service.set_result(idempotency_key, result)
    return result
```
