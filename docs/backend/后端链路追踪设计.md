# 后端链路追踪设计

> 本文档是家健镜系统后端分布式链路追踪的完整设计说明，覆盖 Trace 模型、埋点规范、采样策略、存储查询、告警分析。

## 1. 概述

### 1.1 设计目标

1. 全链路追踪：从请求到响应完整链路
2. 低侵入：业务代码改动最小
3. 高性能：追踪开销 < 5%
4. 可采样：支持动态采样率
5. 可分析：支持慢查询、错误分析

### 1.2 核心概念

| 概念 | 说明 |
| --- | --- |
| Trace | 一次完整请求链路 |
| Span | 链路中的一个操作单元 |
| TraceId | 链路唯一标识 |
| SpanId | 操作单元唯一标识 |
| ParentSpanId | 父操作单元 ID |

## 2. Trace 模型

### 2.1 Span 结构

```python
class Span:
    def __init__(
        self,
        trace_id: str,
        span_id: str,
        parent_span_id: str | None,
        operation_name: str,
        start_time: float,
        duration: float,
        tags: dict,
        logs: list,
    ):
        self.trace_id = trace_id
        self.span_id = span_id
        self.parent_span_id = parent_span_id
        self.operation_name = operation_name
        self.start_time = start_time
        self.duration = duration
        self.tags = tags
        self.logs = logs

    def to_dict(self):
        return {
            "traceId": self.trace_id,
            "spanId": self.span_id,
            "parentSpanId": self.parent_span_id,
            "operationName": self.operation_name,
            "startTime": self.start_time,
            "duration": self.duration,
            "tags": self.tags,
            "logs": self.logs,
        }
```

### 2.2 Trace 上下文传播

```python
class TraceContext:
    def __init__(self, trace_id: str, span_id: str, sampled: bool = True):
        self.trace_id = trace_id
        self.span_id = span_id
        self.sampled = sampled

    def inject(self, headers: dict):
        headers["X-Trace-Id"] = self.trace_id
        headers["X-Span-Id"] = self.span_id
        headers["X-Sampled"] = "1" if self.sampled else "0"

    @classmethod
    def extract(cls, headers: dict) -> "TraceContext | None":
        trace_id = headers.get("X-Trace-Id")
        if not trace_id:
            return None
        return cls(
            trace_id=trace_id,
            span_id=headers.get("X-Span-Id", ""),
            sampled=headers.get("X-Sampled", "1") == "1",
        )
```

## 3. 埋点规范

### 3.1 HTTP 中间件

```python
from fastapi import Request
import time
import uuid

class TraceMiddleware:
    def __init__(self, tracer):
        self.tracer = tracer

    async def __call__(self, request: Request, call_next):
        # 提取或创建 Trace 上下文
        context = TraceContext.extract(dict(request.headers))
        if context is None:
            context = TraceContext(
                trace_id=uuid.uuid4().hex,
                span_id=uuid.uuid4().hex[:16],
            )

        # 开始 Span
        span = self.tracer.start_span(
            operation_name=f"{request.method} {request.url.path}",
            context=context,
            tags={
                "http.method": request.method,
                "http.url": str(request.url),
                "http.client_ip": request.client.host,
            },
        )

        # 处理请求
        start_time = time.time()
        response = await call_next(request)
        duration = time.time() - start_time

        # 结束 Span
        span.set_tag("http.status_code", response.status_code)
        span.finish(duration)

        # 注入 Trace ID 到响应头
        response.headers["X-Trace-Id"] = context.trace_id

        return response
```

### 3.2 数据库埋点

```python
class TracedDatabase:
    def __init__(self, db, tracer):
        self.db = db
        self.tracer = tracer

    async def execute(self, query: str, params: dict = None):
        span = self.tracer.start_span(
            operation_name="db.query",
            tags={
                "db.statement": query,
                "db.type": "postgresql",
            },
        )

        try:
            result = await self.db.execute(query, params)
            span.set_tag("db.rows", result.rowcount)
            return result
        except Exception as e:
            span.set_tag("error", True)
            span.log({"error.message": str(e)})
            raise
        finally:
            span.finish()
```

### 3.3 外部服务调用

```python
class TracedHTTPClient:
    def __init__(self, client, tracer):
        self.client = client
        self.tracer = tracer

    async def get(self, url: str, **kwargs):
        span = self.tracer.start_span(
            operation_name=f"http.get {url}",
            tags={"http.url": url, "http.method": "GET"},
        )

        # 注入 Trace 上下文
        headers = kwargs.pop("headers", {})
        self.tracer.inject(headers)

        try:
            response = await self.client.get(url, headers=headers, **kwargs)
            span.set_tag("http.status_code", response.status_code)
            return response
        except Exception as e:
            span.set_tag("error", True)
            span.log({"error.message": str(e)})
            raise
        finally:
            span.finish()
```

## 4. 采样策略

### 4.1 采样器

```python
class Sampler:
    def should_sample(self, trace_id: str, operation_name: str) -> bool:
        raise NotImplementedError

class ProbabilisticSampler(Sampler):
    def __init__(self, rate: float = 0.1):
        self.rate = rate

    def should_sample(self, trace_id: str, operation_name: str) -> bool:
        # 基于 trace_id 哈希，保证同一 trace 采样一致
        hash_value = int(trace_id, 16) % 10000
        return hash_value < self.rate * 10000

class RateLimitingSampler(Sampler):
    def __init__(self, max_traces_per_second: int = 100):
        self.max_traces_per_second = max_traces_per_second
        self._count = 0
        self._last_reset = time.time()

    def should_sample(self, trace_id: str, operation_name: str) -> bool:
        now = time.time()
        if now - self._last_reset > 1:
            self._count = 0
            self._last_reset = now

        if self._count < self.max_traces_per_second:
            self._count += 1
            return True
        return False

class AdaptiveSampler(Sampler):
    def __init__(self, target_rate: float = 0.1):
        self.target_rate = target_rate
        self._current_rate = target_rate
        self._error_rates = {}

    def should_sample(self, trace_id: str, operation_name: str) -> bool:
        # 错误请求提高采样率
        if operation_name in self._error_rates:
            if self._error_rates[operation_name] > 0.05:
                return True

        hash_value = int(trace_id, 16) % 10000
        return hash_value < self._current_rate * 10000
```

## 5. 存储与查询

### 5.1 Elasticsearch 存储

```python
class ElasticsearchSpanStorage:
    def __init__(self, es_client):
        self.es = es_client

    async def save_span(self, span: Span):
        await self.es.index(
            index=f"traces-{span.start_time.strftime('%Y.%m.%d')}",
            body=span.to_dict(),
        )

    async def query_trace(self, trace_id: str) -> list[Span]:
        result = await self.es.search(
            index="traces-*",
            body={
                "query": {"term": {"traceId": trace_id}},
                "sort": [{"startTime": "asc"}],
                "size": 1000,
            },
        )
        return [Span(**hit["_source"]) for hit in result["hits"]["hits"]]

    async def query_slow_spans(self, duration_ms: int = 1000, limit: int = 100):
        result = await self.es.search(
            index="traces-*",
            body={
                "query": {"range": {"duration": {"gte": duration_ms / 1000}}},
                "sort": [{"duration": "desc"}],
                "size": limit,
            },
        )
        return [Span(**hit["_source"]) for hit in result["hits"]["hits"]]
```

## 6. 告警分析

### 6.1 慢请求告警

```python
class SlowRequestAlert:
    def __init__(self, threshold_ms: int = 3000):
        self.threshold_ms = threshold_ms

    def check(self, span: Span):
        if span.duration * 1000 > self.threshold_ms:
            self._alert(span)

    def _alert(self, span: Span):
        alert = {
            "type": "slow_request",
            "trace_id": span.trace_id,
            "operation": span.operation_name,
            "duration_ms": span.duration * 1000,
            "timestamp": span.start_time,
        }
        # 发送告警
        AlertManager.send(alert)
```

### 6.2 错误率告警

```python
class ErrorRateAlert:
    def __init__(self, threshold: float = 0.05, window_minutes: int = 5):
        self.threshold = threshold
        self.window_minutes = window_minutes
        self._requests = []
        self._errors = []

    def record(self, span: Span):
        self._requests.append(span.start_time)
        if span.tags.get("error"):
            self._errors.append(span.start_time)

        self._cleanup()

        if len(self._requests) > 100:
            error_rate = len(self._errors) / len(self._requests)
            if error_rate > self.threshold:
                self._alert(error_rate)

    def _cleanup(self):
        cutoff = time.time() - self.window_minutes * 60
        self._requests = [t for t in self._requests if t > cutoff]
        self._errors = [t for t in self._errors if t > cutoff]

    def _alert(self, error_rate: float):
        AlertManager.send({
            "type": "high_error_rate",
            "error_rate": error_rate,
            "window_minutes": self.window_minutes,
        })
```
