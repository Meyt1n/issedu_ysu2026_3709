## 2. 配置存储

```python
class ConfigService:
    def __init__(self, redis_client, db):
        self.redis = redis_client
        self.db = db

    async def get_config(self, namespace: str, key: str) -> str:
        # 1. 查缓存
        cached = await self.redis.get(f"config:{namespace}:{key}")
        if cached:
            return cached

        # 2. 查数据库
        config = await self.db.fetchone(
            "SELECT value FROM configs WHERE namespace = ? AND key = ?",
            (namespace, key)
        )
        if config:
            await self.redis.set(f"config:{namespace}:{key}", config['value'])
            return config['value']

        raise ConfigNotFoundError(f"配置 {namespace}:{key} 不存在")
```

## 3. 动态更新

```python
class ConfigWatcher:
    def __init__(self, config_service):
        self.service = config_service
        self.listeners = {}

    def on_change(self, namespace: str, key: str, callback):
        self.listeners.setdefault(f"{namespace}:{key}", []).append(callback)

    async def watch(self):
        pubsub = self.service.redis.pubsub()
        await pubsub.subscribe("config_changes")

        async for message in pubsub.listen():
            if message['type'] == 'message':
                data = json.loads(message['data'])
                key = f"{data['namespace']}:{data['key']}"
                for callback in self.listeners.get(key, []):
                    await callback(data['value'])
```

## 4. 配置版本

```python
class ConfigVersion:
    async def publish(self, namespace: str, key: str, value: str, version: int):
        # 乐观锁
        result = await self.db.execute(
            """UPDATE configs SET value = ?, version = version + 1
               WHERE namespace = ? AND key = ? AND version = ?""",
            (value, namespace, key, version)
        )
        if result.rowcount == 0:
            raise VersionConflictError("配置版本冲突")
```
