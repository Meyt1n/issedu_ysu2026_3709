# 视觉识别后端设计

> 本文档是家健镜系统视觉识别后端的完整设计说明，覆盖任务管理、识别流程、OCR、条码识别、结果融合、复核流程、模型管理。面向后端开发者和算法工程师，作为视觉识别实现的权威依据。

## 1. 视觉识别概述

### 1.1 功能定位

视觉识别模块负责药盒/药瓶图片的自动识别，提取药品名称、规格、生产厂家等信息，辅助家庭成员快速录入药品。

### 1.2 设计原则

1. **隐私优先**：图片在家庭服务器处理，不上传公网
2. **多模态融合**：OCR + 条码 + 包装识别，提高准确率
3. **人机协同**：自动识别 + 人工复核，确保准确
4. **可追溯**：识别过程和结果都有记录
5. **渐进式**：先本地轻量模型，可升级到更精确模型

### 1.3 识别流程

```
用户拍照/上传
    ↓
客户端基础质量检查（模糊、亮度、尺寸）
    ↓
上传文件 → 创建视觉任务（QUEUED）
    ↓
Worker 领取任务 → 预处理（PREPROCESSING）
    ↓
OCR 识别文字 ──┐
条码识别       ├→ 结果融合（FUSING）
包装识别       ──┘
    ↓
结果分类：MATCHED / CONFLICT / UNKNOWN / REVIEW
    ↓
高置信度 → 自动确认（MATCHED）
低置信度/冲突 → 人工复核（REVIEW）
    ↓
确认/修正 → 生成药品事件
```

## 2. 任务管理

### 2.1 任务状态机

```
QUEUED → PREPROCESSING → INFERENCING → FUSING → MATCHED
                                               ↘ CONFLICT
                                               ↘ UNKNOWN
                                               ↘ REVIEW
任意非终态 → CANCELLED
任意步骤 → FAILED
```

| 状态 | 说明 | 可转换到 |
| --- | --- | --- |
| QUEUED | 排队中 | PREPROCESSING, CANCELLED, FAILED |
| PREPROCESSING | 预处理中 | INFERENCING, CANCELLED, FAILED |
| INFERENCING | 识别中 | FUSING, CANCELLED, FAILED |
| FUSING | 融合中 | MATCHED, CONFLICT, UNKNOWN, REVIEW, FAILED |
| MATCHED | 识别成功（高置信） | 终态 |
| CONFLICT | 结果冲突 | 终态（需人工处理） |
| UNKNOWN | 无法识别 | 终态（需手动录入） |
| REVIEW | 待复核 | 终态（人工确认/修正） |
| FAILED | 失败 | 终态 |
| CANCELLED | 已取消 | 终态 |

```python
async def update_task_status(
    task_id: str,
    new_status: str,
    expected_version: int,
    db: AsyncSession,
    **kwargs,
) -> VisionTask:
    task = await db.get(VisionTask, task_id)
    if not task:
        raise NotFoundError("TASK_NOT_FOUND", "任务不存在")

    if task.version != expected_version:
        raise VersionConflictError(
            "VERSION_CONFLICT",
            f"任务版本不匹配（当前 {task.version}，期望 {expected_version}）",
        )

    # 验证状态转换合法
    if new_status not in VALID_TRANSITIONS.get(task.status, []):
        raise BusinessError(
            "INVALID_TRANSITION",
            f"不允许从 {task.status} 转换到 {new_status}",
        )

    task.status = new_status
    task.version += 1
    for key, value in kwargs.items():
        setattr(task, key, value)
    task.updated_at = datetime.now()

    await db.commit()
    return task
```

### 2.4 任务取消

```python
async def cancel_task(
    task_id: str,
    expected_version: int,
    db: AsyncSession,
) -> VisionTask:
    task = await db.get(VisionTask, task_id)

    # 只有非终态可以取消
    if task.status in ("MATCHED", "CONFLICT", "UNKNOWN", "REVIEW", "FAILED", "CANCELLED"):
        raise BusinessError(
            "CANNOT_CANCEL",
            f"任务处于 {task.status} 状态，无法取消",
        )

    task.status = "CANCELLED"
    task.cancelled_at = datetime.now()
    task.version += 1
    await db.commit()

    return task
```

## 3. Worker 处理

### 3.1 Worker 主循环

```python
class VisionWorker:
    def __init__(self, db: AsyncSession, model_path: str | None = None):
        self.db = db
        self.running = False
        self.ocr_engine = OCREngine()
        self.barcode_engine = BarcodeEngine()
        self.package_engine = PackageRecognitionEngine(model_path)

    async def run(self):
        self.running = True
        while self.running:
            task = await self._claim_next_task()
            if not task:
                await asyncio.sleep(1)
                continue

            try:
                await self._process_task(task)
            except Exception as e:
                logger.error("视觉任务处理失败", task_id=task.task_id, error=str(e))
                await self._mark_failed(task, str(e))

    async def _claim_next_task(self) -> VisionTask | None:
        # 原子性领取任务
        async with self.db.begin():
            result = await self.db.execute(
                select(VisionTask)
                .where(VisionTask.status == "QUEUED")
                .order_by(VisionTask.created_at)
                .limit(1)
                .with_for_update(skip_locked=True)
            )
            task = result.scalar_one_or_none()
            if task:
                task.status = "PREPROCESSING"
                task.version += 1
                await self.db.flush()
            return task
```

### 3.2 处理流程

```python
    async def _process_task(self, task: VisionTask):
        # 1. 预处理
        await update_task_status(task.task_id, "PREPROCESSING", task.version, self.db)
        image = await self._preprocess(task.file_reference)

        # 2. OCR 识别
        await update_task_status(task.task_id, "INFERENCING", task.version + 1, self.db)
        ocr_result = await self.ocr_engine.recognize(image)

        # 3. 条码识别
        barcode_result = await self.barcode_engine.recognize(image)

        # 4. 包装识别（可选）
        package_result = await self.package_engine.recognize(image)

        # 5. 结果融合
        await update_task_status(task.task_id, "FUSING", task.version + 2, self.db)
        candidates = await self._fuse_results(ocr_result, barcode_result, package_result)

        # 6. 确定最终状态
        final_status = self._determine_status(candidates)
        await update_task_status(
            task.task_id,
            final_status,
            task.version + 3,
            self.db,
            candidates=candidates,
        )
```

class OCREngine:
    def __init__(self):
        # 使用 PaddleOCR 或 Tesseract
        self.engine = self._init_engine()

    def _init_engine(self):
        try:
            from paddleocr import PaddleOCR
            return PaddleOCR(use_angle_cls=True, lang="ch")
        except ImportError:
            logger.warning("PaddleOCR 未安装，使用 Tesseract")
            return None

    async def recognize(self, image: Image) -> OCRResult:
        if self.engine:
            result = self.engine.ocr(image_array, cls=True)
            return self._parse_paddle_result(result)
        else:
            # fallback 到 Tesseract
            import pytesseract
            text = pytesseract.image_to_string(image, lang="chi_sim+eng")
            return OCRResult(text=text, boxes=[])
```

class BarcodeEngine:
    async def recognize(self, image: Image) -> BarcodeResult:
        try:
            from pyzbar.pyzbar import decode
            results = decode(image)
            if results:
                return BarcodeResult(
                    code=results[0].data.decode(),
                    type=results[0].type,
                    confidence=0.99,
                )
        except ImportError:
            logger.warning("pyzbar 未安装")

        # fallback 到 OpenCV
        try:
            import cv2
            detector = cv2.QRCodeDetector()
            data, _, _ = detector.detectAndDecode(image_array)
            if data:
                return BarcodeResult(code=data, type="QR", confidence=0.9)
        except ImportError:
            pass

        return BarcodeResult(code=None, type=None, confidence=0)
```

## 6. 结果融合

### 6.1 融合策略

```python
class ResultFusion:
    async def fuse(
        self,
        ocr_result: OCRResult,
        barcode_result: BarcodeResult,
        package_result: PackageResult,
    ) -> list[VisionCandidate]:
        candidates = []

        # 1. 条码结果优先级最高
        if barcode_result.code:
            medicine = await self.medicine_db.lookup_by_barcode(barcode_result.code)
            if medicine:
                candidates.append(VisionCandidate(
                    medicine_name=medicine.name,
                    specification=medicine.specification,
                    manufacturer=medicine.manufacturer,
                    match_score=0.95,
                    evidence={"barcode": barcode_result.code, "source": "barcode"},
                    source="barcode",
                ))

        # 2. OCR 结果
        if ocr_result.medicine_names:
            for name in ocr_result.medicine_names[:3]:
                medicines = await self.medicine_db.lookup_by_name(name)
                for med in medicines[:2]:
                    score = self._calculate_ocr_score(med, ocr_result)
                    candidates.append(VisionCandidate(
                        medicine_name=med.name,
                        specification=med.specification,
                        manufacturer=med.manufacturer,
                        match_score=score,
                        evidence={"ocr_text": ocr_result.text[:200]},
                        source="ocr",
                    ))

        # 3. 包装识别结果
        if package_result.medicine_id:
            med = await self.medicine_db.lookup_by_id(package_result.medicine_id)
            if med:
                candidates.append(VisionCandidate(
                    medicine_name=med.name,
                    match_score=package_result.confidence,
                    evidence={"package_match": package_result.medicine_id},
                    source="package",
                ))

        # 4. 去重和排序
        candidates = self._deduplicate(candidates)
        candidates.sort(key=lambda c: c.match_score, reverse=True)

        return candidates[:5]  # 最多返回 5 个候选
```

### 6.2 置信度计算

```python
    def _calculate_ocr_score(self, medicine: MedicineInfo, ocr: OCRResult) -> float:
        score = 0.0

        # 名称匹配
        if medicine.name in ocr.text:
            score += 0.4

        # 规格匹配
        if medicine.specification and medicine.specification in ocr.text:
            score += 0.3

        # 厂家匹配
        if medicine.manufacturer and medicine.manufacturer in ocr.text:
            score += 0.2

        # OCR 置信度
        score += 0.1 * (ocr.confidence if hasattr(ocr, 'confidence') else 0.8)

        return min(score, 1.0)
```

### 6.3 结果分类

```python
    def _determine_status(self, candidates: list[VisionCandidate]) -> str:
        if not candidates:
            return "UNKNOWN"

        top = candidates[0]

        # 高置信度 → 自动匹配
        if top.match_score >= 0.85:
            return "MATCHED"

        # 多个候选且分数接近 → 冲突
        if len(candidates) >= 2 and (top.match_score - candidates[1].match_score) < 0.1:
            return "CONFLICT"

        # 低置信度 → 待复核
        return "REVIEW"
```

async def confirm_review(
    review_task_id: str,
    candidate_id: str,
    user_id: str,
    db: AsyncSession,
) -> ReviewTask:
    review = await db.get(ReviewTask, review_task_id)
    if review.status != "PENDING_REVIEW":
        raise BusinessError("ALREADY_COMPLETED", "复核任务已完成")

    # 找到候选
    candidate = next(
        (c for c in review.candidates if c["candidate_id"] == candidate_id),
        None,
    )
    if not candidate:
        raise NotFoundError("CANDIDATE_NOT_FOUND", "候选不存在")

    review.status = "CONFIRMED"
    review.selected_candidate_id = candidate_id
    review.completed_at = datetime.now()
    review.completed_by = user_id
    await db.commit()

    # 生成药品事件
    await create_medication_added_event(
        member_id=review.member_id,
        medication=candidate,
        actor_user_id=user_id,
        source="vision",
        source_id=review.vision_task_id,
        db=db,
    )

    return review
```

class VisionWorkerPool:
    def __init__(self, num_workers: int = 2):
        self.workers = [VisionWorker() for _ in range(num_workers)]
        self.tasks: list[asyncio.Task] = []

    async def start(self):
        for worker in self.workers:
            self.tasks.append(asyncio.create_task(worker.run()))

    async def stop(self):
        for worker in self.workers:
            worker.running = False
        await asyncio.gather(*self.tasks)

async def process_video(file_reference: str) -> list[Image]:
    # 从视频中抽帧（每秒 2 帧，最多 10 帧）
    import cv2
    cap = cv2.VideoCapture(file_path)
    frames = []
    fps = cap.get(cv2.CAP_PROP_FPS)
    frame_interval = int(fps / 2)  # 每秒 2 帧

    frame_count = 0
    while len(frames) < 10:
        ret, frame = cap.read()
        if not ret:
            break
        if frame_count % frame_interval == 0:
            frames.append(frame)
        frame_count += 1

    cap.release()
    return frames
```

## 10. 错误处理

### 10.1 错误码

| 错误码 | 说明 | 处理 |
| --- | --- | --- |
| INVALID_IMAGE | 图片格式不支持 | 提示重新拍摄 |
| IMAGE_TOO_LARGE | 图片过大 | 提示压缩 |
| IMAGE_TOO_BLURRY | 图片模糊 | 提示重新拍摄 |
| OCR_ENGINE_ERROR | OCR 引擎失败 | 重试或降级 |
| MODEL_NOT_LOADED | 模型未加载 | 使用 fallback |
| MEDICINE_NOT_FOUND | 药品库无匹配 | 提示手动录入 |
| TASK_CANCELLED | 任务已取消 | 正常结束 |

### 10.2 降级策略

- PaddleOCR 不可用 → Tesseract
- 条码识别不可用 → 只用 OCR
- 包装识别不可用 → OCR + 条码
- 所有模型不可用 → 返回 UNKNOWN，引导手动录入
