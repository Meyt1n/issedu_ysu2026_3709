# 后端搜索引擎设计

> 本文档是家健镜系统搜索引擎的完整设计说明，覆盖索引设计、查询优化、全文检索、搜索排序。

## 1. 搜索引擎概述

### 1.1 设计目标

1. 快速检索：毫秒级响应
2. 全文搜索：支持中文分词
3. 相关排序：搜索结果按相关性排序
4. 模糊匹配：支持拼音、错别字
5. 实时索引：数据变更后立即可搜

### 1.2 技术选型

| 方案 | 优点 | 缺点 | 选择 |
| --- | --- | --- | --- |
| Elasticsearch | 功能强大、生态成熟 | 资源消耗大 | 生产环境 |
| PostgreSQL FTS | 无需额外组件 | 功能有限 | 开发环境 |
| Meilisearch | 轻量、快速 | 生态较小 | 备选 |

## 3. 索引管理

### 3.1 索引创建

```python
class IndexManager:
    def __init__(self, es_client):
        self.es = es_client

    async def create_index(self, index_name, mapping):
        if not await self.es.indices.exists(index=index_name):
            await self.es.indices.create(index=index_name, body=mapping)

    async def delete_index(self, index_name):
        await self.es.indices.delete(index=index_name)

    async def rebuild_index(self, index_name, data_loader):
        # 1. 创建临时索引
        temp_index = f"{index_name}_temp"
        await self.create_index(temp_index, MEDICINE_INDEX)

        # 2. 批量导入数据
        batch = []
        async for doc in data_loader():
            batch.append({"index": {"_index": temp_index, "_id": doc["id"]}})
            batch.append(doc)
            if len(batch) >= 1000:
                await self.es.bulk(body=batch)
                batch = []
        if batch:
            await self.es.bulk(body=batch)

        # 3. 别名切换
        await self.es.indices.update_aliases(body={
            "actions": [
                {"remove": {"index": index_name, "alias": index_name}},
                {"add": {"index": temp_index, "alias": index_name}},
            ]
        })

        # 4. 删除旧索引
        await self.es.indices.delete(index=index_name)
```

### 3.2 增量同步

```python
class IndexSync:
    def __init__(self, es_client, db):
        self.es = es_client
        self.db = db

    async def on_medicine_created(self, event):
        medicine = await self.db.get(Medicine, event.medicine_id)
        await self.es.index(
            index="medicines",
            id=medicine.id,
            body=self._to_doc(medicine),
        )

    async def on_medicine_updated(self, event):
        medicine = await self.db.get(Medicine, event.medicine_id)
        await self.es.update(
            index="medicines",
            id=medicine.id,
            body={"doc": self._to_doc(medicine)},
        )

    async def on_medicine_deleted(self, event):
        await self.es.delete(index="medicines", id=event.medicine_id)
```

## 4. 查询设计

### 4.1 药品搜索

```python
class MedicineSearch:
    def __init__(self, es_client):
        self.es = es_client

    async def search(self, keyword, category=None, page=1, size=20):
        query = {
            "bool": {
                "should": [
                    {"match": {"name": {"query": keyword, "boost": 3.0}}},
                    {"match": {"generic_name": {"query": keyword, "boost": 2.0}}},
                    {"match": {"ingredients": keyword}},
                    {"match": {"indications": keyword}},
                    {"match": {"name.pinyin": keyword}},
                ],
                "minimum_should_match": 1,
            }
        }

        if category:
            query["bool"]["filter"] = [{"term": {"category": category}}]

        result = await self.es.search(
            index="medicines",
            body={
                "query": query,
                "from": (page - 1) * size,
                "size": size,
                "highlight": {
                    "fields": {
                        "name": {},
                        "generic_name": {},
                    }
                },
            },
        )

        return {
            "total": result["hits"]["total"]["value"],
            "results": [
                {
                    "id": hit["_id"],
                    "score": hit["_score"],
                    "source": hit["_source"],
                    "highlight": hit.get("highlight", {}),
                }
                for hit in result["hits"]["hits"]
            ],
        }
```

## 5. 搜索排序

### 5.1 相关性排序

```python
def build_search_query(keyword, user_context):
    return {
        "function_score": {
            "query": {
                "bool": {
                    "should": [
                        {"match": {"name": keyword}},
                        {"match": {"indications": keyword}},
                    ],
                }
            },
            "functions": [
                # 热度加权
                {"field_value_factor": {
                    "field": "view_count",
                    "modifier": "log1p",
                    "factor": 0.1,
                }},
                # 时间衰减
                {"gauss": {
                    "created_at": {
                        "origin": "now",
                        "scale": "30d",
                        "decay": 0.5,
                    }
                }},
                # 用户偏好
                {"filter": {"term": {"category": user_context.favorite_category}},
                 "weight": 1.5},
            ],
            "score_mode": "sum",
            "boost_mode": "multiply",
        }
    }
```

- 设置超时时间
- 缓存常用查询

```python
async def search_with_search_after(self, keyword, last_sort_value=None):
    body = {
        "query": {"match": {"name": keyword}},
        "size": 20,
        "sort": [{"created_at": "desc"}, {"_id": "asc"}],
    }
    if last_sort_value:
        body["search_after"] = last_sort_value
    return await self.es.search(index="medicines", body=body)
```
