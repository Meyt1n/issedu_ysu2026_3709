### 2.3 后端国际化中间件

```python
from fastapi import Request
from babel import Locale

class I18nMiddleware:
    def __init__(self, app):
        self.app = app
        self.supported_locales = ["zh-CN", "en-US"]
        self.default_locale = "zh-CN"

    async def __call__(self, request, call_next):
        # 1. 从 Header 获取语言
        accept_language = request.headers.get("Accept-Language", "")
        locale = self._parse_accept_language(accept_language)

        # 2. 从用户偏好获取（如果已登录）
        user_locale = await self._get_user_locale(request)
        if user_locale:
            locale = user_locale

        request.state.locale = locale
        response = await call_next(request)
        response.headers["Content-Language"] = locale
        return response

    def _parse_accept_language(self, header):
        for lang in header.split(","):
            lang_code = lang.split(";")[0].strip()
            if lang_code in self.supported_locales:
                return lang_code
            # 匹配前缀
            for supported in self.supported_locales:
                if supported.startswith(lang_code):
                    return supported
        return self.default_locale
```

### 2.4 翻译服务

```python
import json
from pathlib import Path

class TranslationService:
    def __init__(self, locales_dir):
        self.locales_dir = Path(locales_dir)
        self.translations = {}
        self._load_all()

    def _load_all(self):
        for locale_dir in self.locales_dir.iterdir():
            if locale_dir.is_dir():
                locale = locale_dir.name
                self.translations[locale] = {}
                for file in locale_dir.glob("*.json"):
                    with open(file, "r", encoding="utf-8") as f:
                        self.translations[locale].update(json.load(f))

    def translate(self, key, locale, **kwargs):
        locale_translations = self.translations.get(locale, {})
        default_translations = self.translations.get("zh-CN", {})

        # 查找翻译
        value = self._find_key(locale_translations, key)
        if value is None:
            value = self._find_key(default_translations, key)
        if value is None:
            return key

        # 插值
        if kwargs:
            value = value.format(**kwargs)
        return value

    def _find_key(self, translations, key):
        parts = key.split(".")
        current = translations
        for part in parts:
            if isinstance(current, dict) and part in current:
                current = current[part]
            else:
                return None
        return current
```

## 3. 时区处理

### 3.1 时区存储

- 数据库统一存储 UTC 时间
- API 返回带时区的 ISO 8601 格式
- 前端根据用户时区转换显示

```python
from datetime import datetime, timezone

def to_utc(dt):
    if dt.tzinfo is None:
        dt = dt.replace(tzinfo=timezone.utc)
    return dt.astimezone(timezone.utc)

def from_utc(dt, user_tz):
    return dt.astimezone(user_tz)
```

### 3.2 用药提醒时区

```python
import pytz

def calculate_reminder_times(schedule, user_timezone):
    tz = pytz.timezone(user_timezone)
    now = datetime.now(tz)

    reminders = []
    for time_str in schedule.times:
        hour, minute = map(int, time_str.split(":"))
        reminder_time = now.replace(
            hour=hour, minute=minute, second=0, microsecond=0
        )
        if reminder_time < now:
            reminder_time += timedelta(days=1)
        reminders.append(reminder_time.astimezone(timezone.utc))

    return reminders
```

from babel.numbers import format_decimal, format_currency

def format_number(value, locale):
    return format_decimal(value, locale=locale)
