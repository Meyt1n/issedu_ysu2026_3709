## 2. 图像预处理

### 2.1 图像增强

```python
import cv2
import numpy as np

class ImagePreprocessor:
    def __init__(self, target_size=(512, 512)):
        self.target_size = target_size

    def preprocess(self, image: np.ndarray) -> np.ndarray:
        # 1. 调整大小
        image = cv2.resize(image, self.target_size)

        # 2. 归一化
        image = image.astype(np.float32) / 255.0

        # 3. 标准化
        mean = np.array([0.485, 0.456, 0.406])
        std = np.array([0.229, 0.224, 0.225])
        image = (image - mean) / std

        return image

    def augment(self, image: np.ndarray, mask: np.ndarray = None):
        # 随机翻转
        if np.random.random() > 0.5:
            image = np.fliplr(image)
            if mask is not None:
                mask = np.fliplr(mask)

        # 随机旋转
        angle = np.random.uniform(-15, 15)
        M = cv2.getRotationMatrix2D((256, 256), angle, 1)
        image = cv2.warpAffine(image, M, (512, 512))
        if mask is not None:
            mask = cv2.warpAffine(mask, M, (512, 512))

        # 亮度调整
        brightness = np.random.uniform(0.8, 1.2)
        image = np.clip(image * brightness, 0, 1)

        return image, mask
```

### 2.2 医学图像特定处理

```python
class MedicalImagePreprocessor:
    def window_level(self, image: np.ndarray, window_center: int, window_width: int) -> np.ndarray:
        # CT 图像窗宽窗位调整
        min_visible = window_center - window_width // 2
        max_visible = window_center + window_width // 2
        image = np.clip(image, min_visible, max_visible)
        image = (image - min_visible) / (max_visible - min_visible)
        return image

    def denoise(self, image: np.ndarray) -> np.ndarray:
        # 非局部均值去噪
        return cv2.fastNlMeansDenoising(image.astype(np.uint8), h=10)

    def normalize_intensity(self, image: np.ndarray) -> np.ndarray:
        # Z-score 标准化
        mean = np.mean(image)
        std = np.std(image)
        return (image - mean) / (std + 1e-8)
```

## 3. 病灶检测

### 3.1 YOLO 检测

```python
import torch
from ultralytics import YOLO

class LesionDetector:
    def __init__(self, model_path: str):
        self.model = YOLO(model_path)
        self.class_names = ['结节', '肿块', '钙化', '空洞', '渗出']

    def detect(self, image: np.ndarray, conf_threshold: float = 0.5) -> list[dict]:
        results = self.model(image, conf=conf_threshold)

        detections = []
        for result in results:
            boxes = result.boxes
            for box in boxes:
                x1, y1, x2, y2 = box.xyxy[0].tolist()
                confidence = box.conf[0].item()
                class_id = int(box.cls[0].item())

                detections.append({
                    'class': self.class_names[class_id],
                    'confidence': confidence,
                    'bbox': [int(x1), int(y1), int(x2), int(y2)],
                    'center': [(x1 + x2) / 2, (y1 + y2) / 2],
                    'area': (x2 - x1) * (y2 - y1),
                })

        return detections
```

## 4. 图像分割

### 4.1 U-Net

```python
import torch
import torch.nn as nn

class UNet(nn.Module):
    def __init__(self, in_channels: int = 3, out_channels: int = 1):
        super().__init__()

        # 编码器
        self.enc1 = self.conv_block(in_channels, 64)
        self.enc2 = self.conv_block(64, 128)
        self.enc3 = self.conv_block(128, 256)
        self.enc4 = self.conv_block(256, 512)

        self.pool = nn.MaxPool2d(2)

        # 瓶颈
        self.bottleneck = self.conv_block(512, 1024)

        # 解码器
        self.up4 = nn.ConvTranspose2d(1024, 512, 2, stride=2)
        self.dec4 = self.conv_block(1024, 512)
        self.up3 = nn.ConvTranspose2d(512, 256, 2, stride=2)
        self.dec3 = self.conv_block(512, 256)
        self.up2 = nn.ConvTranspose2d(256, 128, 2, stride=2)
        self.dec2 = self.conv_block(256, 128)
        self.up1 = nn.ConvTranspose2d(128, 64, 2, stride=2)
        self.dec1 = self.conv_block(128, 64)

        self.final = nn.Conv2d(64, out_channels, 1)

    def conv_block(self, in_ch, out_ch):
        return nn.Sequential(
            nn.Conv2d(in_ch, out_ch, 3, padding=1),
            nn.BatchNorm2d(out_ch),
            nn.ReLU(inplace=True),
            nn.Conv2d(out_ch, out_ch, 3, padding=1),
            nn.BatchNorm2d(out_ch),
            nn.ReLU(inplace=True),
        )

    def forward(self, x):
        e1 = self.enc1(x)
        e2 = self.enc2(self.pool(e1))
        e3 = self.enc3(self.pool(e2))
        e4 = self.enc4(self.pool(e3))

        b = self.bottleneck(self.pool(e4))

        d4 = self.dec4(torch.cat([self.up4(b), e4], dim=1))
        d3 = self.dec3(torch.cat([self.up3(d4), e3], dim=1))
        d2 = self.dec2(torch.cat([self.up2(d3), e2], dim=1))
        d1 = self.dec1(torch.cat([self.up1(d2), e1], dim=1))

        return torch.sigmoid(self.final(d1))
```

### 4.2 分割后处理

```python
class SegmentationPostprocessor:
    def remove_small_objects(self, mask: np.ndarray, min_size: int = 100) -> np.ndarray:
        from scipy import ndimage
        labeled, num_features = ndimage.label(mask)
        for i in range(1, num_features + 1):
            if np.sum(labeled == i) < min_size:
                mask[labeled == i] = 0
        return mask

    def morphological_cleanup(self, mask: np.ndarray) -> np.ndarray:
        kernel = np.ones((5, 5), np.uint8)
        mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, kernel)
        mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, kernel)
        return mask

    def calculate_metrics(self, pred: np.ndarray, gt: np.ndarray) -> dict:
        intersection = np.sum(pred * gt)
        union = np.sum(pred) + np.sum(gt) - intersection
        dice = 2 * intersection / (np.sum(pred) + np.sum(gt) + 1e-8)
        iou = intersection / (union + 1e-8)
        return {'dice': dice, 'iou': iou}
```

## 5. 分类识别

### 5.1 疾病分类

```python
import timm

class DiseaseClassifier(nn.Module):
    def __init__(self, num_classes: int, model_name: str = 'efficientnet_b0'):
        super().__init__()
        self.backbone = timm.create_model(model_name, pretrained=True, num_classes=num_classes)

    def forward(self, x):
        return self.backbone(x)

class ClassificationService:
    def __init__(self, model_path: str, class_names: list[str]):
        self.model = DiseaseClassifier(len(class_names))
        self.model.load_state_dict(torch.load(model_path))
        self.model.eval()
        self.class_names = class_names

    def classify(self, image: np.ndarray) -> dict:
        tensor = self._preprocess(image)
        with torch.no_grad():
            output = self.model(tensor)
            probabilities = torch.softmax(output, dim=1)[0]

        top_k = torch.topk(probabilities, k=3)
        results = []
        for i in range(3):
            class_idx = top_k.indices[i].item()
            results.append({
                'class': self.class_names[class_idx],
                'probability': top_k.values[i].item(),
            })

        return {
            'prediction': self.class_names[torch.argmax(probabilities).item()],
            'confidence': torch.max(probabilities).item(),
            'top_k': results,
        }
```

```python
import tensorrt as trt

class TensorRTOptimizer:
    def __init__(self):
        self.logger = trt.Logger(trt.Logger.WARNING)
        self.builder = trt.Builder(self.logger)
        self.network = self.builder.create_network(1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH))
        self.parser = trt.OnnxParser(self.network, self.logger)

    def optimize(self, onnx_path: str, engine_path: str):
        with open(onnx_path, 'rb') as f:
            self.parser.parse(f.read())

        config = self.builder.create_builder_config()
        config.max_workspace_size = 1 << 30
        config.set_flag(trt.BuilderFlag.FP16)

        engine = self.builder.build_engine(self.network, config)
        with open(engine_path, 'wb') as f:
            f.write(engine.serialize())
```

## 7. 可解释性

### 7.1 Grad-CAM

```python
class GradCAM:
    def __init__(self, model: nn.Module, target_layer: str):
        self.model = model
        self.target_layer = target_layer
        self.gradients = None
        self.activations = None

        self._register_hooks()

    def _register_hooks(self):
        def forward_hook(module, input, output):
            self.activations = output

        def backward_hook(module, grad_input, grad_output):
            self.gradients = grad_output[0]

        layer = dict([*self.model.named_modules()])[self.target_layer]
        layer.register_forward_hook(forward_hook)
        layer.register_full_backward_hook(backward_hook)

    def generate(self, input_tensor: torch.Tensor, class_idx: int = None) -> np.ndarray:
        output = self.model(input_tensor)
        if class_idx is None:
            class_idx = torch.argmax(output).item()

        self.model.zero_grad()
        output[0, class_idx].backward()

        weights = torch.mean(self.gradients, dim=(2, 3), keepdim=True)
        cam = torch.sum(weights * self.activations, dim=1, keepdim=True)
        cam = torch.relu(cam)
        cam = cam / (torch.max(cam) + 1e-8)
        cam = torch.nn.functional.interpolate(cam, size=input_tensor.shape[2:], mode='bilinear')

        return cam[0, 0].cpu().numpy()
```
