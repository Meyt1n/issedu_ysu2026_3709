# 算法设计-在线学习与增量更新

> 本文档是家健镜系统机器学习在线学习与增量更新的完整设计说明，覆盖在线学习算法、增量训练、模型更新、A/B 测试、效果监控。

## 1. 概述

### 1.1 设计目标

1. 实时学习：模型能从新数据中持续学习
2. 快速更新：模型更新延迟 < 1 小时
3. 稳定可靠：更新不影响线上服务
4. 可回滚：模型更新可快速回滚
5. 效果可衡量：更新效果可量化评估

### 1.2 学习方式对比

| 方式 | 更新频率 | 数据量 | 复杂度 | 适用场景 |
| --- | --- | --- | --- | --- |
| 批量学习 | 天/周 | 全量 | 低 | 稳定模型 |
| 增量学习 | 小时 | 增量 | 中 | 数据持续增长 |
| 在线学习 | 实时 | 单条 | 高 | 实时反馈 |

## 2. 在线学习算法

### 2.1 随机梯度下降（SGD）

```python
import numpy as np

class OnlineSGD:
    def __init__(self, learning_rate: float = 0.01, n_features: int = 10):
        self.learning_rate = learning_rate
        self.weights = np.zeros(n_features)
        self.bias = 0.0

    def partial_fit(self, X: np.ndarray, y: np.ndarray):
        for x_i, y_i in zip(X, y):
            # 预测
            prediction = self._predict(x_i)

            # 计算梯度
            error = prediction - y_i
            gradient_w = error * x_i
            gradient_b = error

            # 更新参数
            self.weights -= self.learning_rate * gradient_w
            self.bias -= self.learning_rate * gradient_b

    def _predict(self, x: np.ndarray) -> float:
        return np.dot(self.weights, x) + self.bias

    def predict(self, X: np.ndarray) -> np.ndarray:
        return np.array([self._predict(x) for x in X])
```

### 2.2 在线逻辑回归

```python
class OnlineLogisticRegression:
    def __init__(self, learning_rate: float = 0.01, n_features: int = 10):
        self.learning_rate = learning_rate
        self.weights = np.zeros(n_features)
        self.bias = 0.0

    def partial_fit(self, X: np.ndarray, y: np.ndarray):
        for x_i, y_i in zip(X, y):
            # Sigmoid 预测
            z = np.dot(self.weights, x_i) + self.bias
            prediction = 1 / (1 + np.exp(-z))

            # 更新
            error = prediction - y_i
            self.weights -= self.learning_rate * error * x_i
            self.bias -= self.learning_rate * error

    def predict_proba(self, X: np.ndarray) -> np.ndarray:
        z = np.dot(X, self.weights) + self.bias
        return 1 / (1 + np.exp(-z))

    def predict(self, X: np.ndarray) -> np.ndarray:
        return (self.predict_proba(X) >= 0.5).astype(int)
```

### 2.3 FTRL 算法

```python
class FTRL:
    def __init__(self, alpha: float = 0.1, beta: float = 1.0,
                 l1: float = 0.0, l2: float = 0.0, n_features: int = 10):
        self.alpha = alpha
        self.beta = beta
        self.l1 = l1
        self.l2 = l2
        self.n = np.zeros(n_features)  # 累积梯度平方
        self.z = np.zeros(n_features)  # 累积梯度
        self.weights = np.zeros(n_features)

    def partial_fit(self, X: np.ndarray, y: np.ndarray):
        for x_i, y_i in zip(X, y):
            # 预测
            prediction = self._predict(x_i)

            # 计算梯度
            error = prediction - y_i
            gradient = error * x_i

            # FTRL 更新
            self.n += gradient ** 2
            sigma = (np.sqrt(self.n + self.beta) - np.sqrt(self.n)) / self.alpha
            self.z += gradient - sigma * self.weights

            # 计算权重（带 L1 正则）
            for i in range(len(self.weights)):
                if abs(self.z[i]) <= self.l1:
                    self.weights[i] = 0
                else:
                    self.weights[i] = -(self.z[i] - np.sign(self.z[i]) * self.l1) /                         ((self.beta + np.sqrt(self.n[i])) / self.alpha + self.l2)

    def _predict(self, x: np.ndarray) -> float:
        z = np.dot(self.weights, x)
        return 1 / (1 + np.exp(-z))
```

class IncrementalDataPipeline:

class SlidingWindowTrainer:
    def __init__(self, model, window_size: int = 10000):
        self.model = model
        self.window_size = window_size
        self.buffer = []

    def add_sample(self, X, y):
        self.buffer.append((X, y))

        # 窗口满时训练
        if len(self.buffer) >= self.window_size:
            self._train_window()
            self.buffer = self.buffer[-self.window_size // 2:]  # 保留最近一半

    def _train_window(self):
        X_batch = np.array([x for x, _ in self.buffer])
        y_batch = np.array([y for _, y in self.buffer])
        self.model.partial_fit(X_batch, y_batch)
```

        model_path = f"models/{version}.pkl"
        await self.storage.save(model_path, model)

        # 保存元数据
        metadata = {
            "version": version,
            "metrics": metrics,
            "created_at": datetime.utcnow().isoformat(),
            "status": "candidate",
        }
        await self.storage.save(f"models/{version}.json", metadata)

    async def promote_model(self, version: str):
        # 提升为生产模型
        metadata = await self.storage.load(f"models/{version}.json")
        metadata["status"] = "production"
        metadata["promoted_at"] = datetime.utcnow().isoformat()
        await self.storage.save(f"models/{version}.json", metadata)

        self.current_version = version

    async def rollback(self, version: str):
        # 回滚到指定版本
        await self.promote_model(version)
```

```python
class CanaryRelease:
    def __init__(self, model_manager):
        self.model_manager = model_manager
        self.canary_percentage = 0
        self.canary_version = None

    def start_canary(self, version: str, percentage: int = 10):
        self.canary_version = version
        self.canary_percentage = percentage

    def predict(self, X, user_id: str):
        # 基于用户 ID 哈希决定使用哪个模型
        hash_value = int(hashlib.md5(user_id.encode()).hexdigest(), 16) % 100

        if hash_value < self.canary_percentage:
            model = self.model_manager.get_model(self.canary_version)
            version = self.canary_version
        else:
            model = self.model_manager.get_model(self.model_manager.current_version)
            version = self.model_manager.current_version

        prediction = model.predict(X)

        # 记录预测用于后续分析
        self._log_prediction(user_id, version, prediction)

        return prediction, version

    def increase_canary(self, percentage: int):
        self.canary_percentage = min(percentage, 100)

        if self.canary_percentage == 100:
            self.model_manager.promote_model(self.canary_version)
            self.canary_version = None
```

## 5. A/B 测试

### 5.1 实验框架

```python
class ABTestFramework:
    def __init__(self):
        self.experiments = {}

    def create_experiment(self, name: str, variants: list[str], traffic_split: list[float]):
        self.experiments[name] = {
            "variants": variants,
            "traffic_split": traffic_split,
            "metrics": {v: [] for v in variants},
        }

    def assign_variant(self, experiment: str, user_id: str) -> str:
        exp = self.experiments[experiment]
        hash_value = int(hashlib.md5(f"{experiment}:{user_id}".encode()).hexdigest(), 16) % 100

        cumulative = 0
        for variant, split in zip(exp["variants"], exp["traffic_split"]):
            cumulative += split * 100
            if hash_value < cumulative:
                return variant

        return exp["variants"][-1]

    def record_metric(self, experiment: str, variant: str, metric: float):
        self.experiments[experiment]["metrics"][variant].append(metric)

    def analyze(self, experiment: str) -> dict:
        exp = self.experiments[experiment]
        results = {}

        for variant, metrics in exp["metrics"].items():
            if metrics:
                results[variant] = {
                    "mean": np.mean(metrics),
                    "std": np.std(metrics),
                    "count": len(metrics),
                }

        # 显著性检验
        if len(results) >= 2:
            control = list(results.keys())[0]
            treatment = list(results.keys())[1]
            t_stat, p_value = ttest_ind(
                exp["metrics"][control],
                exp["metrics"][treatment],
            )
            results["significance"] = {"t_stat": t_stat, "p_value": p_value}

        return results
```

## 6. 效果监控

### 6.1 模型漂移检测

```python
class ModelDriftDetector:
    def __init__(self, threshold: float = 0.1):
        self.threshold = threshold
        self.baseline_metrics = None

    def set_baseline(self, metrics: dict):
        self.baseline_metrics = metrics

    def detect_drift(self, current_metrics: dict) -> bool:
        if self.baseline_metrics is None:
            return False

        # 计算指标变化
        for key, baseline_value in self.baseline_metrics.items():
            current_value = current_metrics.get(key)
            if current_value is not None:
                change = abs(current_value - baseline_value) / baseline_value
                if change > self.threshold:
                    return True

        return False
```

### 6.2 数据漂移检测

```python
from scipy.stats import ks_2samp

class DataDriftDetector:
    def __init__(self, threshold: float = 0.05):
        self.threshold = threshold
        self.baseline_distributions = {}

    def set_baseline(self, data: pd.DataFrame):
        for col in data.select_dtypes(include=[np.number]).columns:
            self.baseline_distributions[col] = data[col].values

    def detect_drift(self, current_data: pd.DataFrame) -> dict:
        drift_results = {}

        for col, baseline in self.baseline_distributions.items():
            if col in current_data.columns:
                current = current_data[col].values
                stat, p_value = ks_2samp(baseline, current)
                drift_results[col] = {
                    "ks_statistic": stat,
                    "p_value": p_value,
                    "drifted": p_value < self.threshold,
                }

        return drift_results
```
