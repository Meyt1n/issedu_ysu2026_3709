# 算法设计-推荐系统与个性化

> 本文档是家健镜系统推荐系统与个性化的完整设计说明，覆盖推荐架构、协同过滤、内容推荐、混合推荐、冷启动。

## 1. 概述

### 1.1 设计目标

1. 推荐准确率 > 70%
2. 推荐多样性 > 0.5
3. 响应时间 < 200ms
4. 支持实时更新
5. 可解释推荐

### 1.2 推荐场景

| 场景 | 推荐内容 | 目标 |
| --- | --- | --- |
| 首页推荐 | 健康知识、文章 | 提高阅读量 |
| 药品推荐 | 相关药品 | 提高用药依从性 |
| 医生推荐 | 相关医生 | 提高咨询率 |

### 2.2 推荐服务

```python
class RecommendationService:
    def __init__(self):
        self.models = {
            'collaborative': CollaborativeFiltering(),
            'content': ContentBasedFiltering(),
            'popular': PopularRecommender(),
        }
        self.weights = {'collaborative': 0.5, 'content': 0.3, 'popular': 0.2}

    async def recommend(self, user_id: str, scene: str, top_n: int = 10) -> list:
        # 获取各模型推荐结果
        results = {}
        for name, model in self.models.items():
            results[name] = await model.recommend(user_id, top_n * 2)

        # 混合排序
        final_results = self._hybrid_rank(results, self.weights)

        # 过滤已看过的
        final_results = await self._filter_seen(user_id, final_results)

        return final_results[:top_n]

    def _hybrid_rank(self, results: dict, weights: dict) -> list:
        scores = {}
        for model_name, items in results.items():
            weight = weights.get(model_name, 0.1)
            for rank, item in enumerate(items):
                score = weight * (1.0 / (rank + 1))
                scores[item['id']] = scores.get(item['id'], 0) + score

        return sorted(scores.items(), key=lambda x: x[1], reverse=True)
```

## 3. 协同过滤

### 3.1 基于用户的协同过滤

```python
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

class UserBasedCF:
    def __init__(self):
        self.user_item_matrix = None
        self.user_similarity = None

    def fit(self, interactions: pd.DataFrame):
        # 构建用户-物品矩阵
        self.user_item_matrix = interactions.pivot_table(
            index='user_id',
            columns='item_id',
            values='rating',
            fill_value=0,
        )

        # 计算用户相似度
        self.user_similarity = cosine_similarity(self.user_item_matrix)

    def recommend(self, user_id: str, top_n: int = 10) -> list:
        user_idx = self.user_item_matrix.index.get_loc(user_id)
        similar_users = np.argsort(self.user_similarity[user_idx])[::-1][1:11]

        # 推荐相似用户喜欢的物品
        scores = {}
        for similar_idx in similar_users:
            similarity = self.user_similarity[user_idx][similar_idx]
            similar_user_items = self.user_item_matrix.iloc[similar_idx]

            for item_id, rating in similar_user_items.items():
                if rating > 0 and self.user_item_matrix.loc[user_id, item_id] == 0:
                    scores[item_id] = scores.get(item_id, 0) + similarity * rating

        return sorted(scores.items(), key=lambda x: x[1], reverse=True)[:top_n]
```

### 3.2 基于物品的协同过滤

```python
class ItemBasedCF:
    def __init__(self):
        self.item_similarity = None

    def fit(self, interactions: pd.DataFrame):
        user_item_matrix = interactions.pivot_table(
            index='user_id',
            columns='item_id',
            values='rating',
            fill_value=0,
        )
        self.item_similarity = cosine_similarity(user_item_matrix.T)
        self.item_ids = user_item_matrix.columns.tolist()

    def recommend(self, user_id: str, user_items: list, top_n: int = 10) -> list:
        scores = {}
        for item_id in user_items:
            item_idx = self.item_ids.index(item_id)
            similar_items = np.argsort(self.item_similarity[item_idx])[::-1][1:11]

            for similar_idx in similar_items:
                similar_item_id = self.item_ids[similar_idx]
                if similar_item_id not in user_items:
                    similarity = self.item_similarity[item_idx][similar_idx]
                    scores[similar_item_id] = scores.get(similar_item_id, 0) + similarity

        return sorted(scores.items(), key=lambda x: x[1], reverse=True)[:top_n]
```

### 3.3 矩阵分解

```python
class MatrixFactorization:
    def __init__(self, n_factors: int = 50, lr: float = 0.01, reg: float = 0.02, epochs: int = 20):
        self.n_factors = n_factors
        self.lr = lr
        self.reg = reg
        self.epochs = epochs

    def fit(self, interactions: pd.DataFrame):
        self.user_ids = interactions['user_id'].unique()
        self.item_ids = interactions['item_id'].unique()
        self.user_idx = {u: i for i, u in enumerate(self.user_ids)}
        self.item_idx = {i: idx for idx, i in enumerate(self.item_ids)}

        n_users = len(self.user_ids)
        n_items = len(self.item_ids)

        self.P = np.random.normal(0, 0.1, (n_users, self.n_factors))
        self.Q = np.random.normal(0, 0.1, (n_items, self.n_factors))

        for epoch in range(self.epochs):
            for _, row in interactions.iterrows():
                u = self.user_idx[row['user_id']]
                i = self.item_idx[row['item_id']]
                rating = row['rating']

                prediction = np.dot(self.P[u], self.Q[i])
                error = rating - prediction

                self.P[u] += self.lr * (error * self.Q[i] - self.reg * self.P[u])
                self.Q[i] += self.lr * (error * self.P[u] - self.reg * self.Q[i])

    def predict(self, user_id: str, item_id: str) -> float:
        u = self.user_idx[user_id]
        i = self.item_idx[item_id]
        return np.dot(self.P[u], self.Q[i])

    def recommend(self, user_id: str, top_n: int = 10) -> list:
        u = self.user_idx[user_id]
        predictions = np.dot(self.P[u], self.Q.T)
        top_indices = np.argsort(predictions)[::-1][:top_n]
        return [(self.item_ids[i], predictions[i]) for i in top_indices]
```

```python
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import linear_kernel

class ContentBasedRecommender:
    def __init__(self):
        self.tfidf = TfidfVectorizer(stop_words='english')
        self.tfidf_matrix = None
        self.items = None

    def fit(self, items: pd.DataFrame):
        self.items = items
        content = items['title'] + ' ' + items['description'] + ' ' + items['tags']
        self.tfidf_matrix = self.tfidf.fit_transform(content)

    def recommend(self, item_id: str, top_n: int = 10) -> list:
        idx = self.items[self.items['id'] == item_id].index[0]
        cosine_similarities = linear_kernel(self.tfidf_matrix[idx], self.tfidf_matrix).flatten()
        similar_indices = cosine_similarities.argsort()[::-1][1:top_n + 1]

        return [(self.items.iloc[i]['id'], cosine_similarities[i]) for i in similar_indices]

    def recommend_for_user(self, user_history: list, top_n: int = 10) -> list:
        # 基于用户历史推荐
        user_profile = np.zeros(self.tfidf_matrix.shape[1])
        for item_id in user_history:
            idx = self.items[self.items['id'] == item_id].index[0]
            user_profile += self.tfidf_matrix[idx].toarray().flatten()

        user_profile = user_profile / len(user_history)
        similarities = linear_kernel(user_profile.reshape(1, -1), self.tfidf_matrix).flatten()
        top_indices = similarities.argsort()[::-1][:top_n]

        return [(self.items.iloc[i]['id'], similarities[i]) for i in top_indices]
```

## 5. 深度学习推荐

### 5.1 DeepFM

```python
import torch
import torch.nn as nn

class DeepFM(nn.Module):
    def __init__(self, field_dims: list, embed_dim: int = 16, hidden_dims: list = [64, 32]):
        super().__init__()
        self.embeddings = nn.ModuleList([
            nn.Embedding(dim, embed_dim) for dim in field_dims
        ])
        self.num_fields = len(field_dims)

        # FM 部分
        self.fm_linear = nn.Linear(sum(field_dims), 1)

        # Deep 部分
        layers = []
        input_dim = self.num_fields * embed_dim
        for hidden_dim in hidden_dims:
            layers.append(nn.Linear(input_dim, hidden_dim))
            layers.append(nn.ReLU())
            layers.append(nn.Dropout(0.2))
            input_dim = hidden_dim
        layers.append(nn.Linear(input_dim, 1))
        self.deep = nn.Sequential(*layers)

    def forward(self, x):
        embeddings = [emb(x[:, i]) for i, emb in enumerate(self.embeddings)]
        embed_stack = torch.stack(embeddings, dim=1)

        # FM 二阶交互
        sum_square = torch.sum(embed_stack, dim=1) ** 2
        square_sum = torch.sum(embed_stack ** 2, dim=1)
        fm_second = 0.5 * torch.sum(sum_square - square_sum, dim=1, keepdim=True)

        # Deep 部分
        deep_input = embed_stack.view(-1, self.num_fields * embed_stack.shape[2])
        deep_output = self.deep(deep_input)

        return torch.sigmoid(fm_second + deep_output)
```

### 5.2 双塔模型

```python
class TwoTowerModel(nn.Module):
    def __init__(self, user_dim: int, item_dim: int, hidden_dim: int = 64):
        super().__init__()
        self.user_tower = nn.Sequential(
            nn.Linear(user_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
        )
        self.item_tower = nn.Sequential(
            nn.Linear(item_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim),
        )

    def forward(self, user_features, item_features):
        user_embed = self.user_tower(user_features)
        item_embed = self.item_tower(item_features)
        return torch.sum(user_embed * item_embed, dim=1)
```

## 7. 推荐评估

### 7.1 离线评估

```python
class RecommendationEvaluator:
    def __init__(self):
        pass

    def precision_at_k(self, recommended: list, actual: list, k: int = 10) -> float:
        recommended_k = set(recommended[:k])
        actual_set = set(actual)
        return len(recommended_k & actual_set) / k

    def recall_at_k(self, recommended: list, actual: list, k: int = 10) -> float:
        recommended_k = set(recommended[:k])
        actual_set = set(actual)
        return len(recommended_k & actual_set) / len(actual_set) if actual_set else 0

    def ndcg_at_k(self, recommended: list, actual: list, k: int = 10) -> float:
        dcg = 0
        for i, item in enumerate(recommended[:k]):
            if item in actual:
                dcg += 1 / np.log2(i + 2)

        ideal = sorted(actual, reverse=True)[:k]
        idcg = sum(1 / np.log2(i + 2) for i in range(len(ideal)))

        return dcg / idcg if idcg > 0 else 0
```
