### 2.2 分类模型

```python
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.svm import LinearSVC
from sklearn.pipeline import Pipeline

class IntentClassifier:
    def __init__(self):
        self.model = Pipeline([
            ('tfidf', TfidfVectorizer(
                max_features=5000,
                ngram_range=(1, 2),
                sublinear_tf=True,
            )),
            ('clf', LinearSVC()),
        ])

    def train(self, texts, labels):
        self.model.fit(texts, labels)

    def predict(self, text):
        intent = self.model.predict([text])[0]
        confidence = self.model.decision_function([text]).max()
        return {
            'intent': intent,
            'confidence': float(confidence),
        }
```

### 2.3 深度学习分类

```python
import tensorflow as tf
from tensorflow.keras.layers import Input, Embedding, LSTM, Dense, Dropout
from tensorflow.keras.models import Model

class DeepIntentClassifier:
    def __init__(self, vocab_size=10000, embed_dim=128, num_classes=20):
        self.model = self._build_model(vocab_size, embed_dim, num_classes)

    def _build_model(self, vocab_size, embed_dim, num_classes):
        inputs = Input(shape=(None,))
        x = Embedding(vocab_size, embed_dim)(inputs)
        x = LSTM(128, return_sequences=True)(x)
        x = LSTM(64)(x)
        x = Dropout(0.3)(x)
        x = Dense(64, activation='relu')(x)
        outputs = Dense(num_classes, activation='softmax')(x)

        model = Model(inputs, outputs)
        model.compile(
            optimizer='adam',
            loss='sparse_categorical_crossentropy',
            metrics=['accuracy'],
        )
        return model

    def predict(self, text):
        sequence = self.tokenizer.texts_to_sequences([text])
        padded = tf.keras.preprocessing.sequence.pad_sequences(
            sequence, maxlen=50
        )
        probabilities = self.model.predict(padded)[0]
        intent_idx = probabilities.argmax()
        return {
            'intent': self.labels[intent_idx],
            'confidence': float(probabilities[intent_idx]),
        }
```

### 3.2 基于规则的抽取

```python
import re

class RuleBasedEntityExtractor:
    def __init__(self, dictionaries):
        self.dictionaries = dictionaries
        self.patterns = {
            'blood_pressure': r'(\d{2,3})/(\d{2,3})',
            'blood_glucose': r'血糖\s*(\d+\.?\d*)',
            'temperature': r'体温\s*(\d+\.?\d*)',
            'heart_rate': r'心率\s*(\d+)',
            'dosage': r'(\d+)\s*(片|粒|mg|毫升|ml)',
            'frequency': r'(每日|每天|每)\s*(一|二|三|四|1|2|3|4)\s*次',
        }

    def extract(self, text):
        entities = []

        # 1. 词典匹配
        for entity_type, dictionary in self.dictionaries.items():
            for word in dictionary:
                if word in text:
                    entities.append({
                        'type': entity_type,
                        'text': word,
                        'offset': text.index(word),
                        'source': 'dictionary',
                    })

        # 2. 正则匹配
        for entity_type, pattern in self.patterns.items():
            for match in re.finditer(pattern, text):
                entities.append({
                    'type': entity_type,
                    'text': match.group(),
                    'offset': match.start(),
                    'value': match.groups(),
                    'source': 'regex',
                })

        return entities
```

### 3.3 基于模型的抽取（NER）

```python
class NERExtractor:
    def __init__(self, model):
        self.model = model

    def extract(self, text):
        predictions = self.model.predict(text)
        entities = []

        current_entity = None
        for token, pred in zip(text, predictions):
            if pred.startswith('B-'):
                if current_entity:
                    entities.append(current_entity)
                current_entity = {
                    'type': pred[2:],
                    'text': token,
                    'start': token.idx,
                }
            elif pred.startswith('I-') and current_entity:
                current_entity['text'] += token
            else:
                if current_entity:
                    entities.append(current_entity)
                    current_entity = None

        if current_entity:
            entities.append(current_entity)

        return entities
```

### 4.1 语义解析

```python
class SemanticParser:
    def __init__(self, intent_classifier, entity_extractor):
        self.intent_classifier = intent_classifier
        self.entity_extractor = entity_extractor

    def parse(self, text, context=None):
        # 1. 意图识别
        intent_result = self.intent_classifier.predict(text)

        # 2. 实体抽取
        entities = self.entity_extractor.extract(text)

        # 3. 上下文补全
        if context:
            entities = self._fill_from_context(entities, context)

        # 4. 槽位填充
        slots = self._fill_slots(intent_result['intent'], entities)

        return {
            'intent': intent_result['intent'],
            'confidence': intent_result['confidence'],
            'entities': entities,
            'slots': slots,
            'need_clarification': self._need_clarification(slots),
        }

    def _fill_slots(self, intent, entities):
        slot_definitions = SLOT_DEFINITIONS.get(intent, {})
        slots = {}
        for slot_name, slot_type in slot_definitions.items():
            matching = [e for e in entities if e['type'] == slot_type]
            if matching:
                slots[slot_name] = matching[0]
        return slots
```

### 4.2 多轮对话

```python
class DialogManager:
    def __init__(self):
        self.contexts = {}  # user_id -> context

    def process(self, user_id, text):
        context = self.contexts.get(user_id, {})

        # 解析当前输入
        parse_result = self.parser.parse(text, context)

        # 更新上下文
        context['last_intent'] = parse_result['intent']
        context['last_entities'] = parse_result['entities']
        context['slots'] = {
            **context.get('slots', {}),
            **parse_result['slots'],
        }
        self.contexts[user_id] = context

        # 判断是否需要澄清
        if parse_result['need_clarification']:
            return self._ask_clarification(parse_result, context)

        # 执行对应操作
        return self._execute(parse_result, context)
```
