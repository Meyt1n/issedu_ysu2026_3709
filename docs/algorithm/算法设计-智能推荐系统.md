| 健康内容推荐 | 推荐健康文章 | 兴趣、健康状况 | 文章列表 |

## 2. 推荐策略

### 2.1 基于规则的推荐

```python
class RuleBasedRecommender:
    def recommend_medicine(self, symptoms, member_profile):
        candidates = []

        # 基于症状匹配
        for medicine in MEDICINE_DATABASE:
            match_score = 0
            for symptom in symptoms:
                if symptom in medicine.indications:
                    match_score += 1

            # 排除禁忌
            if self._has_contraindication(medicine, member_profile):
                continue

            if match_score > 0:
                candidates.append({
                    "medicine": medicine,
                    "score": match_score,
                    "matched_symptoms": [
                        s for s in symptoms if s in medicine.indications
                    ],
                })

        # 排序
        candidates.sort(key=lambda x: x["score"], reverse=True)
        return candidates[:10]
```

### 2.2 基于内容的推荐

```python
class ContentBasedRecommender:
    def __init__(self, tfidf_vectorizer, content_db):
        self.vectorizer = tfidf_vectorizer
        self.content_db = content_db
        self.content_vectors = self._vectorize_content()

    def _vectorize_content(self):
        texts = [c.text for c in self.content_db]
        return self.vectorizer.fit_transform(texts)

    def recommend(self, user_profile, top_n=10):
        # 用户画像向量化
        user_text = " ".join([
            user_profile.interests,
            user_profile.health_conditions,
            user_profile.recent_searches,
        ])
        user_vector = self.vectorizer.transform([user_text])

        # 计算相似度
        from sklearn.metrics.pairwise import cosine_similarity
        similarities = cosine_similarity(
            user_vector, self.content_vectors
        )[0]

        # 排序
        top_indices = similarities.argsort()[-top_n:][::-1]
        return [
            {
                "content": self.content_db[i],
                "score": float(similarities[i]),
            }
            for i in top_indices
        ]
```

### 2.3 协同过滤推荐

```python
class CollaborativeFilteringRecommender:
    def __init__(self, user_item_matrix):
        self.user_item_matrix = user_item_matrix
        self.user_similarity = self._compute_user_similarity()

    def _compute_user_similarity(self):
        from sklearn.metrics.pairwise import cosine_similarity
        return cosine_similarity(self.user_item_matrix)

    def recommend(self, user_id, top_n=10):
        user_idx = self._get_user_idx(user_id)
        similar_users = self.user_similarity[user_idx].argsort()[::-1][1:11]

        # 汇总相似用户的偏好
        scores = {}
        for similar_idx in similar_users:
            similarity = self.user_similarity[user_idx][similar_idx]
            user_items = self.user_item_matrix[similar_idx]

            for item_idx, rating in enumerate(user_items):
                if rating > 0 and self.user_item_matrix[user_idx][item_idx] == 0:
                    if item_idx not in scores:
                        scores[item_idx] = 0
                    scores[item_idx] += similarity * rating

        # 排序
        top_items = sorted(scores.items(), key=lambda x: x[1], reverse=True)[:top_n]
        return [
            {"item_id": self._get_item_id(idx), "score": score}
            for idx, score in top_items
        ]
```

## 3. 混合推荐

### 3.1 加权混合

```python
class HybridRecommender:
    def __init__(self, recommenders, weights):
        self.recommenders = recommenders
        self.weights = weights

    def recommend(self, user_context, top_n=10):
        all_scores = {}

        for name, recommender in self.recommenders.items():
            weight = self.weights.get(name, 1.0)
            results = recommender.recommend(user_context)

            for result in results:
                item_id = result["item_id"]
                if item_id not in all_scores:
                    all_scores[item_id] = {
                        "item": result["item"],
                        "scores": {},
                        "total_score": 0,
                    }
                all_scores[item_id]["scores"][name] = result["score"]
                all_scores[item_id]["total_score"] += result["score"] * weight

        # 排序
        sorted_items = sorted(
            all_scores.values(),
            key=lambda x: x["total_score"],
            reverse=True,
        )
        return sorted_items[:top_n]
```

## 4. 排序优化

### 4.1 Learning to Rank

```python
class LearningToRank:
    def __init__(self, model):
        self.model = model

    def rank(self, candidates, user_context):
        features = self._extract_features(candidates, user_context)
        scores = self.model.predict(features)

        ranked = sorted(
            zip(candidates, scores),
            key=lambda x: x[1],
            reverse=True,
        )
        return [item for item, score in ranked]

    def _extract_features(self, candidates, user_context):
        features = []
        for candidate in candidates:
            feature_vector = [
                candidate["relevance_score"],
                candidate["popularity"],
                candidate["freshness"],
                self._user_match_score(candidate, user_context),
                self._diversity_score(candidate, candidates),
            ]
            features.append(feature_vector)
        return features
```
