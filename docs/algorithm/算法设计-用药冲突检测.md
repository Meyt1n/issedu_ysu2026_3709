# 算法设计-用药冲突检测

> 本文档是家健镜系统用药冲突检测算法的完整设计说明，覆盖冲突类型、检测算法、评分机制、性能优化。

## 1. 算法概述

### 1.1 目标

自动检测家庭成员用药方案中的潜在冲突，包括成分重复、相互作用、禁忌证、剂量超标等，保障用药安全。

### 1.2 冲突类型

| 类型 | 说明 | 严重程度 |
| --- | --- | --- |
| 成分重复 | 多种药品含相同活性成分 | 高 |
| 相互作用 | 药品间存在不良相互作用 | 高 |
| 禁忌证 | 药品与患者疾病/过敏冲突 | 严重 |
| 剂量超标 | 单日剂量超过安全上限 | 高 |
| 重复用药 | 同类药品重复使用 | 中 |
| 用药不当 | 用法用量不符合说明书 | 中 |

## 3. 检测算法

### 3.1 成分重复检测

```python
class DuplicateIngredientDetector:
    def detect(self, medicines):
        ingredient_map = {}

        for medicine in medicines:
            for ingredient in medicine.ingredients:
                key = ingredient.ingredient_code
                if key not in ingredient_map:
                    ingredient_map[key] = []
                ingredient_map[key].append({
                    "medicine": medicine,
                    "ingredient": ingredient,
                    "daily_dose": self._calculate_daily_dose(medicine, ingredient),
                })

        conflicts = []
        for code, usages in ingredient_map.items():
            if len(usages) > 1:
                total_dose = sum(u["daily_dose"] for u in usages)
                max_dose = self._get_max_dose(code)

                conflicts.append({
                    "type": "duplicate_ingredient",
                    "ingredient": usages[0]["ingredient"].ingredient_name,
                    "medicines": [u["medicine"].name for u in usages],
                    "total_daily_dose": total_dose,
                    "max_safe_dose": max_dose,
                    "exceeds_limit": total_dose > max_dose if max_dose else None,
                    "severity": "high" if total_dose > max_dose else "medium",
                })

        return conflicts

    def _calculate_daily_dose(self, medicine, ingredient):
        frequency = medicine.frequency  # 每日次数
        dose_per_time = ingredient.dose  # 每次含量
        return frequency * dose_per_time

    def _get_max_dose(self, ingredient_code):
        max_dose = SAFE_DOSE_LIMITS.get(ingredient_code)
        return max_dose
```

### 3.2 相互作用检测

```python
class InteractionDetector:
    def __init__(self, interaction_db):
        self.interaction_db = interaction_db

    def detect(self, medicines):
        conflicts = []
        ingredients = self._extract_ingredients(medicines)

        # 两两比较
        for i in range(len(ingredients)):
            for j in range(i + 1, len(ingredients)):
                ing_a = ingredients[i]
                ing_b = ingredients[j]

                interaction = self._find_interaction(
                    ing_a["code"], ing_b["code"]
                )
                if interaction:
                    conflicts.append({
                        "type": "drug_interaction",
                        "ingredient_a": ing_a["name"],
                        "ingredient_b": ing_b["name"],
                        "medicine_a": ing_a["medicine_name"],
                        "medicine_b": ing_b["medicine_name"],
                        "severity": interaction.severity,
                        "description": interaction.description,
                        "recommendation": interaction.recommendation,
                    })

        return conflicts

    def _extract_ingredients(self, medicines):
        ingredients = []
        for medicine in medicines:
            for ing in medicine.ingredients:
                ingredients.append({
                    "code": ing.ingredient_code,
                    "name": ing.ingredient_name,
                    "medicine_name": medicine.name,
                })
        return ingredients

    def _find_interaction(self, code_a, code_b):
        return self.interaction_db.find(code_a, code_b)
```

### 3.3 禁忌证检测

```python
class ContraindicationDetector:
    def __init__(self, contraindication_db):
        self.contraindication_db = contraindication_db

    def detect(self, medicines, member_profile):
        conflicts = []
        conditions = member_profile.medical_conditions
        allergies = member_profile.allergies

        for medicine in medicines:
            for ingredient in medicine.ingredients:
                # 检查疾病禁忌
                for condition in conditions:
                    contraindication = self.contraindication_db.find(
                        ingredient.ingredient_code, condition
                    )
                    if contraindication:
                        conflicts.append({
                            "type": "contraindication",
                            "medicine": medicine.name,
                            "ingredient": ingredient.ingredient_name,
                            "condition": condition,
                            "severity": contraindication.severity,
                            "description": contraindication.description,
                        })

                # 检查过敏
                if ingredient.ingredient_name in allergies:
                    conflicts.append({
                        "type": "allergy",
                        "medicine": medicine.name,
                        "ingredient": ingredient.ingredient_name,
                        "severity": "critical",
                        "description": f"患者对 {ingredient.ingredient_name} 过敏",
                    })

        return conflicts
```

### 3.4 剂量超标检测

```python
class DoseExceedanceDetector:
    def detect(self, medicines):
        conflicts = []

        for medicine in medicines:
            for ingredient in medicine.ingredients:
                daily_dose = medicine.frequency * ingredient.dose
                max_daily = self._get_max_daily_dose(ingredient.ingredient_code)

                if max_daily and daily_dose > max_daily:
                    conflicts.append({
                        "type": "dose_exceedance",
                        "medicine": medicine.name,
                        "ingredient": ingredient.ingredient_name,
                        "daily_dose": daily_dose,
                        "max_daily_dose": max_daily,
                        "exceed_percent": round(
                            (daily_dose - max_daily) / max_daily * 100, 1
                        ),
                        "severity": "high",
                    })

        return conflicts
```

## 4. 综合评分

### 4.1 风险评分

```python
class RiskScorer:
    SEVERITY_WEIGHTS = {
        "critical": 100,
        "high": 50,
        "medium": 20,
        "low": 5,
    }

    def calculate_score(self, conflicts):
        if not conflicts:
            return 0, "safe"

        total_score = sum(
            self.SEVERITY_WEIGHTS.get(c["severity"], 10)
            for c in conflicts
        )

        if total_score >= 100:
            level = "critical"
        elif total_score >= 50:
            level = "high"
        elif total_score >= 20:
            level = "medium"
        else:
            level = "low"

        return total_score, level
```
