## 2. 文本分类

### 2.1 基于 BERT 的分类

```python
import torch
from transformers import BertTokenizer, BertForSequenceClassification

class TextClassifier:
    def __init__(self, model_path: str, num_labels: int):
        self.tokenizer = BertTokenizer.from_pretrained(model_path)
        self.model = BertForSequenceClassification.from_pretrained(
            model_path,
            num_labels=num_labels,
        )
        self.model.eval()

    def classify(self, text: str) -> dict:
        inputs = self.tokenizer(
            text,
            return_tensors='pt',
            truncation=True,
            padding=True,
            max_length=512,
        )

        with torch.no_grad():
            outputs = self.model(**inputs)
            probabilities = torch.softmax(outputs.logits, dim=1)

        predicted_class = torch.argmax(probabilities, dim=1).item()
        confidence = probabilities[0][predicted_class].item()

        return {
            'class': predicted_class,
            'confidence': confidence,
            'probabilities': probabilities[0].tolist(),
        }
```

## 3. 命名实体识别

### 3.1 医疗实体识别

```python
from transformers import BertForTokenClassification

class MedicalNER:
    def __init__(self, model_path: str):
        self.tokenizer = BertTokenizer.from_pretrained(model_path)
        self.model = BertForTokenClassification.from_pretrained(model_path)
        self.model.eval()
        self.label_list = ['O', 'B-DRUG', 'I-DRUG', 'B-DISEASE', 'I-DISEASE',
                           'B-SYMPTOM', 'I-SYMPTOM', 'B-TREATMENT', 'I-TREATMENT']

    def extract_entities(self, text: str) -> list[dict]:
        inputs = self.tokenizer(text, return_tensors='pt', truncation=True)
        tokens = self.tokenizer.convert_ids_to_tokens(inputs['input_ids'][0])

        with torch.no_grad():
            outputs = self.model(**inputs)
            predictions = torch.argmax(outputs.logits, dim=2)[0]

        entities = []
        current_entity = None

        for token, pred in zip(tokens, predictions):
            label = self.label_list[pred.item()]

            if label.startswith('B-'):
                if current_entity:
                    entities.append(current_entity)
                current_entity = {
                    'type': label[2:],
                    'text': token,
                    'start': len(''.join(tokens[:tokens.index(token)])),
                }
            elif label.startswith('I-') and current_entity:
                current_entity['text'] += token
            else:
                if current_entity:
                    entities.append(current_entity)
                    current_entity = None

        if current_entity:
            entities.append(current_entity)

        return entities
```

class RetrievalQA:
    def __init__(self, es_client, embedding_model):
        self.es = es_client
        self.embedding_model = embedding_model

    def answer(self, question: str, top_k: int = 5) -> dict:
        # 1. 检索相关文档
        query_embedding = self.embedding_model.encode(question)
        docs = self._search_similar_docs(query_embedding, top_k)

        # 2. 构建上下文
        context = '
'.join([doc['content'] for doc in docs])

        # 3. 生成回答
        answer = self._generate_answer(question, context)

        return {
            'answer': answer,
            'sources': [doc['title'] for doc in docs],
            'confidence': self._calculate_confidence(docs),
        }

    def _search_similar_docs(self, embedding, top_k: int) -> list:
        # 使用 Elasticsearch 向量搜索
        body = {
            "query": {
                "script_score": {
                    "query": {"match_all": {}},
                    "script": {
                        "source": "cosineSimilarity(params.query_vector, 'embedding') + 1.0",
                        "params": {"query_vector": embedding.tolist()},
                    },
                }
            },
            "size": top_k,
        }
        result = self.es.search(index="health_knowledge", body=body)
        return [hit['_source'] for hit in result['hits']['hits']]
```

### 6.1 多轮对话

class DialogueManager:
    def __init__(self):
        self.sessions = {}

    def process_message(self, session_id: str, message: str) -> dict:
        # 获取或创建会话
        if session_id not in self.sessions:
            self.sessions[session_id] = DialogueSession()

        session = self.sessions[session_id]

        # 识别意图
        intent = self.intent_recognizer.recognize(message)

        # 填充槽位
        slots = self.slot_filler.fill_slots(message)

        # 更新会话状态
        session.update(intent, slots)

        # 检查是否需要追问
        if session.needs_clarification():
            return {'reply': session.get_clarification_question(), 'need_more_info': True}

        # 执行任务
        result = self.execute_task(session)

        return {
            'reply': result['reply'],
            'intent': intent['intent'],
            'slots': session.slots,
            'need_more_info': False,
        }
```

### 6.2 对话状态

```python
class DialogueSession:
    def __init__(self):
        self.intent = None
        self.slots = {}
        self.history = []
        self.created_at = datetime.utcnow()

    def update(self, intent: dict, slots: dict):
        self.intent = intent
        self.slots.update(slots)
        self.history.append({'intent': intent, 'slots': slots})

    def needs_clarification(self) -> bool:
        required_slots = self._get_required_slots()
        return any(slot not in self.slots for slot in required_slots)

    def get_clarification_question(self) -> str:
        required_slots = self._get_required_slots()
        for slot in required_slots:
            if slot not in self.slots:
                return self._get_question_for_slot(slot)
        return ""

    def _get_required_slots(self) -> list[str]:
        slot_requirements = {
            'query_medicine': ['medicine_name'],
            'book_appointment': ['doctor_name', 'time'],
            'set_reminder': ['medicine_name', 'time'],
        }
        return slot_requirements.get(self.intent['intent'], [])
```

class SentimentAnalyzer:

        return {
            'sentiment': labels[result['class']],
            'confidence': result['confidence'],
