## 1. 概述

### 1.1 设计目标

1. 数据不出域：原始数据不离开本地

3. 隐私保护：防止数据泄露
4. 性能可接受：训练效率可接受
5. 模型可用：模型效果接近集中式训练

### 1.2 隐私计算技术

## 2. 联邦学习框架

### 2.1 横向联邦学习

```python
class HorizontalFederatedLearning:
    def __init__(self, model, clients: list, server):
        self.global_model = model
        self.clients = clients
        self.server = server

    def train(self, rounds: int = 10):
        for round in range(rounds):
            # 1. 服务器下发全局模型
            global_weights = self.global_model.get_weights()

            # 2. 客户端本地训练
            client_updates = []
            for client in self.clients:
                local_weights = client.train(global_weights, epochs=1)
                client_updates.append({
                    "weights": local_weights,
                    "samples": client.sample_count,
                })

            # 3. 服务器聚合
            new_weights = self._fedavg(client_updates)
            self.global_model.set_weights(new_weights)

            # 4. 评估
            metrics = self._evaluate()
            print(f"Round {round}: {metrics}")

    def _fedavg(self, updates: list) -> list:
        total_samples = sum(u["samples"] for u in updates)
        averaged_weights = []

        for i in range(len(updates[0]["weights"])):
            weighted_sum = sum(
                u["weights"][i] * u["samples"] for u in updates
            )
            averaged_weights.append(weighted_sum / total_samples)

        return averaged_weights
```

### 2.3 联邦平均算法

```python
class FedAvg:
    def aggregate(self, client_updates: list) -> dict:
        '''
        client_updates: [{"weights": ..., "samples": n}, ...]
        '''
        total_samples = sum(u["samples"] for u in client_updates)

        # 按样本数加权平均
        aggregated = {}
        for key in client_updates[0]["weights"].keys():
            aggregated[key] = sum(
                u["weights"][key] * u["samples"] for u in client_updates
            ) / total_samples

        return aggregated
```

## 3. 安全聚合

### 3.1 安全聚合协议

```python
class SecureAggregation:
    def __init__(self, clients: list):
        self.clients = clients
        self.key_pairs = {}

    def setup(self):
        # 为每对客户端生成密钥对
        for i, client_a in enumerate(self.clients):
            for client_b in self.clients[i+1:]:
                # 协商共享密钥
                shared_key = self._diffie_hellman(client_a, client_b)
                self.key_pairs[(client_a.id, client_b.id)] = shared_key

    def aggregate(self, client_updates: list) -> dict:
        # 1. 客户端添加掩码
        masked_updates = []
        for update in client_updates:
            masked = self._add_mask(update)
            masked_updates.append(masked)

        # 2. 服务器聚合掩码后的更新
        aggregated = self._sum_updates(masked_updates)

        # 3. 移除掩码（需要存活客户端协助）
        result = self._remove_masks(aggregated)

        return result

    def _add_mask(self, update: dict) -> dict:
        # 使用与其他客户端的共享密钥添加掩码
        masked = {}
        for key, value in update.items():
            mask = self._compute_mask(key)
            masked[key] = value + mask
        return masked

    def _diffie_hellman(self, client_a, client_b) -> bytes:
        # 简化的 Diffie-Hellman 密钥交换
        return b"shared_secret"
```

### 3.2 秘密分享

```python
class SecretSharing:
    def __init__(self, threshold: int, n_parties: int):
        self.threshold = threshold
        self.n_parties = n_parties

    def share(self, secret: float) -> list[float]:
        # Shamir 秘密分享
        coefficients = [secret] + [random.uniform(-1, 1) for _ in range(self.threshold - 1)]

        shares = []
        for i in range(1, self.n_parties + 1):
            share = sum(c * (i ** j) for j, c in enumerate(coefficients))
            shares.append(share)

        return shares

    def reconstruct(self, shares: list[tuple[int, float]]) -> float:
        # Lagrange 插值
        result = 0
        for i, (x_i, y_i) in enumerate(shares):
            numerator = 1
            denominator = 1
            for j, (x_j, _) in enumerate(shares):
                if i != j:
                    numerator *= -x_j
                    denominator *= (x_i - x_j)
            result += y_i * numerator / denominator
        return result
```

## 4. 差分隐私

### 4.1 拉普拉斯机制

```python
class LaplaceMechanism:
    def __init__(self, epsilon: float, sensitivity: float):
        self.epsilon = epsilon
        self.sensitivity = sensitivity
        self.scale = sensitivity / epsilon

    def add_noise(self, value: float) -> float:
        noise = np.random.laplace(0, self.scale)
        return value + noise

    def add_noise_to_vector(self, vector: np.ndarray) -> np.ndarray:
        noise = np.random.laplace(0, self.scale, size=vector.shape)
        return vector + noise
```

### 4.2 高斯机制

```python
class GaussianMechanism:
    def __init__(self, epsilon: float, delta: float, sensitivity: float):
        self.epsilon = epsilon
        self.delta = delta
        self.sensitivity = sensitivity
        self.sigma = sensitivity * np.sqrt(2 * np.log(1.25 / delta)) / epsilon

    def add_noise(self, value: float) -> float:
        noise = np.random.normal(0, self.sigma)
        return value + noise
```

### 4.3 差分隐私 SGD

```python
class DPSGD:
    def __init__(self, model, l2_norm_clip: float, noise_multiplier: float, learning_rate: float):
        self.model = model
        self.l2_norm_clip = l2_norm_clip
        self.noise_multiplier = noise_multiplier
        self.learning_rate = learning_rate

    def train_step(self, x_batch, y_batch):
        gradients = []

        # 1. 逐样本计算梯度
        for x_i, y_i in zip(x_batch, y_batch):
            grad = self._compute_gradient(x_i, y_i)
            # 2. 梯度裁剪
            clipped_grad = self._clip_gradient(grad)
            gradients.append(clipped_grad)

        # 3. 聚合梯度
        avg_grad = np.mean(gradients, axis=0)

        # 4. 添加噪声
        noise_std = self.noise_multiplier * self.l2_norm_clip / len(x_batch)
        noisy_grad = avg_grad + np.random.normal(0, noise_std, avg_grad.shape)

        # 5. 更新参数
        self.model.weights -= self.learning_rate * noisy_grad

    def _clip_gradient(self, gradient: np.ndarray) -> np.ndarray:
        norm = np.linalg.norm(gradient)
        if norm > self.l2_norm_clip:
            return gradient * (self.l2_norm_clip / norm)
        return gradient
```

## 5. 同态加密

### 5.1 加法同态

```python
class AdditiveHomomorphicEncryption:
    def __init__(self, key_size: int = 2048):
        self.public_key, self.private_key = self._generate_keys(key_size)

    def encrypt(self, plaintext: int) -> int:
        # Paillier 加密
        return self._paillier_encrypt(plaintext)

    def decrypt(self, ciphertext: int) -> int:
        return self._paillier_decrypt(ciphertext)

    def add(self, ciphertext_a: int, ciphertext_b: int) -> int:
        # 密文相加等于明文相加
        return (ciphertext_a * ciphertext_b) % (self.public_key.n ** 2)

    def add_plaintext(self, ciphertext: int, plaintext: int) -> int:
        # 密文加明文
        return (ciphertext * pow(self.public_key.g, plaintext, self.public_key.n ** 2)) % (self.public_key.n ** 2)
```

### 5.2 隐私求交

```python
class PrivateSetIntersection:
    def __init__(self):
        pass

    def compute_intersection(self, set_a: set, set_b: set) -> set:
        # 基于哈希的隐私求交
        # 双方各自哈希，求交集
        hash_a = {hash(x) for x in set_a}
        hash_b = {hash(x) for x in set_b}

        common_hashes = hash_a & hash_b

        # 还原原始值
        intersection = {x for x in set_a if hash(x) in common_hashes}
        return intersection
```

## 6. 联邦学习系统架构

### 6.1 客户端

```python
class FederatedClient:
    def __init__(self, client_id: str, model, data):
        self.client_id = client_id
        self.model = model
        self.data = data

    def train(self, global_weights, epochs: int = 1) -> dict:
        # 1. 加载全局模型
        self.model.set_weights(global_weights)

        # 2. 本地训练
        for epoch in range(epochs):
            for batch in self.data:
                self.model.train_step(batch)

        # 3. 返回模型更新
        return {
            "client_id": self.client_id,
            "weights": self.model.get_weights(),
            "samples": len(self.data),
        }
```

### 6.2 服务器

```python
class FederatedServer:
    def __init__(self, model, aggregation_strategy="fedavg"):
        self.global_model = model
        self.aggregation_strategy = aggregation_strategy
        self.clients = []

    def register_client(self, client):
        self.clients.append(client)

    def run_round(self, fraction: float = 0.1) -> dict:
        # 1. 选择客户端
        selected_clients = self._select_clients(fraction)

        # 2. 下发全局模型
        global_weights = self.global_model.get_weights()

        # 3. 客户端训练
        updates = []
        for client in selected_clients:
            update = client.train(global_weights)
            updates.append(update)

        # 4. 聚合
        new_weights = self._aggregate(updates)
        self.global_model.set_weights(new_weights)

        return {"round_complete": True, "clients_participated": len(updates)}
```
