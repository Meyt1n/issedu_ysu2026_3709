### 2.3 图谱存储

```python
from neo4j import GraphDatabase

class KnowledgeGraph:
    def __init__(self, uri, user, password):
        self.driver = GraphDatabase.driver(uri, auth=(user, password))

    def create_entity(self, entity_type, properties):
        with self.driver.session() as session:
            session.run(
                f"CREATE (e:{entity_type} $props) RETURN e",
                props=properties,
            )

    def create_relation(self, source, relation, target, properties=None):
        with self.driver.session() as session:
            session.run(
                f"MATCH (a), (b) WHERE a.id = $source AND b.id = $target "
                f"CREATE (a)-[r:{relation} $props]->(b)",
                source=source,
                target=target,
                props=properties or {},
            )

    def query(self, cypher, params=None):
        with self.driver.session() as session:
            result = session.run(cypher, params or {})
            return [record.data() for record in result]
```

## 3. 实体抽取

### 3.1 基于词典的抽取

```python
class DictionaryEntityExtractor:
    def __init__(self, entity_dictionaries):
        self.dictionaries = entity_dictionaries

    def extract(self, text):
        entities = []

        for entity_type, dictionary in self.dictionaries.items():
            for entity_name in dictionary:
                if entity_name in text:
                    entities.append({
                        "text": entity_name,
                        "type": entity_type,
                        "offset": text.index(entity_name),
                    })

        return entities
```

### 3.2 基于模型的抽取

```python
class ModelEntityExtractor:
    def __init__(self, model):
        self.model = model

    def extract(self, text):
        # 使用 NER 模型抽取实体
        predictions = self.model.predict(text)

        entities = []
        for pred in predictions:
            entities.append({
                "text": pred["text"],
                "type": pred["label"],
                "offset": pred["start"],
                "confidence": pred["score"],
            })

        return entities
```

### 4.1 基于规则的抽取

class RuleRelationExtractor:
    def __init__(self, patterns):
        self.patterns = patterns

    def extract(self, text, entities):
        relations = []

        for pattern in self.patterns:
            matches = pattern["regex"].finditer(text)
            for match in matches:
                source = self._find_entity(entities, match.group("source"))
                target = self._find_entity(entities, match.group("target"))

                if source and target:
                    relations.append({
                        "source": source,
                        "relation": pattern["relation"],
                        "target": target,
                        "confidence": pattern.get("confidence", 0.9),
                    })

        return relations
```

class ModelRelationExtractor:
    def __init__(self, model):
        self.model = model

    def extract(self, text, entities):
        relations = []

        # 两两实体对
        for i, e1 in enumerate(entities):
            for e2 in entities[i+1:]:
                relation = self.model.predict_relation(text, e1, e2)
                if relation:
                    relations.append({
                        "source": e1,
                        "relation": relation["type"],
                        "target": e2,
                        "confidence": relation["score"],
                    })

        return relations
```

## 5. 图谱应用

### 5.1 智能问答

```python
class GraphQA:
    def __init__(self, kg):
        self.kg = kg

    def answer(self, question):
        # 1. 实体识别
        entities = self._extract_entities(question)

        # 2. 关系识别
        relation = self._extract_relation(question)

        # 3. 图谱查询
        cypher = self._build_query(entities, relation)
        results = self.kg.query(cypher)

        # 4. 生成回答
        return self._generate_answer(results)

    def _build_query(self, entities, relation):
        if relation == "indicates":
            return (
                "MATCH (m:medicine)-[:indicates]->(d:disease) "
                "WHERE m.name = $medicine_name "
                "RETURN d.name as disease",
                {"medicine_name": entities[0]["text"]},
            )
```

### 5.2 用药推荐

```python
class GraphMedicineRecommender:
    def __init__(self, kg):
        self.kg = kg

    def recommend(self, symptoms, member_profile):
        # 1. 根据症状找疾病
        diseases = self.kg.query('''
            MATCH (d:disease)-[:has_symptom]->(s:symptom)
            WHERE s.name IN $symptoms
            RETURN d.name as disease, count(s) as match_count
            ORDER BY match_count DESC
            LIMIT 5
        ''', {"symptoms": symptoms})

        # 2. 根据疾病找药品
        recommendations = []
        for disease in diseases:
            medicines = self.kg.query('''
                MATCH (m:medicine)-[:indicates]->(d:disease)
                WHERE d.name = $disease
                AND NOT (m)-[:contraindicated]->(:disease {name: $condition})
                RETURN m.name as medicine
            ''', {
                "disease": disease["disease"],
                "condition": member_profile.condition,
            })
            recommendations.extend(medicines)

        return recommendations
```

### 5.3 知识推理

```python
class GraphReasoning:
    def __init__(self, kg):
        self.kg = kg

    def infer_interactions(self, medicines):
        # 多跳推理：药品-成分-相互作用-成分-药品
        return self.kg.query('''
            MATCH (m1:medicine)-[:has_ingredient]->(i1:ingredient)
                 -[:interacts_with]->(i2:ingredient)
                 <-[:has_ingredient]-(m2:medicine)
            WHERE m1.name IN $medicines AND m2.name IN $medicines
            AND m1.name <> m2.name
            RETURN m1.name as medicine1, m2.name as medicine2,
                   i1.name as ingredient1, i2.name as ingredient2
        ''', {"medicines": medicines})
```
