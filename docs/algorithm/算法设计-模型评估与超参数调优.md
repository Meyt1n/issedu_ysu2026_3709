## 2. 评估指标

### 2.1 分类指标

```python
from sklearn.metrics import (
    accuracy_score, precision_score, recall_score, f1_score,
    roc_auc_score, confusion_matrix, classification_report
)

class ClassificationEvaluator:
    def evaluate(self, y_true, y_pred, y_prob=None):
        metrics = {
            'accuracy': accuracy_score(y_true, y_pred),
            'precision': precision_score(y_true, y_pred, average='weighted'),
            'recall': recall_score(y_true, y_pred, average='weighted'),
            'f1': f1_score(y_true, y_pred, average='weighted'),
        }

        if y_prob is not None:
            metrics['roc_auc'] = roc_auc_score(y_true, y_prob, multi_class='ovr')

        # 混淆矩阵
        cm = confusion_matrix(y_true, y_pred)
        metrics['confusion_matrix'] = cm.tolist()

        # 分类报告
        metrics['report'] = classification_report(y_true, y_pred)

        return metrics
```

### 2.2 回归指标

```python
from sklearn.metrics import (
    mean_absolute_error, mean_squared_error, r2_score
)
import numpy as np

class RegressionEvaluator:
    def evaluate(self, y_true, y_pred):
        mse = mean_squared_error(y_true, y_pred)
        return {
            'mae': mean_absolute_error(y_true, y_pred),
            'mse': mse,
            'rmse': np.sqrt(mse),
            'r2': r2_score(y_true, y_pred),
            'mape': np.mean(np.abs((y_true - y_pred) / y_true)) * 100,
        }
```

### 2.3 排序指标

```python
class RankingEvaluator:
    def ndcg_at_k(self, y_true, y_score, k=10):
        # 归一化折损累计增益
        order = np.argsort(y_score)[::-1]
        y_true_sorted = np.take(y_true, order[:k])

        gains = 2 ** y_true_sorted - 1
        discounts = np.log2(np.arange(len(y_true_sorted)) + 2)
        dcg = np.sum(gains / discounts)

        ideal_order = np.argsort(y_true)[::-1]
        ideal_sorted = np.take(y_true, ideal_order[:k])
        ideal_gains = 2 ** ideal_sorted - 1
        idcg = np.sum(ideal_gains / discounts)

        return dcg / idcg if idcg > 0 else 0

    def map_at_k(self, y_true, y_score, k=10):
        # 平均精度均值
        order = np.argsort(y_score)[::-1]
        y_true_sorted = np.take(y_true, order[:k])

        precision_at_k = []
        num_relevant = 0
        for i, rel in enumerate(y_true_sorted):
            if rel > 0:
                num_relevant += 1
                precision_at_k.append(num_relevant / (i + 1))

        return np.mean(precision_at_k) if precision_at_k else 0
```

## 3. 交叉验证

### 3.1 K 折交叉验证

```python
from sklearn.model_selection import KFold, StratifiedKFold, cross_val_score

class CrossValidator:
    def k_fold(self, model, X, y, k=5):
        kf = KFold(n_splits=k, shuffle=True, random_state=42)
        scores = cross_val_score(model, X, y, cv=kf, scoring='f1_weighted')
        return {
            'mean': scores.mean(),
            'std': scores.std(),
            'scores': scores.tolist(),
        }

    def stratified_k_fold(self, model, X, y, k=5):
        skf = StratifiedKFold(n_splits=k, shuffle=True, random_state=42)
        scores = cross_val_score(model, X, y, cv=skf, scoring='f1_weighted')
        return {
            'mean': scores.mean(),
            'std': scores.std(),
            'scores': scores.tolist(),
        }
```

### 3.2 时间序列交叉验证

```python
from sklearn.model_selection import TimeSeriesSplit

class TimeSeriesCrossValidator:
    def validate(self, model, X, y, n_splits=5):
        tscv = TimeSeriesSplit(n_splits=n_splits)
        scores = []

        for train_idx, test_idx in tscv.split(X):
            X_train, X_test = X.iloc[train_idx], X.iloc[test_idx]
            y_train, y_test = y.iloc[train_idx], y.iloc[test_idx]

            model.fit(X_train, y_train)
            score = model.score(X_test, y_test)
            scores.append(score)

        return {
            'mean': np.mean(scores),
            'std': np.std(scores),
            'scores': scores,
        }
```

from sklearn.model_selection import LeaveOneOut

class LeaveOneOutValidator:
    def validate(self, model, X, y):

        scores = cross_val_score(model, X, y, cv=loo, scoring='accuracy')

            'mean': scores.mean(),
            'std': scores.std(),

## 4. 超参数调优

### 4.1 网格搜索

```python
from sklearn.model_selection import GridSearchCV

class GridSearchTuner:
    def __init__(self, param_grid, scoring='f1_weighted', cv=5):
        self.param_grid = param_grid
        self.scoring = scoring
        self.cv = cv

    def tune(self, model, X, y):
        grid_search = GridSearchCV(
            estimator=model,
            param_grid=self.param_grid,
            scoring=self.scoring,
            cv=self.cv,
            n_jobs=-1,
            verbose=1,
        )
        grid_search.fit(X, y)

        return {
            'best_params': grid_search.best_params_,
            'best_score': grid_search.best_score_,
            'best_estimator': grid_search.best_estimator_,
            'cv_results': grid_search.cv_results_,
        }
```

## 5. 学习曲线

### 5.1 学习曲线绘制

```python
from sklearn.model_selection import learning_curve

class LearningCurveAnalyzer:
    def plot(self, model, X, y, train_sizes=None):
        if train_sizes is None:
            train_sizes = np.linspace(0.1, 1.0, 10)

        train_sizes, train_scores, val_scores = learning_curve(
            model, X, y,
            train_sizes=train_sizes,
            cv=5,
            scoring='f1_weighted',
            n_jobs=-1,
        )

        return {
            'train_sizes': train_sizes.tolist(),
            'train_mean': train_scores.mean(axis=1).tolist(),
            'train_std': train_scores.std(axis=1).tolist(),
            'val_mean': val_scores.mean(axis=1).tolist(),
            'val_std': val_scores.std(axis=1).tolist(),
        }

    def diagnose(self, train_score, val_score):
        if train_score > 0.9 and val_score < 0.7:
            return '过拟合：训练集表现好，验证集表现差'
        elif train_score < 0.7 and val_score < 0.7:
            return '欠拟合：训练集和验证集表现都差'
        elif abs(train_score - val_score) < 0.05:
            return '拟合良好：训练集和验证集表现接近'
        else:
            return '需要更多数据或调整模型'
```

## 7. 模型选择

### 7.1 模型对比

```python
class ModelComparator:
    def compare(self, models, X, y, cv=5):
        results = {}

        for name, model in models.items():
            scores = cross_val_score(model, X, y, cv=cv, scoring='f1_weighted')
            results[name] = {
                'mean': scores.mean(),
                'std': scores.std(),
                'scores': scores.tolist(),
            }

        # 排序
        sorted_results = sorted(
            results.items(),
            key=lambda x: x[1]['mean'],
            reverse=True,
        )

        return sorted_results
```
