# 算法设计-时序异常检测与预警

> 本文档是家健镜系统时序异常检测与预警的完整设计说明，覆盖统计方法、机器学习方法、深度学习方法、预警规则、可视化。

## 1. 概述

### 1.1 设计目标

1. 异常检测准确率 > 90%
2. 误报率 < 5%
3. 实时检测延迟 < 10秒
4. 支持多指标联合检测
5. 可解释的预警原因

### 1.2 异常类型

| 类型 | 说明 | 示例 |
| --- | --- | --- |
| 点异常 | 单个数据点异常 | 血压突然升高 |
| 上下文异常 | 特定上下文下异常 | 夜间心率过高 |
| 集体异常 | 连续数据异常 | 连续3天血糖偏高 |
| 趋势异常 | 趋势变化异常 | 体重持续下降 |
| 季节性异常 | 违反季节规律 | 夏季血压异常 |

## 2. 统计方法

### 2.1 3σ 原则

```python
import numpy as np

class ThreeSigmaDetector:
    def __init__(self, window_size: int = 30):
        self.window_size = window_size
        self.history = []

    def detect(self, value: float) -> dict:
        self.history.append(value)
        if len(self.history) < self.window_size:
            return {"anomaly": False, "reason": "数据不足"}

        recent = self.history[-self.window_size:]
        mean = np.mean(recent)
        std = np.std(recent)

        if std == 0:
            return {"anomaly": False}

        z_score = abs(value - mean) / std
        is_anomaly = z_score > 3

        return {
            "anomaly": is_anomaly,
            "z_score": z_score,
            "mean": mean,
            "std": std,
            "threshold": mean + 3 * std,
            "reason": f"偏离均值 {z_score:.1f} 个标准差" if is_anomaly else None,
        }
```

### 2.2 IQR 方法

```python
class IQRDetector:
    def __init__(self, window_size: int = 30, k: float = 1.5):
        self.window_size = window_size
        self.k = k
        self.history = []

    def detect(self, value: float) -> dict:
        self.history.append(value)
        if len(self.history) < self.window_size:
            return {"anomaly": False}

        recent = self.history[-self.window_size:]
        q1 = np.percentile(recent, 25)
        q3 = np.percentile(recent, 75)
        iqr = q3 - q1

        lower_bound = q1 - self.k * iqr
        upper_bound = q3 + self.k * iqr

        is_anomaly = value < lower_bound or value > upper_bound

        return {
            "anomaly": is_anomaly,
            "q1": q1,
            "q3": q3,
            "iqr": iqr,
            "lower_bound": lower_bound,
            "upper_bound": upper_bound,
        }
```

### 2.3 移动平均

```python
class MovingAverageDetector:
    def __init__(self, window_size: int = 7, threshold: float = 2.0):
        self.window_size = window_size
        self.threshold = threshold
        self.history = []

    def detect(self, value: float) -> dict:
        self.history.append(value)
        if len(self.history) < self.window_size + 1:
            return {"anomaly": False}

        # 计算移动平均
        ma = np.mean(self.history[-(self.window_size + 1):-1])
        std = np.std(self.history[-(self.window_size + 1):-1])

        deviation = abs(value - ma)
        is_anomaly = deviation > self.threshold * std

        return {
            "anomaly": is_anomaly,
            "moving_average": ma,
            "deviation": deviation,
            "threshold": self.threshold * std,
        }
```

## 3. 机器学习方法

### 3.1 Isolation Forest

```python
from sklearn.ensemble import IsolationForest

class IsolationForestDetector:
    def __init__(self, contamination: float = 0.05, n_estimators: int = 100):
        self.model = IsolationForest(
            contamination=contamination,
            n_estimators=n_estimators,
            random_state=42,
        )
        self.fitted = False

    def fit(self, X: np.ndarray):
        self.model.fit(X)
        self.fitted = True

    def detect(self, X: np.ndarray) -> dict:
        if not self.fitted:
            raise RuntimeError("模型未训练")

        predictions = self.model.predict(X)
        scores = self.model.decision_function(X)

        return {
            "anomaly": predictions == -1,
            "anomaly_score": scores,
            "anomaly_indices": np.where(predictions == -1)[0],
        }
```

## 4. 深度学习方法

### 4.1 Autoencoder

```python
import torch
import torch.nn as nn

class Autoencoder(nn.Module):
    def __init__(self, input_dim: int, hidden_dim: int = 64, latent_dim: int = 16):
        super().__init__()
        self.encoder = nn.Sequential(
            nn.Linear(input_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, latent_dim),
        )
        self.decoder = nn.Sequential(
            nn.Linear(latent_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, input_dim),
        )

    def forward(self, x):
        encoded = self.encoder(x)
        decoded = self.decoder(encoded)
        return decoded

class AutoencoderDetector:
    def __init__(self, model: Autoencoder, threshold: float = 0.1):
        self.model = model
        self.threshold = threshold
        self.criterion = nn.MSELoss()

    def detect(self, X: torch.Tensor) -> dict:
        self.model.eval()
        with torch.no_grad():
            reconstructed = self.model(X)
            errors = torch.mean((X - reconstructed) ** 2, dim=1)

        is_anomaly = errors > self.threshold
        return {
            "anomaly": is_anomaly.numpy(),
            "reconstruction_error": errors.numpy(),
            "threshold": self.threshold,
        }
```

### 4.2 LSTM 预测

```python
class LSTMPredictor(nn.Module):
    def __init__(self, input_size: int, hidden_size: int = 64, num_layers: int = 2):
        super().__init__()
        self.lstm = nn.LSTM(input_size, hidden_size, num_layers, batch_first=True)
        self.fc = nn.Linear(hidden_size, input_size)

    def forward(self, x):
        lstm_out, _ = self.lstm(x)
        output = self.fc(lstm_out[:, -1, :])
        return output

class LSTMAnomalyDetector:
    def __init__(self, model: LSTMPredictor, threshold: float = 0.5):
        self.model = model
        self.threshold = threshold

    def detect(self, sequence: np.ndarray) -> dict:
        self.model.eval()
        with torch.no_grad():
            input_tensor = torch.FloatTensor(sequence).unsqueeze(0)
            prediction = self.model(input_tensor)
            actual = sequence[-1]
            error = np.mean((prediction.numpy() - actual) ** 2)

        is_anomaly = error > self.threshold
        return {
            "anomaly": is_anomaly,
            "prediction_error": error,
            "predicted": prediction.numpy(),
            "actual": actual,
        }
```

### 4.3 Transformer 异常检测

```python
class TransformerAnomalyDetector(nn.Module):
    def __init__(self, input_dim: int, d_model: int = 128, nhead: int = 8, num_layers: int = 3):
        super().__init__()
        self.embedding = nn.Linear(input_dim, d_model)
        self.pos_encoder = nn.Parameter(torch.randn(1, 100, d_model) * 0.02)
        encoder_layer = nn.TransformerEncoderLayer(d_model=d_model, nhead=nhead)
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers=num_layers)
        self.decoder = nn.Linear(d_model, input_dim)

    def forward(self, x):
        x = self.embedding(x) + self.pos_encoder[:, :x.size(1), :]
        encoded = self.transformer(x)
        return self.decoder(encoded[:, -1, :])
```

class CorrelationDetector:
    def __init__(self, window_size: int = 30):
        self.window_size = window_size
        self.history = {}

    def add_data(self, metric: str, value: float):
        if metric not in self.history:

### 5.2 多变量异常评分

```python
class MultivariateScorer:
    def __init__(self, weights: dict):
        self.weights = weights

    def score(self, anomaly_scores: dict) -> float:
        total_weight = sum(self.weights.values())
        weighted_score = sum(
            anomaly_scores.get(metric, 0) * weight
            for metric, weight in self.weights.items()
        )
        return weighted_score / total_weight

    def is_anomaly(self, anomaly_scores: dict, threshold: float = 0.7) -> bool:
        return self.score(anomaly_scores) > threshold
```

## 6. 预警规则

### 6.1 阈值预警

```python
class ThresholdAlert:
    def __init__(self, rules: list):
        self.rules = rules

    def check(self, metric: str, value: float) -> list[dict]:
        alerts = []
        for rule in self.rules:
            if rule['metric'] != metric:
                continue

            if rule['type'] == 'above' and value > rule['threshold']:
                alerts.append({
                    "level": rule['level'],
                    "message": f"{metric} 超过阈值 {rule['threshold']}",
                    "value": value,
                    "threshold": rule['threshold'],
                })
            elif rule['type'] == 'below' and value < rule['threshold']:
                alerts.append({
                    "level": rule['level'],
                    "message": f"{metric} 低于阈值 {rule['threshold']}",
                    "value": value,
                    "threshold": rule['threshold'],
                })

        return alerts
```

### 6.2 持续异常预警

```python
class PersistentAnomalyAlert:
    def __init__(self, consecutive_count: int = 3):
        self.consecutive_count = consecutive_count
        self.anomaly_streak = {}

    def check(self, metric: str, is_anomaly: bool) -> dict | None:
        if metric not in self.anomaly_streak:
            self.anomaly_streak[metric] = 0

        if is_anomaly:
            self.anomaly_streak[metric] += 1
        else:
            self.anomaly_streak[metric] = 0

        if self.anomaly_streak[metric] >= self.consecutive_count:
            return {
                "metric": metric,
                "streak": self.anomaly_streak[metric],
                "message": f"{metric} 连续 {self.anomaly_streak[metric]} 次异常",
            }

        return None
```

### 6.3 预警分级

```python
class AlertLevel:
    INFO = "info"        # 提示
    WARNING = "warning"  # 警告
    CRITICAL = "critical"  # 严重

class AlertManager:
    def __init__(self):
        self.alert_channels = {
            AlertLevel.INFO: ["app_push"],
            AlertLevel.WARNING: ["app_push", "sms"],
            AlertLevel.CRITICAL: ["app_push", "sms", "phone_call"],
        }

    async def send_alert(self, alert: dict):
        channels = self.alert_channels.get(alert['level'], [])
        for channel in channels:
            await self._send_to_channel(channel, alert)

    async def _send_to_channel(self, channel: str, alert: dict):
        if channel == "app_push":
            await self._push_service.send(alert)
        elif channel == "sms":
            await self._sms_service.send(alert)
        elif channel == "phone_call":
            await self._call_service.send(alert)
```
