# 算法设计-异常检测

> 本文档是家健镜系统异常检测算法的完整设计说明，覆盖体征异常、行为异常、用药异常、设备异常。

## 1. 异常检测概述

### 1.1 目标

自动识别家庭成员健康数据和系统运行中的异常模式，及时预警潜在问题。

### 1.2 异常类型

| 类型 | 说明 | 检测方法 |
| --- | --- | --- |
| 体征异常 | 血压、血糖、心率超出正常范围 | 阈值法 |
| 趋势异常 | 体征数据趋势异常 | 统计过程控制 |
| 行为异常 | 作息、活动异常 | 行为模式匹配 |
| 用药异常 | 漏服、错服、过量 | 规则匹配 |
| 设备异常 | 设备离线、数据异常 | 状态监控 |

## 2. 体征异常检测

### 2.1 阈值检测

```python
class VitalAnomalyDetector:
    NORMAL_RANGES = {
        "blood_pressure_systolic": (90, 140),
        "blood_pressure_diastolic": (60, 90),
        "blood_glucose_fasting": (3.9, 6.1),
        "blood_glucose_postprandial": (3.9, 7.8),
        "heart_rate": (60, 100),
        "temperature": (36.0, 37.3),
        "weight_bmi": (18.5, 24.0),
    }

    def detect(self, record):
        anomalies = []
        vital_type = record.type
        value = record.value

        if vital_type in self.NORMAL_RANGES:
            low, high = self.NORMAL_RANGES[vital_type]

            if value < low:
                anomalies.append({
                    "type": "below_normal",
                    "vital": vital_type,
                    "value": value,
                    "normal_range": (low, high),
                    "severity": self._get_severity(value, low, high, "below"),
                })
            elif value > high:
                anomalies.append({
                    "type": "above_normal",
                    "vital": vital_type,
                    "value": value,
                    "normal_range": (low, high),
                    "severity": self._get_severity(value, low, high, "above"),
                })

        return anomalies

    def _get_severity(self, value, low, high, direction):
        if direction == "above":
            deviation = (value - high) / high
        else:
            deviation = (low - value) / low

        if deviation > 0.3:
            return "critical"
        elif deviation > 0.15:
            return "high"
        else:
            return "medium"
```

### 2.2 趋势检测

```python
class TrendAnomalyDetector:
    def detect(self, records, window=7):
        if len(records) < window:
            return []

        anomalies = []
        recent = records[-window:]
        values = [r.value for r in recent]

        # 计算趋势
        trend = self._calculate_trend(values)

        # 检测持续上升/下降
        if self._is_consistent_trend(values, "increasing"):
            anomalies.append({
                "type": "consistent_increase",
                "vital": records[0].type,
                "trend": trend,
                "duration": window,
                "severity": "medium",
            })
        elif self._is_consistent_trend(values, "decreasing"):
            anomalies.append({
                "type": "consistent_decrease",
                "vital": records[0].type,
                "trend": trend,
                "duration": window,
                "severity": "medium",
            })

        # 检测突变
        if len(values) >= 2:
            change = abs(values[-1] - values[-2]) / values[-2]
            if change > 0.2:
                anomalies.append({
                    "type": "sudden_change",
                    "vital": records[0].type,
                    "change_percent": round(change * 100, 1),
                    "severity": "high",
                })

        return anomalies

    def _calculate_trend(self, values):
        import numpy as np
        x = np.arange(len(values))
        slope = np.polyfit(x, values, 1)[0]
        return round(float(slope), 4)

    def _is_consistent_trend(self, values, direction):
        for i in range(1, len(values)):
            if direction == "increasing" and values[i] <= values[i-1]:
                return False
            if direction == "decreasing" and values[i] >= values[i-1]:
                return False
        return True
```

### 2.3 统计过程控制

```python
class SPCDetector:
    def detect(self, records):
        if len(records) < 20:
            return []

        values = [r.value for r in records]
        mean = sum(values) / len(values)
        std = (sum((v - mean) ** 2 for v in values) / len(values)) ** 0.5

        anomalies = []
        latest = values[-1]

        # 3-sigma 规则
        if abs(latest - mean) > 3 * std:
            anomalies.append({
                "type": "out_of_control",
                "vital": records[0].type,
                "value": latest,
                "mean": round(mean, 2),
                "std": round(std, 2),
                "sigma_deviation": round(abs(latest - mean) / std, 2),
                "severity": "high",
            })

        return anomalies
```

## 3. 行为异常检测

### 3.1 作息异常

```python
class SleepAnomalyDetector:
    def detect(self, sleep_records):
        if len(sleep_records) < 7:
            return []

        anomalies = []

        # 计算平均入睡时间
        sleep_times = [r.bed_time for r in sleep_records]
        avg_sleep_time = self._average_time(sleep_times)

        # 检测入睡时间偏差
        latest = sleep_records[-1]
        deviation = self._time_deviation(latest, avg_sleep_time)

        if deviation > 120:  # 偏差超过 2 小时
            anomalies.append({
                "type": "sleep_time_deviation",
                "expected": avg_sleep_time,
                "actual": latest,
                "deviation_minutes": deviation,
                "severity": "medium",
            })

        # 检测睡眠时长异常
        latest_duration = sleep_records[-1].duration
        if latest_duration < 4:
            anomalies.append({
                "type": "insufficient_sleep",
                "duration": latest_duration,
                "severity": "high",
            })
        elif latest_duration > 10:
            anomalies.append({
                "type": "excessive_sleep",
                "duration": latest_duration,
                "severity": "medium",
            })

        return anomalies
```

## 4. 用药异常检测

### 4.1 漏服检测

```python
class MedicationAnomalyDetector:
    def detect(self, records, schedule):
        anomalies = []

        # 计算漏服率
        total = len(records)
        missed = sum(1 for r in records if r.status == "missed")
        missed_rate = missed / total if total > 0 else 0

        if missed_rate > 0.3:
            anomalies.append({
                "type": "high_missed_rate",
                "missed_rate": round(missed_rate, 2),
                "total_doses": total,
                "missed_doses": missed,
                "severity": "high",
            })

        # 检测连续漏服
        consecutive_missed = 0
        max_consecutive = 0
        for r in reversed(records):
            if r.status == "missed":
                consecutive_missed += 1
                max_consecutive = max(max_consecutive, consecutive_missed)
            else:
                break

        if max_consecutive >= 3:
            anomalies.append({
                "type": "consecutive_missed",
                "count": max_consecutive,
                "severity": "high",
            })

        return anomalies
```

## 5. 设备异常检测

### 5.1 设备状态监控

```python
class DeviceAnomalyDetector:
    def detect(self, device):
        anomalies = []

        # 离线检测
        if device.status == "offline":
            offline_duration = (datetime.now() - device.last_seen).total_seconds() / 3600
            if offline_duration > 24:
                anomalies.append({
                    "type": "device_offline",
                    "device_id": device.id,
                    "offline_hours": round(offline_duration, 1),
                    "severity": "medium",
                })

        # 电量低
        if device.battery_level is not None and device.battery_level < 20:
            anomalies.append({
                "type": "low_battery",
                "device_id": device.id,
                "battery_level": device.battery_level,
                "severity": "low",
            })

        # 数据异常
        if device.data_quality < 0.8:
            anomalies.append({
                "type": "poor_data_quality",
                "device_id": device.id,
                "quality_score": device.data_quality,
                "severity": "medium",
            })

        return anomalies
```
