## 2. 数据清洗

### 2.1 缺失值处理

```python
import pandas as pd
import numpy as np

class MissingValueHandler:
    def __init__(self):
        self.fill_values = {}

    def fit(self, df: pd.DataFrame):
        # 数值型：中位数
        numeric_cols = df.select_dtypes(include=[np.number]).columns
        self.fill_values["numeric"] = df[numeric_cols].median()

        # 类别型：众数
        categorical_cols = df.select_dtypes(include=["object"]).columns
        self.fill_values["categorical"] = df[categorical_cols].mode().iloc[0]

    def transform(self, df: pd.DataFrame) -> pd.DataFrame:
        df = df.copy()

        # 数值型填充
        for col, value in self.fill_values["numeric"].items():
            df[col] = df[col].fillna(value)

        # 类别型填充
        for col, value in self.fill_values["categorical"].items():
            df[col] = df[col].fillna(value)

        # 时间序列：前向填充
        if "timestamp" in df.columns:
            df = df.sort_values("timestamp")
            df = df.ffill()

        return df

    def fit_transform(self, df: pd.DataFrame) -> pd.DataFrame:
        self.fit(df)
        return self.transform(df)
```

### 2.2 异常值处理

```python
class OutlierHandler:
    def __init__(self, method: str = "iqr", threshold: float = 1.5):
        self.method = method
        self.threshold = threshold
        self.bounds = {}

    def fit(self, df: pd.DataFrame, columns: list[str]):
        for col in columns:
            if self.method == "iqr":
                Q1 = df[col].quantile(0.25)
                Q3 = df[col].quantile(0.75)
                IQR = Q3 - Q1
                self.bounds[col] = (
                    Q1 - self.threshold * IQR,
                    Q3 + self.threshold * IQR,
                )
            elif self.method == "zscore":
                mean = df[col].mean()
                std = df[col].std()
                self.bounds[col] = (
                    mean - self.threshold * std,
                    mean + self.threshold * std,
                )

    def transform(self, df: pd.DataFrame) -> pd.DataFrame:
        df = df.copy()
        for col, (lower, upper) in self.bounds.items():
            # 截断（Winsorize）
            df[col] = df[col].clip(lower, upper)
        return df
```

## 3. 特征构造

### 3.1 时间特征

```python
class TimeFeatureEngineer:
    def transform(self, df: pd.DataFrame, time_col: str) -> pd.DataFrame:
        df = df.copy()
        df[time_col] = pd.to_datetime(df[time_col])

        # 基础时间特征
        df["year"] = df[time_col].dt.year
        df["month"] = df[time_col].dt.month
        df["day"] = df[time_col].dt.day
        df["hour"] = df[time_col].dt.hour
        df["minute"] = df[time_col].dt.minute
        df["dayofweek"] = df[time_col].dt.dayofweek
        df["dayofyear"] = df[time_col].dt.dayofyear
        df["weekofyear"] = df[time_col].dt.isocalendar().week

        # 周期性特征（正弦余弦编码）
        df["month_sin"] = np.sin(2 * np.pi * df["month"] / 12)
        df["month_cos"] = np.cos(2 * np.pi * df["month"] / 12)
        df["hour_sin"] = np.sin(2 * np.pi * df["hour"] / 24)
        df["hour_cos"] = np.cos(2 * np.pi * df["hour"] / 24)
        df["dayofweek_sin"] = np.sin(2 * np.pi * df["dayofweek"] / 7)
        df["dayofweek_cos"] = np.cos(2 * np.pi * df["dayofweek"] / 7)

        # 是否周末
        df["is_weekend"] = df["dayofweek"].isin([5, 6]).astype(int)

        # 时间段
        df["time_period"] = pd.cut(
            df["hour"],
            bins=[0, 6, 12, 18, 24],
            labels=["凌晨", "上午", "下午", "晚上"],
            include_lowest=True,
        )

        return df
```

### 3.2 统计特征

```python
class StatisticalFeatureEngineer:
    def transform(self, df: pd.DataFrame, group_col: str, value_col: str) -> pd.DataFrame:
        df = df.copy()

        # 分组统计
        group_stats = df.groupby(group_col)[value_col].agg([
            "mean", "std", "min", "max", "median", "count",
        ])
        group_stats.columns = [f"{value_col}_{col}" for col in group_stats.columns]

        df = df.merge(group_stats, left_on=group_col, right_index=True, how="left")

        # 滑动窗口特征
        df = df.sort_values(["user_id", "timestamp"])
        df[f"{value_col}_rolling_mean_7"] = (
            df.groupby("user_id")[value_col]
            .rolling(window=7, min_periods=1)
            .mean()
            .reset_index(level=0, drop=True)
        )
        df[f"{value_col}_rolling_std_7"] = (
            df.groupby("user_id")[value_col]
            .rolling(window=7, min_periods=1)
            .std()
            .reset_index(level=0, drop=True)
        )

        # 变化率特征
        df[f"{value_col}_diff"] = df.groupby("user_id")[value_col].diff()
        df[f"{value_col}_pct_change"] = df.groupby("user_id")[value_col].pct_change()

        return df
```

class CrossFeatureEngineer:
    def transform(self, df: pd.DataFrame) -> pd.DataFrame:
        df = df.copy()

        # 数值交叉
        df["bmi"] = df["weight"] / (df["height"] / 100) ** 2

        df["season_weather"] = df["season"].astype(str) + "_" + df["weather"].astype(str)

        return df

## 4. 特征编码

### 4.1 类别编码

```python
from sklearn.preprocessing import LabelEncoder, OneHotEncoder

class CategoricalEncoder:
    def __init__(self):
        self.label_encoders = {}
        self.onehot_encoders = {}

    def fit(self, df: pd.DataFrame, high_cardinality_cols: list, low_cardinality_cols: list):
        # 高基数：Label Encoding
        for col in high_cardinality_cols:
            le = LabelEncoder()
            le.fit(df[col].astype(str))
            self.label_encoders[col] = le

        # 低基数：One-Hot Encoding
        for col in low_cardinality_cols:
            ohe = OneHotEncoder(sparse=False, handle_unknown="ignore")
            ohe.fit(df[[col]])
            self.onehot_encoders[col] = ohe

    def transform(self, df: pd.DataFrame) -> pd.DataFrame:
        df = df.copy()

        for col, le in self.label_encoders.items():
            df[col] = le.transform(df[col].astype(str))

        for col, ohe in self.onehot_encoders.items():
            encoded = ohe.transform(df[[col]])
            feature_names = ohe.get_feature_names_out([col])
            encoded_df = pd.DataFrame(encoded, columns=feature_names, index=df.index)
            df = pd.concat([df.drop(col, axis=1), encoded_df], axis=1)

        return df
```

### 4.2 目标编码

```python
class TargetEncoder:
    def __init__(self, smoothing: float = 10):
        self.smoothing = smoothing
        self.encoding_map = {}

    def fit(self, df: pd.DataFrame, col: str, target: str):
        global_mean = df[target].mean()
        agg = df.groupby(col)[target].agg(["mean", "count"])

        # 平滑处理
        agg["encoded"] = (
            (agg["count"] * agg["mean"] + self.smoothing * global_mean)
            / (agg["count"] + self.smoothing)
        )
        self.encoding_map[col] = agg["encoded"].to_dict()
        self.global_mean = global_mean

    def transform(self, df: pd.DataFrame) -> pd.DataFrame:
        df = df.copy()
        for col, mapping in self.encoding_map.items():
            df[f"{col}_target_encoded"] = df[col].map(mapping).fillna(self.global_mean)
        return df
```

### 4.3 数值标准化

```python
from sklearn.preprocessing import StandardScaler, MinMaxScaler

class NumericalScaler:
    def __init__(self, method: str = "standard"):
        self.method = method
        self.scalers = {}

    def fit(self, df: pd.DataFrame, columns: list[str]):
        for col in columns:
            if self.method == "standard":
                scaler = StandardScaler()
            elif self.method == "minmax":
                scaler = MinMaxScaler()
            scaler.fit(df[[col]])
            self.scalers[col] = scaler

    def transform(self, df: pd.DataFrame) -> pd.DataFrame:
        df = df.copy()
        for col, scaler in self.scalers.items():
            df[col] = scaler.transform(df[[col]])
        return df
```

## 5. 特征选择

### 5.1 过滤法

```python
from sklearn.feature_selection import SelectKBest, f_classif, mutual_info_classif

class FilterFeatureSelector:
    def __init__(self, k: int = 20, method: str = "f_classif"):
        self.k = k
        self.method = method
        self.selector = None

    def fit(self, X: pd.DataFrame, y: pd.Series):
        if self.method == "f_classif":
            score_func = f_classif
        elif self.method == "mutual_info":
            score_func = mutual_info_classif

        self.selector = SelectKBest(score_func=score_func, k=self.k)
        self.selector.fit(X, y)
        self.selected_features = X.columns[self.selector.get_support()]

    def transform(self, X: pd.DataFrame) -> pd.DataFrame:
        return X[self.selected_features]
```

### 5.2 包裹法

```python
from sklearn.feature_selection import RFE
from sklearn.ensemble import RandomForestClassifier

class WrapperFeatureSelector:
    def __init__(self, n_features: int = 20):
        self.n_features = n_features
        self.selector = RFE(
            estimator=RandomForestClassifier(n_estimators=100),
            n_features_to_select=n_features,
        )

    def fit(self, X: pd.DataFrame, y: pd.Series):
        self.selector.fit(X, y)
        self.selected_features = X.columns[self.selector.support_]

    def transform(self, X: pd.DataFrame) -> pd.DataFrame:
        return X[self.selected_features]
```

### 5.3 嵌入法

```python
class EmbeddedFeatureSelector:
    def __init__(self, threshold: float = 0.01):
        self.threshold = threshold
        self.model = RandomForestClassifier(n_estimators=100)

    def fit(self, X: pd.DataFrame, y: pd.Series):
        self.model.fit(X, y)
        importances = pd.Series(self.model.feature_importances_, index=X.columns)
        self.selected_features = importances[importances > self.threshold].index

    def transform(self, X: pd.DataFrame) -> pd.DataFrame:
        return X[self.selected_features]
```

## 6. 数据增强

### 6.1 时间序列增强

```python
class TimeSeriesAugmentor:
    def jitter(self, x: np.ndarray, sigma: float = 0.03) -> np.ndarray:
        return x + np.random.normal(loc=0, scale=sigma, size=x.shape)

    def scaling(self, x: np.ndarray, sigma: float = 0.1) -> np.ndarray:
        factor = np.random.normal(loc=1, scale=sigma)
        return x * factor

    def permutation(self, x: np.ndarray, n_segments: int = 5) -> np.ndarray:
        segments = np.array_split(x, n_segments)
        np.random.shuffle(segments)
        return np.concatenate(segments)

    def time_warp(self, x: np.ndarray, sigma: float = 0.2) -> np.ndarray:
        # 简单的时间扭曲
        warp = np.cumsum(np.random.normal(1, sigma, len(x)))
        warp = warp / warp[-1] * (len(x) - 1)
        return np.interp(warp, np.arange(len(x)), x)
```

## 7. 预处理流水线

```python
from sklearn.pipeline import Pipeline

class PreprocessingPipeline:
    def __init__(self):
        self.pipeline = Pipeline([
            ("missing_handler", MissingValueHandler()),
            ("outlier_handler", OutlierHandler()),
            ("time_features", TimeFeatureEngineer()),
            ("stat_features", StatisticalFeatureEngineer()),
            ("cross_features", CrossFeatureEngineer()),
            ("categorical_encoder", CategoricalEncoder()),
            ("numerical_scaler", NumericalScaler()),
            ("feature_selector", FilterFeatureSelector(k=30)),
        ])

    def fit(self, X, y=None):
        self.pipeline.fit(X, y)

    def transform(self, X):
        return self.pipeline.transform(X)

    def fit_transform(self, X, y=None):
        return self.pipeline.fit_transform(X, y)
```
