## 2. 数据预处理

### 2.1 数据清洗

```python
class TimeSeriesPreprocessor:
    def clean(self, series):
        # 1. 去除异常值
        series = self._remove_outliers(series)

        # 2. 填充缺失值
        series = self._fill_missing(series)

        # 3. 重采样
        series = self._resample(series)

        # 4. 归一化
        series = self._normalize(series)

        return series

    def _remove_outliers(self, series):
        # 使用 3-sigma 原则
        mean = series.mean()
        std = series.std()
        return series[(series >= mean - 3 * std) & (series <= mean + 3 * std)]

    def _fill_missing(self, series):
        # 线性插值
        return series.interpolate(method='linear')

    def _resample(self, series):
        # 按天重采样，取平均值
        return series.resample('D').mean()

    def _normalize(self, series):
        # Min-Max 归一化
        return (series - series.min()) / (series.max() - series.min())
```

### 2.2 特征工程

```python
class FeatureExtractor:
    def extract(self, series):
        features = {}

        # 时间特征
        features['day_of_week'] = series.index.dayofweek
        features['month'] = series.index.month
        features['is_weekend'] = (series.index.dayofweek >= 5).astype(int)

        # 统计特征
        features['rolling_mean_7'] = series.rolling(7).mean()
        features['rolling_std_7'] = series.rolling(7).std()
        features['rolling_mean_30'] = series.rolling(30).mean()

        # 滞后特征
        for lag in [1, 3, 7, 14, 30]:
            features[f'lag_{lag}'] = series.shift(lag)

        # 差分特征
        features['diff_1'] = series.diff(1)
        features['diff_7'] = series.diff(7)

        return pd.DataFrame(features)
```

## 3. 预测模型

### 3.1 统计模型

```python
from statsmodels.tsa.arima.model import ARIMA
from statsmodels.tsa.holtwinters import ExponentialSmoothing

class ARIMAPredictor:
    def __init__(self, order=(1, 1, 1)):
        self.order = order
        self.model = None

    def fit(self, series):
        self.model = ARIMA(series, order=self.order).fit()

    def predict(self, steps=7):
        forecast = self.model.forecast(steps=steps)
        return forecast

class ESPredictor:
    def fit(self, series):
        self.model = ExponentialSmoothing(
            series,
            trend='add',
            seasonal='add',
            seasonal_periods=7,
        ).fit()

    def predict(self, steps=7):
        return self.model.forecast(steps)
```

### 3.2 机器学习模型

```python
from sklearn.ensemble import GradientBoostingRegressor
from sklearn.model_selection import TimeSeriesSplit

class GBRTTPredictor:
    def __init__(self):
        self.model = GradientBoostingRegressor(
            n_estimators=100,
            max_depth=3,
            learning_rate=0.1,
        )
        self.feature_extractor = FeatureExtractor()

    def fit(self, series):
        features = self.feature_extractor.extract(series)
        target = series.shift(-1)  # 预测下一个值

        # 去除 NaN
        valid = features.notna().all(axis=1) & target.notna()
        self.model.fit(features[valid], target[valid])

    def predict(self, series, steps=7):
        predictions = []
        current = series.copy()

        for _ in range(steps):
            features = self.feature_extractor.extract(current)
            next_value = self.model.predict(features.iloc[[-1]])[0]
            predictions.append(next_value)
            current = pd.concat([current, pd.Series([next_value])])

        return predictions
```

### 3.3 深度学习模型

```python
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense, Dropout

class LSTMPredictor:
    def __init__(self, lookback=30):
        self.lookback = lookback
        self.model = self._build_model()

    def _build_model(self):
        model = Sequential([
            LSTM(64, return_sequences=True, input_shape=(self.lookback, 1)),
            Dropout(0.2),
            LSTM(32),
            Dropout(0.2),
            Dense(16, activation='relu'),
            Dense(1),
        ])
        model.compile(optimizer='adam', loss='mse')
        return model

    def _create_sequences(self, data):
        X, y = [], []
        for i in range(len(data) - self.lookback):
            X.append(data[i:i + self.lookback])
            y.append(data[i + self.lookback])
        return np.array(X), np.array(y)

    def fit(self, series, epochs=50, batch_size=32):
        data = series.values.reshape(-1, 1)
        X, y = self._create_sequences(data)
        self.model.fit(X, y, epochs=epochs, batch_size=batch_size, verbose=0)

    def predict(self, series, steps=7):
        predictions = []
        current = series.values[-self.lookback:].reshape(1, self.lookback, 1)

        for _ in range(steps):
            next_value = self.model.predict(current, verbose=0)[0][0]
            predictions.append(next_value)
            current = np.roll(current, -1)
            current[0, -1, 0] = next_value

        return predictions
```

## 4. 模型集成

### 4.1 集成预测

```python
class EnsemblePredictor:
    def __init__(self):
        self.predictors = [
            ARIMAPredictor(),
            ESPredictor(),
            GBRTTPredictor(),
            LSTMPredictor(),
        ]
        self.weights = [0.2, 0.2, 0.3, 0.3]

    def fit(self, series):
        for predictor in self.predictors:
            predictor.fit(series)

    def predict(self, series, steps=7):
        predictions = []
        for predictor in self.predictors:
            pred = predictor.predict(series, steps)
            predictions.append(pred)

        # 加权平均
        ensemble = np.zeros(steps)
        for pred, weight in zip(predictions, self.weights):
            ensemble += np.array(pred) * weight

        return ensemble
```

## 5. 模型评估

### 5.1 评估指标

```python
class TimeSeriesEvaluator:
    def evaluate(self, actual, predicted):
        return {
            'mae': self._mae(actual, predicted),
            'mse': self._mse(actual, predicted),
            'rmse': self._rmse(actual, predicted),
            'mape': self._mape(actual, predicted),
            'r2': self._r2(actual, predicted),
        }

    def _mae(self, actual, predicted):
        return np.mean(np.abs(actual - predicted))

    def _mse(self, actual, predicted):
        return np.mean((actual - predicted) ** 2)

    def _rmse(self, actual, predicted):
        return np.sqrt(self._mse(actual, predicted))

    def _mape(self, actual, predicted):
        return np.mean(np.abs((actual - predicted) / actual)) * 100

    def _r2(self, actual, predicted):
        ss_res = np.sum((actual - predicted) ** 2)
        ss_tot = np.sum((actual - np.mean(actual)) ** 2)
        return 1 - ss_res / ss_tot
```

### 6.2 异常预警

```python
class AnomalyPredictor:
    def predict_anomaly_probability(self, series, threshold):
        # 预测未来值
        predictions = self.model.predict(series, steps=7)

        # 计算超过阈值的概率
        probabilities = []
        for pred in predictions:
            if pred > threshold:
                prob = min(1.0, (pred - threshold) / threshold + 0.5)
            else:
                prob = max(0.0, 0.5 - (threshold - pred) / threshold)
            probabilities.append(prob)

        return probabilities
```
