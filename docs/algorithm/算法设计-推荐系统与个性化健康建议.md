## 3. 推荐算法

### 3.1 协同过滤

```python
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

class CollaborativeFiltering:
    def __init__(self):
        self.user_item_matrix = None
        self.user_similarity = None
        self.item_similarity = None

    def fit(self, interactions: list[dict]):
        # 构建用户-物品矩阵
        users = list(set(i['user_id'] for i in interactions))
        items = list(set(i['item_id'] for i in interactions))

        self.user_idx = {u: i for i, u in enumerate(users)}
        self.item_idx = {it: i for i, it in enumerate(items)}

        matrix = np.zeros((len(users), len(items)))
        for inter in interactions:
            u = self.user_idx[inter['user_id']]
            it = self.item_idx[inter['item_id']]
            matrix[u][it] = inter.get('weight', 1)

        self.user_item_matrix = matrix
        self.user_similarity = cosine_similarity(matrix)
        self.item_similarity = cosine_similarity(matrix.T)

    def recommend_user_based(self, user_id: str, top_n: int = 10) -> list:
        if user_id not in self.user_idx:
            return []

        user_idx = self.user_idx[user_id]
        similar_users = np.argsort(self.user_similarity[user_idx])[::-1][1:11]

        scores = np.zeros(self.user_item_matrix.shape[1])
        for sim_user in similar_users:
            sim = self.user_similarity[user_idx][sim_user]
            scores += sim * self.user_item_matrix[sim_user]

        # 排除已交互的物品
        interacted = np.where(self.user_item_matrix[user_idx] > 0)[0]
        scores[interacted] = -1

        top_items = np.argsort(scores)[::-1][:top_n]
        return [
            {'item_id': list(self.item_idx.keys())[i], 'score': scores[i]}
            for i in top_items if scores[i] > 0
        ]

    def recommend_item_based(self, item_id: str, top_n: int = 10) -> list:
        if item_id not in self.item_idx:
            return []

        item_idx = self.item_idx[item_id]
        similar_items = np.argsort(self.item_similarity[item_idx])[::-1][1:top_n+1]

        return [
            {'item_id': list(self.item_idx.keys())[i],
             'similarity': self.item_similarity[item_idx][i]}
            for i in similar_items
        ]
```

### 3.2 基于内容推荐

```python
class ContentBasedRecommender:
    def __init__(self):
        self.tfidf = None
        self.item_features = None
        self.item_ids = []

    def fit(self, items: list[dict]):
        from sklearn.feature_extraction.text import TfidfVectorizer

        self.item_ids = [item['id'] for item in items]
        contents = [item.get('description', '') + ' ' + item.get('tags', '') for item in items]

        self.tfidf = TfidfVectorizer(max_features=1000)
        self.item_features = self.tfidf.fit_transform(contents)

    def recommend(self, user_profile: UserProfile, top_n: int = 10) -> list:
        # 构建用户偏好向量
        user_text = ' '.join(user_profile.preferences.get('tags', []))
        user_vector = self.tfidf.transform([user_text])

        # 计算相似度
        similarities = cosine_similarity(user_vector, self.item_features)[0]

        top_indices = np.argsort(similarities)[::-1][:top_n]
        return [
            {'item_id': self.item_ids[i], 'score': similarities[i]}
            for i in top_indices if similarities[i] > 0
        ]
```

### 3.3 混合推荐

```python
class HybridRecommender:
    def __init__(self, cf: CollaborativeFiltering, cbr: ContentBasedRecommender):
        self.cf = cf
        self.cbr = cbr
        self.weights = {'cf': 0.6, 'cbr': 0.4}

    def recommend(self, user_id: str, user_profile: UserProfile, top_n: int = 10) -> list:
        # 协同过滤推荐
        cf_results = self.cf.recommend_user_based(user_id, top_n * 2)
        cf_scores = {r['item_id']: r['score'] for r in cf_results}

        # 基于内容推荐
        cbr_results = self.cbr.recommend(user_profile, top_n * 2)
        cbr_scores = {r['item_id']: r['score'] for r in cbr_results}

        # 合并结果
        all_items = set(cf_scores.keys()) | set(cbr_scores.keys())
        combined = []
        for item_id in all_items:
            cf_score = cf_scores.get(item_id, 0)
            cbr_score = cbr_scores.get(item_id, 0)
            total = self.weights['cf'] * cf_score + self.weights['cbr'] * cbr_score
            combined.append({'item_id': item_id, 'score': total})

        combined.sort(key=lambda x: x['score'], reverse=True)
        return combined[:top_n]
```

```python
class PopularRecommender:
    @staticmethod
    def get_trending(items: list[dict], top_n: int = 10) -> list:
        # 按近期热度排序
        scored = []
        for item in items:
            score = (
                item.get('views', 0) * 0.3 +
                item.get('likes', 0) * 0.4 +
                item.get('shares', 0) * 0.3
            )
            # 时间衰减
            age_days = (datetime.now() - item['created_at']).days
            decay = 0.95 ** age_days
            scored.append({'item': item, 'score': score * decay})

        scored.sort(key=lambda x: x['score'], reverse=True)
        return [s['item'] for s in scored[:top_n]]
```

## 6. 效果评估

### 6.1 评估指标

```python
class RecommendationEvaluator:
    @staticmethod
    def precision_at_k(recommended: list, actual: list, k: int) -> float:
        recommended_k = recommended[:k]
        hits = len(set(recommended_k) & set(actual))
        return hits / k if k > 0 else 0

    @staticmethod
    def recall_at_k(recommended: list, actual: list, k: int) -> float:
        recommended_k = recommended[:k]
        hits = len(set(recommended_k) & set(actual))
        return hits / len(actual) if actual else 0

    @staticmethod
    def ndcg_at_k(recommended: list, actual: list, k: int) -> float:
        dcg = 0
        for i, item in enumerate(recommended[:k]):
            if item in actual:
                dcg += 1 / np.log2(i + 2)

        ideal = sorted([1] * len(actual), reverse=True)
        idcg = sum(1 / np.log2(i + 2) for i in range(min(k, len(actual))))

        return dcg / idcg if idcg > 0 else 0

    @staticmethod
    def coverage(recommended_lists: list[list], all_items: set) -> float:
        recommended_items = set()
        for recs in recommended_lists:
            recommended_items.update(recs)
        return len(recommended_items) / len(all_items) if all_items else 0
```
